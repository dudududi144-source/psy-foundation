   340→  // ── Audio init (EXACTLY like psy) ──
   341→  private ensureAudio(): void {
   342→    if (this.ctx) { if (this.ctx.state === 'suspended') this.ctx.resume(); return; }
   343→    this.ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
   344→
   345→    // Simple chain: voices → master → safetyLimiter → analyser → destination
   346→    // R6: Safety limiter added to prevent clipping (engine + radio sum)
   347→    this.master = this.ctx.createGain();
   348→    this.master.gain.value = 0.9;
   349→    this.analyser = this.ctx.createAnalyser();
   350→    this.analyser.fftSize = 512;
   351→    this.analyser.smoothingTimeConstant = 0.7;
   352→    // R6: Safety limiter — brickwall-style, only activates on peaks near 0dBFS
   353→    this.safetyLimiter = this.ctx.createDynamicsCompressor();
   354→    this.safetyLimiter.threshold.value = -1.0;   // only catches peaks above -1dB
   355→    this.safetyLimiter.knee.value = 0;            // hard knee
   356→    this.safetyLimiter.ratio.value = 20;          // 20:1 = brickwall-ish
   357→    this.safetyLimiter.attack.value = 0.003;      // 3ms (fast)
   358→    this.safetyLimiter.release.value = 0.05;      // 50ms
   359→    this.master.connect(this.safetyLimiter);
   360→    this.safetyLimiter.connect(this.analyser);
   361→    this.analyser.connect(this.ctx.destination);
   362→
   363→    // Delay (like psy)
   364→    this.delaySend = this.ctx.createGain();
   365→    this.delaySend.gain.value = 1.0;
   366→    this.delay = this.ctx.createDelay(2.0);
   367→    this.delay.delayTime.value = 0.3;
   368→    const wet = this.ctx.createGain(); wet.gain.value = 0.22;
   369→    const fb = this.ctx.createGain(); fb.gain.value = 0.34;
   370→    this.delaySend.connect(this.delay);
   371→    this.delay.connect(wet); wet.connect(this.master);
   372→    this.delay.connect(fb); fb.connect(this.delay);
   373→
   374→    // Noise buffer for hats
   375→    const len = Math.floor(this.ctx.sampleRate * 0.25);
   376→    this.noiseBuf = this.ctx.createBuffer(1, len, this.ctx.sampleRate);
   377→    const nd = this.noiseBuf.getChannelData(0);
   378→    for (let i = 0; i < len; i++) nd[i] = Math.random() * 2 - 1;
   379→
   380→    // Load learning
   381→    this.learningData = loadLearning();
   382→
   383→    // F1.18 — Initialize MusicalTransport (the SINGLE source of truth for musical time)
   384→    // All beat/bar/phase/bpm reads come from transport.snapshot().
   385→    // The PLL is an observer that feeds observations to Transport.
   386→    this.transport = new MusicalTransport(() => this.ctx!.currentTime, {
   387→      initialBpm: PRESETS[0].bpm,
   388→    });
   389→    this.transportAdapter = new TransportAdapter(this.transport);
   390→
   391→    // F2.5 — Initialize RadioObservationLayer
   392→    // The SINGLE entry point for radio analysis. Produces timestamped
   393→    // RadioBeatObservation that feeds Transport.observeBeat().
   394→    this.radioLayer = new RadioObservationLayer({
   395→      ...DEFAULT_RADIO_CONFIG,
   396→      sampleRate: this.ctx.sampleRate,
   397→      fftSize: this.radioAnalyser?.fftSize ?? 512,
   398→    });
   399→
   400→    // ── PER-ROLE BUSES (from architecture review) ──
   401→    // Each voice connects to its role bus → engineBus → gentle comp → master
   402→    this.kickBus = this.ctx.createGain(); this.kickBus.gain.value = 0.9;
   403→    this.bassBus = this.ctx.createGain(); this.bassBus.gain.value = 0.85;
   404→    this.leadBus = this.ctx.createGain(); this.leadBus.gain.value = 0.7;
   405→    this.hatBus = this.ctx.createGain(); this.hatBus.gain.value = 0.6;
   406→    
   407→    this.engineBus = this.ctx.createGain();
   408→    this.engineBus.gain.value = 0.8;
   409→
   410→    // Gentle compressor on engine bus only (not on radio)
   411→    this.comp = this.ctx.createDynamicsCompressor();
   412→    this.comp.threshold.value = -18;
   413→    this.comp.knee.value = 18;
   414→    this.comp.ratio.value = 2;
   415→    this.comp.attack.value = 0.015;
   416→    this.comp.release.value = 0.12;
   417→
   418→    // Connect: role buses → engineBus → comp → master
   419→    this.kickBus.connect(this.engineBus);
   420→    this.bassBus.connect(this.engineBus);
   421→    this.leadBus.connect(this.engineBus);
   422→    this.hatBus.connect(this.engineBus);
   423→    this.engineBus.connect(this.comp);
   424→    this.comp.connect(this.master);
   425→  }

   428→  private kick(t: number): void {
   429→    if (!this.ctx || !this.kickBus) return;
   430→    const osc = this.ctx.createOscillator(), gain = this.ctx.createGain();
   431→    osc.type = 'sine';
   432→    osc.frequency.setValueAtTime(150, t);
   433→    osc.frequency.exponentialRampToValueAtTime(44, t + 0.09);
   434→    gain.gain.setValueAtTime(1.0, t);
   435→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.3);
   436→    osc.connect(gain); gain.connect(this.kickBus);
   437→    osc.start(t); osc.stop(t + 0.32);
   438→  }
   439→
   440→  private hat(t: number, lvl: number): void {
   441→    if (!this.ctx || !this.hatBus || !this.noiseBuf) return;
   442→    const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf;
   443→    const hp = this.ctx.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 7000;
   444→    const gain = this.ctx.createGain();
   445→    gain.gain.setValueAtTime(Math.max(0.001, lvl), t);
   446→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.05);
   447→    src.connect(hp); hp.connect(gain); gain.connect(this.hatBus);
   448→    src.start(t); src.stop(t + 0.06);
   449→  }

   451→  private bass(t: number, freq: number, v: Variant): void {
   452→    if (!this.ctx || !this.bassBus) return;
   453→    const osc = this.ctx.createOscillator(); osc.type = v.bassWave; osc.frequency.value = freq;
   454→    const filter = this.ctx.createBiquadFilter(); filter.type = 'lowpass'; filter.Q.value = v.bassQ;
   455→    const fStart = Math.max(60, v.bassCut), fEnd = Math.max(80, v.bassCut * 0.35);
   456→    filter.frequency.setValueAtTime(fStart, t);
   457→    filter.frequency.exponentialRampToValueAtTime(fEnd, t + 0.16);
   458→    const gain = this.ctx.createGain();
   459→    gain.gain.setValueAtTime(0.0001, t);
   460→    gain.gain.exponentialRampToValueAtTime(0.85, t + 0.006);
   461→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.2);
   462→    osc.connect(filter); filter.connect(gain); gain.connect(this.bassBus);
   463→    if (this.delaySend) { const send = this.ctx.createGain(); send.gain.value = 0.12; gain.connect(send); send.connect(this.delaySend); }
   464→    osc.start(t); osc.stop(t + 0.22);
   465→  }

   467→  private lead(t: number, freq: number, v: Variant, accent: boolean): void {
   468→    if (!this.ctx || !this.leadBus) return;
   469→    const peakCut = Math.max(200, v.leadCut * (accent ? 1.25 : 1));
   470→    const o1 = this.ctx.createOscillator(), o2 = this.ctx.createOscillator();
   471→    o1.type = v.leadWave; o2.type = v.leadWave;
   472→    o1.frequency.value = freq; o2.frequency.value = freq * Math.pow(2, 9 / 1200);
   473→    const filter = this.ctx.createBiquadFilter(); filter.type = 'lowpass'; filter.Q.value = v.leadQ;
   474→    filter.frequency.setValueAtTime(180, t);
   475→    filter.frequency.exponentialRampToValueAtTime(peakCut, t + 0.02);
   476→    filter.frequency.exponentialRampToValueAtTime(240, t + 0.22);
   477→    const gain = this.ctx.createGain();
   478→    const peak = Math.max(0.05, v.leadLvl * (accent ? 1 : 0.7));
   479→    gain.gain.setValueAtTime(0.0001, t);
   480→    gain.gain.exponentialRampToValueAtTime(peak, t + 0.008);
   481→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.24);
   482→    o1.connect(filter); o2.connect(filter); filter.connect(gain); gain.connect(this.leadBus);
   483→    if (this.delaySend) { const send = this.ctx.createGain(); send.gain.value = 0.3; gain.connect(send); send.connect(this.delaySend); }
   484→    o1.start(t); o2.start(t); o1.stop(t + 0.26); o2.stop(t + 0.26);
   485→  }

   547→  // ── Scheduler — reads Transport for ALL musical time ──
   548→  // F1.18: setInterval wakes the scheduler (25ms). The scheduler reads
   549→  // transport.snapshot() to get the beat grid, then schedules 16th notes
   550→  // directly. NO independent nextNoteTime, step, or barCount.
   551→  //
   552→  // PLAYBACK REALITY FIX: The previous version called predictBeats(0.15)
   553→  // which only returned BEAT boundaries within 150ms. At 145 BPM, beats
   554→  // are 414ms apart — so predictBeats returned an EMPTY ARRAY most ticks,
   555→  // causing silence. The fix: compute 16th-note times directly from the
   556→  // Transport's beat grid (beatTime + k * stepDur), not from beat boundaries.
   557→  //
   558→  // Policy for tab suspension: DROP STALE EVENTS.
   559→  private scheduler(): void {
   560→    if (!this.ctx || !this.transport) return;
   561→    try {
   562→      const now = this.ctx.currentTime;
   563→      const snap = this.transport.snapshot();
   564→      const stepDur = snap.beatDuration / 4; // 16th note duration
   565→
   566→      // Compute the next 16th-note step from the Transport's beat grid.
   567→      // snap.beatTime = AudioContext time of the most recent beat boundary.
   568→      // 16th notes are at: beatTime + k * stepDur (k = 0,1,2,3 within each beat).
   569→      const elapsedSinceBeat = now - snap.beatTime;
   570→      const stepsSinceBeat = Math.floor(elapsedSinceBeat / stepDur);
   571→
   572→      // Next 16th note to schedule (one step ahead of current position)
   573→      let stepTime = snap.beatTime + (stepsSinceBeat + 1) * stepDur;
   574→      let stepIdx = snap.beatIndex * 4 + stepsSinceBeat + 1;
   575→
   576→      // Schedule all 16th notes within the schedule-ahead window
   577→      while (stepTime < now + this.scheduleAheadTime) {
   578→        if (stepTime > now && stepIdx > this.lastScheduledBeatIndex) {
   579→          this.scheduleStep(stepIdx, stepTime);
   580→          this.lastScheduledBeatIndex = stepIdx;
   581→        }
   582→        stepIdx++;
   583→        stepTime += stepDur;
   584→      }
   585→    } catch (e) {}
   586→  }
   587→
   588→  // F1.18: scheduleStep receives the GLOBAL step index from Transport.
   589→  // The 16-step pattern position is derived: step16 = stepIndex % 16.
   590→  // Bar position comes from Transport: snap.bar.
   591→  private scheduleStep(stepIndex: number, time: number): void {
   592→    if (!this.transport) return;
   593→    const snap = this.transport.snapshot();
   594→    const p = this.getPreset();
   595→    const v = this.getVariant();
   596→    const s16 = stepIndex % 16;
   597→
   598→    // ── PATTERN MUTATION: evolve every 8 bars ──
   599→    // F1.18: bar comes from Transport snapshot, not from barCount++
   600→    if (s16 === 0) {
   601→      const currentBar = snap.bar;
   602→      if (currentBar !== this.lastMutatedBar && currentBar % 8 === 0 && currentBar > 0) {
   603→        const basePattern = this.livePattern || p.patterns;
   604→        const mutated = mutatePattern(basePattern, this.occupancy, this.musicState.density);
   605→        if (mutated) {
   606→          this.livePattern = mutated;
   607→          console.log('[MUTATE] pattern evolved, bar', currentBar);
   608→        }
   609→        this.lastMutatedBar = currentBar;
   610→      }
   611→    }
   612→
   613→    // Use live pattern if available, else preset pattern
   614→    const pat = this.livePattern || p.patterns;
   615→
   616→    // Use detected root if locked, else preset root
   617→    const root = this.harmonicLocked && this.harmonicRoot ? this.harmonicRoot : p.root;
   618→
   619→    // ── ARRANGER: decide what to play based on occupancy ──
   620→    const kickAvailable = this.occupancy.kick < 0.7;
   621→    const bassAvailable = this.occupancy.bass < 0.75;
   622→    const leadAvailable = this.occupancy.lead < 0.85;
   623→    const density = this.musicState.density;
   624→
   625→    // Kick: only if role is available AND density allows
   626→    if (kickAvailable && pat.kick && pat.kick[s16] && Math.random() < density) {
   627→      this.kick(time);
   628→    }
   629→    // Hats: always available (fills high end)
   630→    if (pat.hat && pat.hat[s16]) this.hat(time, v.hatLvl || 0.1);
   631→
   632→    // Bass: only if role is available
   633→    if (bassAvailable) {
   634→      const bn = pat.bass ? pat.bass[s16] : null;
   635→      if (bn !== null && bn !== undefined) this.bass(time, mtof(root + bn), v);
   636→    }
   637→
   638→    // Lead: only if role is available AND density is high enough
   639→    if (leadAvailable && density > 0.4) {
   640→      const ln = pat.lead ? pat.lead[s16] : null;
   641→      if (ln !== null && ln !== undefined) this.lead(time, mtof(root + 24 + ln), v, s16 % 4 === 0);
   642→    }
   643→  }

   689→      this.radioSource.connect(this.radioGain);
   690→      this.radioGain.connect(this.radioAnalyser);
   691→      this.radioAnalyser.connect(this.master!);
   692→
   693→      // R3: RadioStateGate — mark connecting BEFORE play()

   700→      this.radioOn = true;
   701→      this.updateMixMode();
   702→      this.startDetection();
   703→      this.emit();
   704→      return true;
   705→    } catch (e) { console.error(e); return false; }

   736→  private detect(): void {
   737→    if (!this.radioAnalyser || !this.ctx || !this.radioLayer) return;
   738→    if (!this.radioFreqBuf || this.radioFreqBuf.length !== this.radioAnalyser.frequencyBinCount) {
   739→      this.radioFreqBuf = new Uint8Array(this.radioAnalyser.frequencyBinCount);
   740→    }
   741→    const fd = this.radioFreqBuf;
   742→    this.radioAnalyser.getByteFrequencyData(fd);
   743→
   744→    // F2.5 — Get time-domain data for RadioObservationLayer
   745→    const tdBuf = this.melodyObserver.ensureTimeDomainBuf(this.radioAnalyser);
   746→    this.radioAnalyser.getFloatTimeDomainData(tdBuf);
   747→
   748→    // F2.5 — Process through RadioObservationLayer (the SINGLE entry point)
   749→    // This replaces: RadioStateGate, inline beat detection, inline pitch observation
   750→    const audioTime = this.ctx.currentTime;
   751→    const radioSnap = this.radioLayer.process(tdBuf, fd, audioTime);
   752→
   753→    // F2.5 — Feed beat observations to Transport (the ONLY crossing point)
   754→    // RadioObservationLayer produces timestamped RadioBeatObservation.
   755→    // Only { time, confidence, source } crosses into Transport.
   756→    if (radioSnap.beat) {
   757→      this.transport!.observeBeat({
   758→        time: radioSnap.beat.timestamp.observedAt,
   759→        confidence: radioSnap.beat.confidence,
   760→        source: 'radio',
   761→      });
   762→      this.kickCount++;
   763→    }
   764→
   765→    // F2.5 — Update syncStatus from observation state (not loudness)
   766→    if (this.radioOn) {
   767→      const obsState = radioSnap.signal.observationState;
   768→      if (obsState === 'FOLLOWING') {
   769→        this.syncStatus = 'following';
   770→      } else if (obsState === 'LOCKING') {
   771→        this.syncStatus = 'listening';
   772→      } else if (obsState === 'SIGNAL_PRESENT') {
   773→        this.syncStatus = 'listening';
   774→      } else if (obsState === 'DEGRADED') {
   775→        this.syncStatus = 'listening'; // still listening, but degraded
   776→      } else if (obsState === 'LOST' || obsState === 'NO_SIGNAL') {
   777→        this.syncStatus = 'no_signal';
   778→      }
   779→    }
   780→
   781→    // F2.5 — Update occupancy from radio layer (for arranger decisions)
   782→    this.occupancy = radioSnap.occupancy;
   783→
   784→    // Update radio level for UI
   785→    this.radioLevel = radioSnap.signal.spectralEnergy;
   786→    this.radioRms = this.radioRms * 0.85 + radioSnap.signal.rms * 0.15;
   787→    this.radioBands = {
   788→      low: radioSnap.occupancy.kick,
   789→      mid: radioSnap.occupancy.lead,
   790→      high: radioSnap.occupancy.hats,
   791→    };
   792→
   793→    // ── ROLE DUCKING (based on occupancy from radio layer) ──
   794→    if (this.kickBus && this.bassBus && this.leadBus && this.hatBus && this.ctx) {
   795→      const now = this.ctx.currentTime;
   796→      const kickGain = this.occupancy.kick > 0.7 ? 0.05 : 0.9;
   797→      this.kickBus.gain.setTargetAtTime(kickGain, now, 0.03);
   798→      const bassGain = this.occupancy.bass > 0.75 ? 0.35 : 0.85;
   799→      this.bassBus.gain.setTargetAtTime(bassGain, now, 0.05);
   800→      const leadGain = this.occupancy.lead > 0.85 ? 0.55 : 0.7;
   801→      this.leadBus.gain.setTargetAtTime(leadGain, now, 0.08);
   802→      this.hatBus.gain.setTargetAtTime(0.6, now, 0.08);
   803→    }
   804→
   805→    // ── ENERGY HISTORY (for relative energy, not absolute) ──
   806→    this.energyHistory.push(radioSnap.signal.spectralEnergy);
   807→    if (this.energyHistory.length > 32) this.energyHistory.shift();
   808→
   809→    // ── MUSIC STATE UPDATE ──
   810→    if (this.energyHistory.length >= 8) {
   811→      const recent = this.energyHistory.slice(-4).reduce((a, b) => a + b, 0) / 4;
   812→      const older = this.energyHistory.slice(-8, -4).reduce((a, b) => a + b, 0) / 4;
   813→      this.musicState.energy = recent;
   814→      this.musicState.energySlope = recent - older;
   815→    }
   816→
   817→    this.musicState.radioRoles = { ...this.occupancy };
   818→    this.musicState.bpm = this.transport ? this.transport.snapshot().bpm : 145;
   819→
   820→    // ── STYLE DETECTION (with hysteresis, using AudioContext time) ──
   821→    const detectedStyle = this.classifyStyle();
   822→    if (detectedStyle) {
   823→      const audioNow = this.ctx.currentTime;
   824→      if (detectedStyle !== this.styleCandidate) {
   825→        this.styleCandidate = detectedStyle;
   826→        this.styleCandidateSince = audioNow;
   827→      }
   828→      if (this.styleCandidate && audioNow - this.styleCandidateSince > 8) {
   829→        if (this.styleCandidate !== this.currentStyle) {
   830→          this.currentStyle = this.styleCandidate;
   831→        }
   832→      }
   833→    }
   834→    this.musicState.style = this.currentStyle;
   835→
   836→    // ── COMPETITIVE DENSITY CONTROL ──
   837→    const delta = this.musicState.energySlope;
   838→    if (delta > 0.18) {
   839→      this.musicState.density = Math.max(0.3, this.musicState.density * 0.75);
   840→    } else if (delta < -0.18) {
   841→      this.musicState.density = Math.min(0.9, this.musicState.density * 1.15);
   842→    } else {
   843→      this.musicState.density += (0.7 - this.musicState.density) * 0.05;
   844→    }
   845→
   846→    // ── LEARNING (record kicks when locked) ──
   847→    const transportSnap = this.transport!.snapshot();
   848→    if (transportSnap.locked && radioSnap.beat) {
   849→      if (this.learningData) {
   850→        this.learningData = recordKick(this.learningData, Math.round(transportSnap.bpm));
   851→        this.learningData = deriveInsights(this.learningData);
   852→        saveLearning(this.learningData);
   853→      }
   854→      this.updateDelayTime();
   855→      this.updateMixMode();
   856→    }
   857→
   858→    // Engine level
   859→    if (this.analyser) {
   860→      const d = new Uint8Array(this.analyser.frequencyBinCount);
   861→      this.analyser.getByteFrequencyData(d);
   862→      let s = 0; for (let i = 0; i < d.length; i++) s += d[i];
   863→      this.engineLevel = s / (d.length * 255);
   864→    }
   865→
   866→    this.emit();
   867→  }

   881→  // ── Style classifier ──
   882→  // F1.18: BPM comes from Transport — single source of truth
   883→  private classifyStyle(): Style | null {
   884→    const o = this.occupancy;
   885→    const bpm = this.transport ? this.transport.snapshot().bpm : 145;
   886→
   887→    // Full On: high kick + high bass + high highs + fast
   888→    if (o.kick > 0.7 && o.bass > 0.6 && o.hats > 0.4 && bpm > 143) {
   889→      return 'fullOn';
   890→    }
   891→    // Dark: high bass + low highs + slow
   892→    if (o.bass > 0.6 && o.hats < 0.3 && bpm < 142) {
   893→      return 'dark';
   894→    }
   895→    // Progressive: moderate everything + stable energy
   896→    if (o.kick < 0.6 && o.bass > 0.4 && this.musicState.energySlope < 0.1) {
   897→      return 'progressive';
   898→    }
   899→    // Acid: mid-heavy + high energy
   900→    if (o.lead > 0.6 && this.musicState.energy > 0.5) {
   901→      return 'acid';
   902→    }
   903→    return null;
   904→  }
