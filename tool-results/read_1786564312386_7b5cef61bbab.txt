    32→function fullOnContext(overrides: Partial<MusicalContext> = {}): MusicalContext {
    33→  return baseContextForStyle('full-on', overrides)
    34→}
    35→
    36→function adaptAt(
    37→  adaptation: CompositionAdaptation,
    38→  baseContext: MusicalContext,
    39→  radio: RadioMusicalContext,
    40→  phraseBar = 4
    41→): AdaptedCompositionIntent {
    42→  const opportunities = buildOpportunityMap(radio)
    43→  return adaptation.adapt({
    44→    baseContext,
    45→    radio,
    46→    opportunities,
    47→    currentBar: 0,
    48→    phraseBar,
    49→  })
    50→}

    52→function allPressuresIn01(intent: AdaptedCompositionIntent): boolean {
    53→  const vals = [
    54→    intent.groovePressure,
    55→    intent.bassPressure,
    56→    intent.leadPressure,
    57→    intent.counterPressure,
    58→    intent.texturePressure,
    59→    intent.densityTarget,
    60→    intent.tensionTarget,
    61→    intent.noveltyTarget,
    62→    intent.restPressure,
    63→    intent.confidence,
    64→  ]
    65→  return vals.every((v) => v >= 0 && v <= 1 && !Number.isNaN(v))
    66→}

    68→function composeBase(
    69→  styleName: string,
    70→  seed: number,
    71→  bars: number
    72→): { section: ComposedSection; baseContext: MusicalContext; engine: CompositionEngine } {
    73→  const baseContext = baseContextForStyle(styleName)
    74→  const engine = new CompositionEngine({ seed, context: baseContext })
    75→  const section = engine.composeSection({ bars })
    76→  return { section, baseContext, engine }
    77→}

   451→// ====================== 16. Negative learning ======================
   452→
   453→describe('Negative learning', () => {
   454→  test('bass-heavy + competing bass → next time less bass competition', () => {
   455→    const base = fullOnContext()
   456→    const { section } = composeBase('full-on', 11, 32)
   457→
   458→    // Use a mid-confidence BASS_HEAVY radio so the first adaptation keeps
   459→    // some bass notes (blend = 0.5 → bassPressure ~0.45, above the 0.4
   460→    // silence threshold). This produces measurable bass competition that
   461→    // the learning loop can then penalise.
   462→    const radio = createRadioContext({
   463→      bpm: 145,
   464→      bpmConfidence: 0.8,
   465→      key: 4,
   466→      scale: 'phrygian-dominant',
   467→      keyConfidence: 0.75,
   468→      energy: 0.7,
   469→      density: 0.6,
   470→      energyConfidence: 0.7,
   471→      kickOccupancy: 0.5,
   472→      bassOccupancy: 0.85,
   473→      percussionOccupancy: 0.4,
   474→      leadOccupancy: 0.45,
   475→      harmonicOccupancy: 0.3,
   476→      syncopation: 0.3,
   477→      style: 'full-on',
   478→      styleConfidence: 0.7,
   479→      sectionLikelihood: 'GROOVE',
   480→      confidence: 0.5, // mid confidence → blend = 0.5
   481→      available: true,
   482→    })
   483→
   484→    // Run 1: no learning bias.
   485→    const adaptation = new CompositionAdaptation()
   486→    const intent1 = adaptAt(adaptation, base, radio)
   487→    // Mid-confidence BASS_HEAVY → bassPressure ~0.45 (not silenced).
   488→    expect(intent1.bassPressure).toBeGreaterThan(0.4)
   489→    const adapted1 = applyAdaptation(section.bars, intent1)
   490→    const competition1 = bassCompetition({ bars: adapted1 }, radio)
   491→    expect(competition1).toBeGreaterThan(0)
   492→
   493→    // Negative outcome: foundation's bass competed with radio's bass.
   494→    adaptation.reinforce('bass', false)
   495→    expect(adaptation.biasFor('bass')).toBeLessThan(0)
   496→
   497→    // Run 2: with learned bias.
   498→    const intent2 = adaptAt(adaptation, base, radio)
   499→    const adapted2 = applyAdaptation(section.bars, intent2)
   500→    const competition2 = bassCompetition({ bars: adapted2 }, radio)
   501→
   502→    expect(intent2.bassPressure).toBeLessThan(intent1.bassPressure)
   503→    expect(competition2).toBeLessThanOrEqual(competition1)
   504→  })
   505→})
