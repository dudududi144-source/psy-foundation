   664→   664→/**
   665→   665→ * Auto-select presets by detected scale + genre + energy.
   666→   666→ * Used by the learning system to pick the best sounds for the current context.
   667→   667→ */
   668→   668→export function autoSelect(
   669→   669→  scaleDegrees: number[],
   670→   670→  genre: Genre = 'PSYTRANCE',
   671→   671→  energy: number = 0.7,
   672→   672→  cat?: Category,
   673→   673→): SoundPreset[] {
   674→   674→  let candidates = SOUND_BANK;
   675→   675→  if (cat) candidates = candidates.filter(p => p.cat === cat);
   676→   676→  // Prefer matching genre, fall back to ANY
   677→   677→  const genreMatch = candidates.filter(p => p.genre === genre || p.genre === 'ANY');
   678→   678→  candidates = genreMatch.length > 0 ? genreMatch : candidates;
   679→   679→  // Filter by energy (±0.2 tolerance)
   680→   680→  candidates = candidates.filter(p => Math.abs((p.energyLevel || 0.5) - energy) < 0.25);
   681→   681→  // Sort by how many scale degrees match
   682→   682→  candidates.sort((a, b) => {
   683→   683→    const aMatch = (a.scaleDegrees || []).filter(d => scaleDegrees.includes(d)).length;
   684→   684→    const bMatch = (b.scaleDegrees || []).filter(d => scaleDegrees.includes(d)).length;
   685→   685→    return bMatch - aMatch;
   686→   686→  });
   687→   687→  return candidates.slice(0, 5);
   688→   688→}

   690→   690→export function bankStats() {
   691→   691→  const cats: Record<string, number> = {};
   692→   692→  const genres: Record<string, number> = {};
   693→   693→  SOUND_BANK.forEach(p => {
   694→   694→    cats[p.cat] = (cats[p.cat] || 0) + 1;
   695→   695→    genres[p.genre] = (genres[p.genre] || 0) + 1;
   696→   696→  });
   697→   697→  return { total: SOUND_BANK.length, cats, genres };
   698→   698→}
