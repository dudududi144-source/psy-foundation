     1→/**
     2→ * Continuous Trainer — runs entirely client-side.
     3→ *
     4→ * This solves the Cloudflare edge runtime limitation (server can't run heavy DSP).
     5→ * The trainer runs in the browser, periodically:
     6→ *   1. Renders a short snippet with current params (deterministic, offline)
     7→ *   2. Analyzes it (FFT, dynamics, spectrum)
     8→ *   3. Compares to the live reference profile
     9→ *   4. Adjusts 1-3 parameters
    10→ *   5. Applies the new params to the LIVE engine (real-time)
    11→ *   6. Repeats
    12→ *
    13→ * This enables CONTINUOUS LEARNING while radio + engine play together.
    14→ *
    15→ * The learned params are saved to localStorage for future sessions.
    16→ */
    17→

    44→  running: boolean;
    45→  iterations: LearningIteration[];

    48→  currentParams: Record<string, number>;
    49→  bestParams: Record<string, number>;
    50→  acceptedCount: number;
    51→  rejectedCount: number;
    52→  totalIterations: number;
    53→  learnedKnowledge: Record<string, { value: number; score: number; attempts: number }>;
    54→  // Per-voice scores (from perVoiceAnalyzer)
    55→  perVoiceScores?: {
    56→    kick: number;
    57→    bass: number;
    58→    lead: number;
    59→    spectral: number;
    60→  };
    61→  // Per-voice analysis details
    62→  voiceAnalysis?: {
    63→    kick: { lufs: number; decayMs: number; active: boolean };
    64→    bass: { lufs: number; bandEnergy: number; active: boolean };
    65→    lead: { lufs: number; centroidHz: number; active: boolean };
    66→    hat: { lufs: number; active: boolean };
    67→  };
    68→}

   111→
   112→  constructor(config?: Partial<ContinuousTrainerConfig>) {
   113→    this.config = { ...DEFAULT_CONFIG, ...config };
   114→    const dna = getWorldDNA(this.config.worldId);
   115→    const defaults = dna ? {
   116→      kickDecay: dna.kickDecayTarget,
   117→      kickFundamental: dna.kickFundamentalTarget,

   119→      bassResonance: dna.bassResonanceTarget,
   120→      leadCutoff: dna.leadCutoffTarget,
   121→      leadDetune: dna.leadDetuneTarget,
   122→      padCutoff: 1200,
   123→      duck: 0.4,
   124→    } : {};
   125→
   126→    // Load saved params from localStorage
   127→    let savedParams: Record<string, number> = {};
   128→    if (this.config.saveToLocalStorage && typeof window !== 'undefined') {
   129→      try {
   130→        const saved = localStorage.getItem(STORAGE_KEY);

   132→          const parsed = JSON.parse(saved);
   133→          if (parsed[this.config.worldId]) {

   136→        }
   137→      } catch { /* ignore */ }
   138→    }
   139→
   140→    this.registry = createParameterRegistry({ ...defaults, ...savedParams });
   141→    this.state = {
   142→      worldId: this.config.worldId,
   143→      running: false,
   144→      iterations: [],
   145→      currentScore: 0,
   146→      bestScore: 0,
   147→      currentParams: registryToOverrides(this.registry),
   148→      bestParams: registryToOverrides(this.registry),
   149→      acceptedCount: 0,
   150→      rejectedCount: 0,
   151→      totalIterations: 0,
   152→      learnedKnowledge: {},
   153→    };
   154→  }
   155→
   156→  /** Set the live engine reference so learned params can be applied. */

   160→
   161→  private worker: Worker | null = null;
   162→  private pendingRequestId = 0;
   163→  private waitingForWorker = false;
   164→
   165→  /** Start continuous learning. */
   166→  start(referenceProfile: ReferenceProfile): void {
   167→    if (this.running) return;
   168→    this.running = true;
   169→    this.state.running = true;
   170→    this.onStateChangeCallback?.(this.state);
   171→
   172→    // Initialize Web Worker for off-main-thread rendering
   173→    try {
   174→      this.worker = new Worker(new URL('./renderWorker.ts', import.meta.url));
   175→      this.worker.onmessage = (e: MessageEvent) => {
   176→        this.handleWorkerResponse(e.data, referenceProfile);
   177→      };
   178→      this.worker.onerror = (err) => {
   179→        console.warn('[Trainer] Worker error, falling back to main thread:', err);
   180→        this.worker = null;
   181→      };
   182→    } catch {
   183→      console.warn('[Trainer] Could not create Worker, using main thread');
   184→      this.worker = null;
   185→    }
   186→
   187→    // Run first iteration
   188→    this.runIteration(referenceProfile);
   189→
   190→    // Schedule periodic iterations
   191→    this.intervalId = setInterval(() => {
   192→      this.runIteration(referenceProfile);
   193→    }, this.config.iterationIntervalMs);
   194→  }
   195→
   196→  /** Stop learning. */
   197→  stop(): void {
   198→    this.running = false;
   199→    this.state.running = false;
   200→    if (this.intervalId) {
   201→      clearInterval(this.intervalId);
   202→      this.intervalId = null;
   203→    }
   204→    if (this.worker) {
   205→      this.worker.terminate();
   206→      this.worker = null;
   207→    }
   208→    this.waitingForWorker = false;
   209→    this.onStateChangeCallback?.(this.state);
   210→  }
   211→
   212→  isRunning(): boolean {
   213→    return this.running;
   214→  }
   215→
   216→  getState(): LearningState {
   217→    return { ...this.state };
   218→  }
   219→
   220→  getCurrentParams(): Record<string, number> {
   221→    return registryToOverrides(this.registry);
   222→  }
   223→
   224→  onIteration(cb: (iter: LearningIteration) => void): void {
   225→    this.onIterationCallback = cb;
   226→  }
   227→
   228→  onStateChange(cb: (state: LearningState) => void): void {
   229→    this.onStateChangeCallback = cb;
   230→  }
   231→
   232→  onParamsApplied(cb: (params: Record<string, number>) => void): void {
   233→    this.onParamsAppliedCallback = cb;
   234→  }
   235→
   236→  /** Run a single learning iteration — sends work to Worker if available. */
   237→  private runIteration(referenceProfile: ReferenceProfile): void {
   238→    if (!this.running) return;
   239→    if (this.waitingForWorker) return; // don't stack requests
   240→
   241→    const { worldId, seed, renderDuration } = this.config;
   242→    const dna = getWorldDNA(worldId);
   243→    if (!dna) return;
   244→
   245→    const currentOverrides = registryToOverrides(this.registry);
   246→
   247→    // If Worker is available, send the work there (non-blocking!)
   248→    if (this.worker) {
   249→      this.waitingForWorker = true;
   250→      this.pendingRequestId++;
   251→      this.worker.postMessage({
   252→        type: 'train',
   253→        id: this.pendingRequestId,
   254→        seed,
   255→        worldId,
   256→        duration: renderDuration,
   257→        currentParams: currentOverrides,
   258→        referenceProfile,
   259→        maxChangesPerIteration: this.config.maxChangesPerIteration,
   260→      });
   261→      // The Worker will call handleWorkerResponse when done.
   262→      // The main thread is FREE — engine keeps playing!
   263→      return;
   264→    }
   265→
   266→    // Fallback: run on main thread (will block briefly)
   267→    this.runIterationMainThread(referenceProfile);
   268→  }
   269→
   270→  /** Handle response from Worker. */
   271→  private handleWorkerResponse(data: any, referenceProfile: ReferenceProfile): void {
   272→    this.waitingForWorker = false;
   273→    if (!this.running) return;
   274→
   275→    if (data.error) {
   276→      console.warn('[Trainer] Worker error:', data.error);
   277→      return;
   278→    }
   279→
   280→    if (data.type !== 'train') return;
   281→
   282→    const { oldScore, newScore, changes, accepted, perVoiceScores, voiceAnalysis } = data;
   283→
   284→    // Update state
   285→    this.state.currentScore = newScore;
   286→    if (perVoiceScores) this.state.perVoiceScores = perVoiceScores;
   287→    if (voiceAnalysis) this.state.voiceAnalysis = voiceAnalysis;
   288→
   289→    if (accepted) {
   290→      // Apply changes to local registry
   291→      this.registry = applyChanges(this.registry, changes);
   292→      this.state.acceptedCount++;
   293→      if (newScore > this.state.bestScore) {
   294→        this.state.bestScore = newScore;
   295→        this.state.bestParams = registryToOverrides(this.registry);
   296→      }
   297→
   298→      // Apply to live engine
   299→      if (this.config.autoApplyToEngine && this.engineRef) {
   300→        const overrides = registryToOverrides(this.registry);
   301→        this.engineRef.setWorld(overrides);
   302→        this.onParamsAppliedCallback?.(overrides);
   303→      }
   304→
   305→      // Save to localStorage
   306→      if (this.config.saveToLocalStorage) {
   307→        this.saveParams();
   308→      }
   309→
   310→      // Update learned knowledge
   311→      for (const c of changes) {
   312→        if (!this.state.learnedKnowledge[c.name]) {
   313→          this.state.learnedKnowledge[c.name] = { value: c.newValue, score: newScore, attempts: 1 };
   314→        } else {
   315→          this.state.learnedKnowledge[c.name].value = c.newValue;
   316→          this.state.learnedKnowledge[c.name].score = newScore;
   317→          this.state.learnedKnowledge[c.name].attempts++;
   318→        }
   319→      }
   320→    } else {
   321→      this.state.rejectedCount++;
   322→      for (const c of changes) {
   323→        this.triedChanges.add(`${c.name}:${c.delta > 0 ? '+' : '-'}`);
   324→      }
   325→    }
   326→
   327→    this.state.totalIterations++;
   328→
   329→    const iteration: LearningIteration = {
   330→      iteration: this.state.totalIterations,
   331→      timestamp: Date.now(),
   332→      changes,
   333→      oldScore,
   334→      newScore,
   335→      scoreDelta: newScore - oldScore,
   336→      accepted,
   337→      reason: accepted
   338→        ? `score improved by ${(newScore - oldScore).toFixed(1)}`
   339→        : `score dropped by ${Math.abs(newScore - oldScore).toFixed(1)}`,
   340→      topProblem: changes[0]?.name || 'none',
   341→    };
   342→
   343→    this.state.iterations.push(iteration);
   344→    if (this.state.iterations.length > 100) {
   345→      this.state.iterations = this.state.iterations.slice(-100);
   346→    }
   347→
   348→    this.onIterationCallback?.(iteration);
   349→    this.onStateChangeCallback?.(this.state);
   350→  }
   351→
   352→  /** Run iteration on main thread (fallback when Worker unavailable). */
   353→  private runIterationMainThread(referenceProfile: ReferenceProfile): void {
   354→
   355→    try {
   356→      const { worldId, seed, renderDuration } = this.config;
   357→      const dna = getWorldDNA(worldId);
   358→      if (!dna) return;
   359→
   360→      const currentOverrides = registryToOverrides(this.registry);
   361→
   362→      // 1. Render full mix for overall score
   363→      const currentRender = renderLite(worldId, renderDuration, {
   364→        paramOverrides: currentOverrides,
   365→      });
   366→
   367→      // 2. Analyze full mix
   368→      const analysis = analyzeAudio(currentRender.samplesL, currentRender.samplesR, SR);
   369→      const metrics = this.analysisToMetrics(analysis, worldId);
   370→
   371→      // 3. Full mix score against reference
   372→      const scoreResult = computeReferenceScore(metrics, referenceProfile, dna.bpmTarget);
   373→
   374→      // 4. Per-voice analysis (isolates each voice for targeted learning)
   375→      let perVoiceComparison: ReturnType<typeof comparePerVoiceToReference> | null = null;
   376→      try {
   377→        const perVoiceReport = analyzePerVoice(seed, worldId, renderDuration, currentOverrides);
   378→        perVoiceComparison = comparePerVoiceToReference(perVoiceReport, referenceProfile);
   379→
   380→        // Update state with per-voice scores
   381→        this.state.perVoiceScores = {
   382→          kick: perVoiceComparison.kickScore,
   383→          bass: perVoiceComparison.bassScore,
   384→          lead: perVoiceComparison.leadScore,
   385→          spectral: perVoiceComparison.spectralScore,
   386→        };
   387→        this.state.voiceAnalysis = {
   388→          kick: { lufs: perVoiceReport.kick.lufs, decayMs: perVoiceReport.kick.decayMs, active: perVoiceReport.kick.active },
   389→          bass: { lufs: perVoiceReport.bass.lufs, bandEnergy: perVoiceReport.bass.bandEnergy, active: perVoiceReport.bass.active },
   390→          lead: { lufs: perVoiceReport.lead.lufs, centroidHz: perVoiceReport.lead.centroidHz, active: perVoiceReport.lead.active },
   391→          hat: { lufs: perVoiceReport.hat.lufs, active: perVoiceReport.hat.active },
   392→        };
   393→      } catch (e) {
   394→        // per-voice analysis can fail on some worlds — continue with full-mix only
   395→      }
   396→
   397→      // 5. Combined score: 60% full-mix + 40% per-voice (if available)
   398→      const oldScore = perVoiceComparison
   399→        ? Math.round(scoreResult.total * 0.6 + perVoiceComparison.totalScore * 0.4)
   400→        : scoreResult.total;
   401→
   402→      // Update current score
   403→      this.state.currentScore = oldScore;
   404→      if (oldScore > this.state.bestScore) {
   405→        this.state.bestScore = oldScore;
   406→        this.state.bestParams = { ...currentOverrides };
   407→      }
   408→
   409→      // 6. Propose changes — use per-voice errors if available, else full-mix problems
   410→      const changes = perVoiceComparison && perVoiceComparison.errors.length > 0
   411→        ? this.proposeChangesFromPerVoice(perVoiceComparison, referenceProfile)
   412→        : this.proposeChanges(scoreResult, referenceProfile);
   413→
   414→      if (changes.length === 0) {
   415→        // Random exploration
   416→        const randomParam = this.registry[Math.floor(Math.random() * this.registry.length)];
   417→        const randomDir = Math.random() > 0.5 ? 1 : -1;
   418→        const newValue = adjustParameter(randomParam, randomDir * randomParam.step * 2);
   419→        if (newValue !== randomParam.current) {
   420→          changes.push({
   421→            name: randomParam.name,
   422→            oldValue: randomParam.current,
   423→            newValue,
   424→            delta: newValue - randomParam.current,
   425→          });
   426→        }
   427→      }
   428→
   429→      if (changes.length === 0) return;
   430→
   431→      // 5. Render with new params
   432→      const newRegistry = applyChanges(this.registry, changes);
   433→      const newRender = renderLite(worldId, renderDuration, {
   434→        paramOverrides: registryToOverrides(newRegistry),
   435→      });
   436→      const newAnalysis = analyzeAudio(newRender.samplesL, newRender.samplesR, SR);
   437→      const newMetrics = this.analysisToMetrics(newAnalysis, worldId);
   438→      const newScoreResult = computeReferenceScore(newMetrics, referenceProfile, dna.bpmTarget);
   439→      const newScore = newScoreResult.total;
   440→
   441→      // 6. Accept or reject
   442→      const accepted = newScore > oldScore;
   443→      const scoreDelta = newScore - oldScore;
   444→      const topProblem = scoreResult.topProblems[0]?.name || 'none';
   445→
   446→      if (accepted) {
   447→        this.registry = newRegistry;
   448→        this.state.currentScore = newScore;
   449→        this.state.acceptedCount++;
   450→        if (newScore > this.state.bestScore) {
   451→          this.state.bestScore = newScore;
   452→          this.state.bestParams = registryToOverrides(newRegistry);
   453→        }
   454→        // Clear tried changes on success — the landscape changed
   455→        this.triedChanges.clear();
   456→
   457→        // Apply to live engine if enabled
   458→        if (this.config.autoApplyToEngine && this.engineRef) {
   459→          const overrides = registryToOverrides(this.registry);
   460→          this.engineRef.setWorld(overrides);
   461→          this.onParamsAppliedCallback?.(overrides);
   462→        }
   463→
   464→        // Save to localStorage
   465→        if (this.config.saveToLocalStorage) {
   466→          this.saveParams();
   467→        }
   468→
   469→        // Update learned knowledge
   470→        for (const c of changes) {
   471→          if (!this.state.learnedKnowledge[c.name]) {
   472→            this.state.learnedKnowledge[c.name] = { value: c.newValue, score: newScore, attempts: 1 };
   473→          } else {
   474→            this.state.learnedKnowledge[c.name].value = c.newValue;
   475→            this.state.learnedKnowledge[c.name].score = newScore;
   476→            this.state.learnedKnowledge[c.name].attempts++;
   477→          }
   478→        }
   479→      } else {
   480→        this.state.rejectedCount++;
   481→        for (const c of changes) {
   482→          this.triedChanges.add(`${c.name}:${c.delta > 0 ? '+' : '-'}`);
   483→        }
   484→      }
   485→
   486→      this.state.totalIterations++;
   487→
   488→      const iteration: LearningIteration = {
   489→        iteration: this.state.totalIterations,
   490→        timestamp: Date.now(),
   491→        changes,
   492→        oldScore,
   493→        newScore,
   494→        scoreDelta,
   495→        accepted,
   496→        reason: accepted
   497→          ? `score improved by ${scoreDelta.toFixed(1)}`
   498→          : `score dropped by ${Math.abs(scoreDelta).toFixed(1)}`,
   499→        topProblem,
   500→      };
   501→
   502→      this.state.iterations.push(iteration);
   503→      // Keep last 100 iterations
   504→      if (this.state.iterations.length > 100) {
   505→        this.state.iterations = this.state.iterations.slice(-100);
   506→      }
   507→
   508→      this.onIterationCallback?.(iteration);
   509→      this.onStateChangeCallback?.(this.state);
   510→
   511→    } catch (err) {
   512→      console.error('[Trainer] iteration error:', err);
   513→    }
   514→  }

   516→  /** Propose changes based on per-voice analysis errors. */
   517→  private proposeChangesFromPerVoice(
   518→    perVoice: ReturnType<typeof comparePerVoiceToReference>,
   519→    referenceProfile: ReferenceProfile,
   520→  ): ParameterChange[] {
   521→    const changes: ParameterChange[] = [];
   522→    const usedParams = new Set<string>();
   523→
   524→    const isTried = (name: string, dir: number) =>
   525→      this.triedChanges.has(`${name}:${dir > 0 ? '+' : '-'}`);
   526→
   527→    // Sort errors by magnitude (biggest error first)
   528→    const sortedErrors = [...perVoice.errors].sort((a, b) => Math.abs(b.error) - Math.abs(a.error));
   529→
   530→    for (const err of sortedErrors.slice(0, this.config.maxChangesPerIteration)) {
   531→      let paramName: string | null = null;
   532→      let delta = 0;
   533→
   534→      // Map per-voice errors to parameter changes
   535→      if (err.voice === 'kick' && err.metric === 'decay') {
   536→        const param = this.registry.find(p => p.name === 'kickDecay');
   537→        if (param && !usedParams.has('kickDecay')) {
   538→          paramName = 'kickDecay';
   539→          const refDecaySec = err.refValue / 1000;
   540→          const direction = refDecaySec > param.current ? 1 : -1;
   541→          if (isTried('kickDecay', direction)) {
   542→            if (isTried('kickDecay', -direction)) { paramName = null; }
   543→            else { delta = -direction * param.step * 2; }
   544→          } else {
   545→            delta = (refDecaySec - param.current) * 0.5;
   546→          }
   547→          if (paramName) usedParams.add('kickDecay');
   548→        }
   549→      } else if (err.voice === 'bass' && err.metric === 'energy') {
   550→        const param = this.registry.find(p => p.name === 'bassCutoff');
   551→        if (param && !usedParams.has('bassCutoff')) {
   552→          paramName = 'bassCutoff';
   553→          // If our bass energy is too low, raise cutoff for more brightness
   554→          const direction = err.error < 0 ? 1 : -1;
   555→          if (isTried('bassCutoff', direction)) {
   556→            if (isTried('bassCutoff', -direction)) { paramName = null; }
   557→            else { delta = -direction * param.step * 2; }
   558→          } else { delta = direction * 80; }
   559→          if (paramName) usedParams.add('bassCutoff');
   560→        }
   561→      } else if (err.voice === 'lead' && err.metric === 'centroid') {
   562→        const param = this.registry.find(p => p.name === 'leadCutoff');
   563→        if (param && !usedParams.has('leadCutoff')) {
   564→          paramName = 'leadCutoff';
   565→          const direction = err.error < 0 ? 1 : -1;
   566→          if (isTried('leadCutoff', direction)) {
   567→            if (isTried('leadCutoff', -direction)) { paramName = null; }
   568→            else { delta = -direction * param.step * 2; }
   569→          } else { delta = direction * 500; }
   570→          if (paramName) usedParams.add('leadCutoff');
   571→        }
   572→      }
   573→
   574→      if (paramName) {
   575→        const param = this.registry.find(p => p.name === paramName);
   576→        if (param) {
   577→          const newValue = adjustParameter(param, delta);
   578→          if (newValue !== param.current) {
   579→            changes.push({
   580→              name: paramName,
   581→              oldValue: param.current,
   582→              newValue,
   583→              delta: newValue - param.current,
   584→            });
   585→          }
   586→        }
   587→      }
   588→    }
   589→
   590→    return changes;
   591→  }

   593→  /** Propose parameter changes based on score problems. */
   594→  private proposeChanges(
   595→    scoreResult: ReturnType<typeof computeReferenceScore>,
   596→    referenceProfile: ReferenceProfile,
   597→  ): ParameterChange[] {
   598→    const changes: ParameterChange[] = [];
   599→    const usedParams = new Set<string>();
   600→
   601→    const isTried = (name: string, dir: number) =>
   602→      this.triedChanges.has(`${name}:${dir > 0 ? '+' : '-'}`);
   603→
   604→    for (const problem of scoreResult.topProblems.slice(0, this.config.maxChangesPerIteration)) {
   605→      let paramName: string | null = null;
   606→      let delta = 0;
   607→
   608→      switch (problem.name) {
   609→        case 'Kick Decay': {
   610→          const param = this.registry.find(p => p.name === 'kickDecay');
   611→          if (param && !usedParams.has('kickDecay')) {
   612→            paramName = 'kickDecay';
   613→            const refDecaySec = referenceProfile.kickDecayMs.mean / 1000;
   614→            const direction = refDecaySec > param.current ? 1 : -1;
   615→            if (isTried('kickDecay', direction)) {
   616→              if (isTried('kickDecay', -direction)) { paramName = null; }
   617→              else { delta = -direction * param.step * 2; }
   618→            } else {
   619→              delta = (refDecaySec - param.current) * 0.5;
   620→            }
   621→            if (paramName) usedParams.add('kickDecay');
   622→          }
   623→          break;
   624→        }
   625→        case 'Bass Decay': {
   626→          const param = this.registry.find(p => p.name === 'bassCutoff');
   627→          if (param && !usedParams.has('bassCutoff')) {
   628→            paramName = 'bassCutoff';
   629→            const direction = problem.error > 0 ? -1 : 1;
   630→            if (isTried('bassCutoff', direction)) {
   631→              if (isTried('bassCutoff', -direction)) { paramName = null; }
   632→              else { delta = -direction * param.step * 2; }
   633→            } else { delta = direction * 60; }
   634→            if (paramName) usedParams.add('bassCutoff');
   635→          }
   636→          break;
   637→        }
   638→        case 'Spectral Balance': {
   639→          const param = this.registry.find(p => p.name === 'leadCutoff');
   640→          if (param && !usedParams.has('leadCutoff')) {
   641→            paramName = 'leadCutoff';
   642→            const direction = problem.error < 0 ? 1 : -1;
   643→            if (isTried('leadCutoff', direction)) {
   644→              if (isTried('leadCutoff', -direction)) { paramName = null; }
   645→              else { delta = -direction * param.step * 2; }
   646→            } else { delta = direction * 400; }
   647→            if (paramName) usedParams.add('leadCutoff');
   648→          }
   649→          break;
   650→        }
   651→        case 'Transient Density':
   652→        case 'Loudness':
   653→        case 'Energy': {
   654→          const param = this.registry.find(p => p.name === 'duck');
   655→          if (param && !usedParams.has('duck')) {
   656→            paramName = 'duck';
   657→            const direction = problem.error < 0 ? 1 : -1;
   658→            if (isTried('duck', direction)) {
   659→              if (isTried('duck', -direction)) { paramName = null; }
   660→              else { delta = -direction * param.step; }
   661→            } else { delta = direction * 0.1; }
   662→            if (paramName) usedParams.add('duck');
   663→          }
   664→          break;
   665→        }
   666→      }
   667→
   668→      if (paramName) {
   669→        const param = this.registry.find(p => p.name === paramName);
   670→        if (param) {
   671→          const newValue = adjustParameter(param, delta);
   672→          if (newValue !== param.current) {
   673→            changes.push({
   674→              name: paramName,
   675→              oldValue: param.current,
   676→              newValue,
   677→              delta: newValue - param.current,
   678→            });
   679→          }
   680→        }
   681→      }
   682→    }
   683→
   684→    return changes;
   685→  }

   687→  /** Convert analysis to reference metrics format. */
   688→  private analysisToMetrics(analysis: ReturnType<typeof analyzeAudio>, worldId: string): ReferenceMetrics {
   689→    const sp = analysis.spectrum;
   690→    const d = analysis.dynamics;
   691→    const le = analysis.lowEnd;
   692→    const tr = analysis.transients;
   693→    const dur = analysis.duration || 1;
   694→
   695→    return {
   696→      bpm: LITE_WORLDS[worldId]?.bpm || 142,
   697→      bpmConfidence: 0.9,
   698→      rms: d.rms,
   699→      peak: d.peak,
   700→      lufs: d.lufs,
   701→      crestFactor: d.crest,
   702→      subEnergy: le.subRms,
   703→      lowEnergy: le.kickRms + le.bassRms,
   704→      midEnergy: sp.bands.find(b => b.name === '500-2k')?.energy || 0,
   705→      highEnergy: sp.bands.find(b => b.name === '2k-8k')?.energy || 0,
   706→      airEnergy: sp.bands.find(b => b.name === '8k-20k')?.energy || 0,
   707→      spectralCentroid: sp.centroidHz,
   708→      spectralFlatness: sp.flatness,
   709→      spectralRolloff: sp.rolloff,
   710→      transientDensity: tr.count / dur,
   711→      kickDensity: tr.count / dur * 0.3,
   712→      hatDensity: tr.count / dur * 0.4,
   713→      percussionDensity: tr.count / dur,
   714→      stereoWidth: 0.35,
   715→      kickDecayMs: le.kickDecay * 1000,
   716→      bassDecayMs: le.bassDecay * 1000,
   717→      rhythmicRegularity: tr.consistency,
   718→      repetitionScore: 0.5,
   719→      energy: Math.min(1, d.rms * 3),
   720→      overallConfidence: 0.8,
   721→      timestamp: Date.now(),
   722→      sourceStream: 'self',
   723→    };
   724→  }

   726→  /** Save learned params to localStorage for future sessions. */
   727→  private saveParams(): void {
   728→    if (typeof window === 'undefined') return;
   729→    try {
   730→      const existing = localStorage.getItem(STORAGE_KEY);
   731→      const all = existing ? JSON.parse(existing) : {};
   732→      all[this.config.worldId] = registryToOverrides(this.registry);
   733→      all[this.config.worldId]._bestScore = this.state.bestScore;
   734→      all[this.config.worldId]._lastUpdated = Date.now();
   735→      localStorage.setItem(STORAGE_KEY, JSON.stringify(all));
   736→    } catch { /* ignore */ }
   737→  }
   738→
   739→  /** Load previously learned params from localStorage. */
   740→  static loadLearnedParams(worldId: string): Record<string, number> | null {
   741→    if (typeof window === 'undefined') return null;
   742→    try {
   743→      const saved = localStorage.getItem(STORAGE_KEY);
   744→      if (!saved) return null;
   745→      const parsed = JSON.parse(saved);
   746→      return parsed[worldId] || null;
   747→    } catch {
   748→      return null;
   749→    }
   750→  }

   752→  /** Reset learning (clear all stored knowledge). */
   753→  reset(): void {
   754→    this.stop();
   755→    this.triedChanges.clear();
   756→    this.state = {
   757→      worldId: this.config.worldId,
   758→      running: false,
   759→      iterations: [],
   760→      currentScore: 0,
   761→      bestScore: 0,
   762→      currentParams: registryToOverrides(this.registry),
   763→      bestParams: registryToOverrides(this.registry),
   764→      acceptedCount: 0,
   765→      rejectedCount: 0,
   766→      totalIterations: 0,
   767→      learnedKnowledge: {},
   768→    };
   769→    if (typeof window !== 'undefined') {
   770→      localStorage.removeItem(STORAGE_KEY);
   771→    }
   772→  }
