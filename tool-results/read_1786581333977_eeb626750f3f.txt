   147→   147→/** Snap a MIDI to the nearest in-scale tone within +/- 6 semitones. */
   148→   148→function snapToScale(midi: number, rootPc: number, scale: Scale): number {
   149→   149→  if (isInScale(rootPc, scale, midi)) return midi
   150→   150→  for (let offset = 1; offset <= 6; offset++) {
   151→   151→    if (isInScale(rootPc, scale, midi + offset)) return midi + offset
   152→   152→    if (isInScale(rootPc, scale, midi - offset)) return midi - offset
   153→   153→  }
   154→   154→  return midi
   155→   155→}

   157→   157→function mean(values: number[]): number {
   158→   158→  if (values.length === 0) return 0
   159→   159→  let s = 0
   160→   160→  for (const v of values) s += v
   161→   161→  return s / values.length
   162→   162→}

   164→   164→function pcsOf(midis: number[]): number[] {
   165→   165→  const set = new Set<number>()
   166→   166→  for (const m of midis) set.add(((m % 12) + 12) % 12)
   167→   167→  return Array.from(set).sort((a, b) => a - b)
   168→   168→}
   169→   169→
   170→   170→function intervalsOf(midis: number[]): number[] {
   171→   171→  const out: number[] = []
   172→   172→  for (let i = 1; i < midis.length; i++) {
   173→   173→    const prev = midis[i - 1]
   174→   174→    const cur = midis[i]
   175→   175→    if (prev !== undefined && cur !== undefined) out.push(cur - prev)
   176→   176→  }
   177→   177→  return out
   178→   178→}

   180→   180→/**
   181→   181→ * Extract a PhraseMaterial from a Motif. The material is rendered within a
   182→   182→ * single bar (stepsPerBar); notes whose step exceeds stepsPerBar are wrapped
   183→   183→ * modulo stepsPerBar so the material always fits one bar.
   184→   184→ */
   185→   185→export function motifToPhraseMaterial(motif: Motif, stepsPerBar: number): PhraseMaterial {
   186→   186→  const notes = motif.notes
   187→   187→  const pitchContour: number[] = []
   188→   188→  const rhythmPattern: number[] = []
   189→   189→  const accentPattern: number[] = []
   190→   190→  const noteDurations: number[] = []
   191→   191→  for (const n of notes) {
   192→   192→    pitchContour.push(n.midi)
   193→   193→    rhythmPattern.push(((n.step % stepsPerBar) + stepsPerBar) % stepsPerBar)
   194→   194→    accentPattern.push(n.velocity)
   195→   195→    noteDurations.push(n.durationSteps)
   196→   196→  }
   197→   197→  const intervals = intervalsOf(pitchContour)
   198→   198→  return {
   199→   199→    motifId: motif.id,
   200→   200→    pitchContour,
   201→   201→    intervalSequence: intervals,
   202→   202→    rhythmPattern,
   203→   203→    onsetPositions: rhythmPattern.slice(),
   204→   204→    accentPattern,
   205→   205→    noteDurations,
   206→   206→    registerProfile: mean(pitchContour),
   207→   207→    harmonicTargets: pcsOf(pitchContour),
   208→   208→    stepsPerBar,
   209→   209→    transformHistory: motif.transformHistory.slice(),
   210→   210→    // F21 shape fields (derived from the motif).
   211→   211→    rhythmicCell: deriveRhythmicCell(rhythmPattern),
   212→   212→    intervalCell: deriveIntervalCell(intervals),
   213→   213→    contour: classifyContour(intervals),
   214→   214→    accentShape: { weights: accentPattern.slice(), climaxPosition: findClimax(accentPattern) },
   215→   215→    densityShape: [Math.min(1, pitchContour.length / stepsPerBar)],
   216→   216→    registerShape: [mean(pitchContour)],
   217→   217→    harmonicTargetShape: pcsOf(pitchContour),
   218→   218→    cadenceTarget:
   219→   219→      pitchContour.length > 0
   220→   220→        ? ((pitchContour[pitchContour.length - 1] ?? 0 % 12) + 12) % 12
   221→   221→        : null,
   222→   222→    phraseArc: buildPhraseArc(1, mean(pitchContour), 0.3),
   223→   223→    developmentHistory: [],
   224→   224→  }
   225→   225→}

   285→   285→/** Derive a repeating rhythmic cell from a rhythm pattern (find the shortest repeating unit). */
   286→   286→function deriveRhythmicCell(rhythmPattern: number[]): number[] {
   287→   287→  if (rhythmPattern.length === 0) return [0]
   288→   288→  if (rhythmPattern.length === 1) return [0]
   289→   289→  const offsets: number[] = []
   290→   290→  for (let i = 1; i < rhythmPattern.length; i++) {
   291→   291→    const cur = rhythmPattern[i] ?? 0
   292→   292→    const first = rhythmPattern[0] ?? 0
   293→   293→    offsets.push(cur - first)
   294→   294→  }
   295→   295→  // If the pattern is short enough, use it directly.
   296→   296→  if (rhythmPattern.length <= 4) return [0, ...offsets]
   297→   297→  // Otherwise look for a repeating cell of length 2-4.
   298→   298→  for (let cellLen = 2; cellLen <= 4; cellLen++) {
   299→   299→    if (rhythmPattern.length >= cellLen * 2) {
   300→   300→      const cell = rhythmPattern.slice(0, cellLen)
   301→   301→      let repeats = true
   302→   302→      for (let i = cellLen; i < rhythmPattern.length; i++) {
   303→   303→        if (rhythmPattern[i] !== cell[i % cellLen]) {
   304→   304→          repeats = false
   305→   305→          break
   306→   306→        }
   307→   307→      }
   308→   308→      if (repeats) {
   309→   309→        const cellOffsets = [0]
   310→   310→        for (let i = 1; i < cellLen; i++) {
   311→   311→          const cur = cell[i] ?? 0
   312→   312→          const first = cell[0] ?? 0
   313→   313→          cellOffsets.push(cur - first)
   314→   314→        }
   315→   315→        return cellOffsets
   316→   316→      }
   317→   317→    }
   318→   318→  }
   319→   319→  return [0, ...offsets]
   320→   320→}

   322→   322→/** Derive a repeating interval cell from an interval sequence. */
   323→   323→function deriveIntervalCell(intervals: number[]): number[] {
   324→   324→  if (intervals.length === 0) return [0]
   325→   325→  if (intervals.length <= 3) return intervals.slice()
   326→   326→  // Look for a repeating cell of length 2-3.
   327→   327→  for (let cellLen = 2; cellLen <= 3; cellLen++) {
   328→   328→    if (intervals.length >= cellLen * 2) {
   329→   329→      const cell = intervals.slice(0, cellLen)
   330→   330→      let repeats = true
   331→   331→      for (let i = cellLen; i < intervals.length; i++) {
   332→   332→        if (intervals[i] !== cell[i % cellLen]) {
   333→   333→          repeats = false
   334→   334→          break
   335→   335→        }
   336→   336→      }
   337→   337→      if (repeats) return cell.slice()
   338→   338→    }
   339→   339→  }
   340→   340→  // Fall back to the first 3 intervals.
   341→   341→  return intervals.slice(0, 3)
   342→   342→}

   344→   344→/** Classify a contour from an interval sequence. */
   345→   345→function classifyContour(intervals: number[]): ContourShape {
   346→   346→  if (intervals.length === 0) return 'flat'
   347→   347→  let up = 0
   348→   348→  let down = 0
   349→   349→  for (const iv of intervals) {
   350→   350→    if (iv > 0) up++
   351→   351→    else if (iv < 0) down++
   352→   352→  }
   353→   353→  if (up === intervals.length) return 'ascending'
   354→   354→  if (down === intervals.length) return 'descending'
   355→   355→  // Arch: rises then falls.
   356→   356→  const midpoint = Math.floor(intervals.length / 2)
   357→   357→  const firstHalfUp = intervals.slice(0, midpoint).filter((iv) => iv > 0).length
   358→   358→  const secondHalfDown = intervals.slice(midpoint).filter((iv) => iv < 0).length
   359→   359→  if (firstHalfUp >= midpoint * 0.6 && secondHalfDown >= (intervals.length - midpoint) * 0.6)
   360→   360→    return 'arch'
   361→   361→  // Valley: falls then rises.
   362→   362→  const firstHalfDown = intervals.slice(0, midpoint).filter((iv) => iv < 0).length
   363→   363→  const secondHalfUp = intervals.slice(midpoint).filter((iv) => iv > 0).length
   364→   364→  if (firstHalfDown >= midpoint * 0.6 && secondHalfUp >= (intervals.length - midpoint) * 0.6)
   365→   365→    return 'valley'
   366→   366→  // Wave: frequent direction changes.
   367→   367→  let changes = 0
   368→   368→  for (let i = 1; i < intervals.length; i++) {
   369→   369→    if (Math.sign(intervals[i] ?? 0) !== Math.sign(intervals[i - 1] ?? 0) && intervals[i] !== 0)
   370→   370→      changes++
   371→   371→  }
   372→   372→  if (changes >= intervals.length * 0.4) return 'wave'
   373→   373→  return 'flat'
   374→   374→}

   376→   376→/** Find the climax position (fraction 0..1) in an accent pattern. */
   377→   377→function findClimax(accents: number[]): number {
   378→   378→  if (accents.length === 0) return 0.5
   379→   379→  let maxIdx = 0
   380→   380→  let maxVal = -1
   381→   381→  for (let i = 0; i < accents.length; i++) {
   382→   382→    if (accents[i] ?? 0 > maxVal) {
   383→   383→      maxVal = accents[i] ?? 0
   384→   384→      maxIdx = i
   385→   385→    }
   386→   386→  }
   387→   387→  return accents.length > 1 ? maxIdx / (accents.length - 1) : 0.5
   388→   388→}

   390→   390→/**
   391→   391→ * Build a PhraseArc for a phrase of `bars` bars. The arc has 6 stages
   392→   392→ * (OPEN → ESTABLISH → DEVELOP → FOCAL → RELEASE → CADENCE) distributed across
   393→   393→ * the bars. The focal/climax point falls around 60-70% of the phrase.
   394→   394→ */
   395→   395→export function buildPhraseArc(
   396→   396→  bars: number,
   397→   397→  registerCenter: number,
   398→   398→  baseTension: number
   399→   399→): PhraseArc {
   400→   400→  const stages: PhraseArcStage[] = []
   401→   401→  const focalBar = Math.max(1, Math.floor(bars * 0.6))
   402→   402→  const cadenceBar = bars - 1
   403→   403→  // Distribute bars across stages proportionally.
   404→   404→  const fractions = [0.1, 0.2, 0.25, 0.15, 0.15, 0.15] // OPEN, ESTABLISH, DEVELOP, FOCAL, RELEASE, CADENCE
   405→   405→  const labels: PhraseArcStage['stage'][] = [
   406→   406→    'OPEN',
   407→   407→    'ESTABLISH',
   408→   408→    'DEVELOP',
   409→   409→    'FOCAL',
   410→   410→    'RELEASE',
   411→   411→    'CADENCE',
   412→   412→  ]
   413→   413→  let cursor = 0
   414→   414→  const tensionTrajectory: number[] = []
   415→   415→  for (let i = 0; i < 6; i++) {
   416→   416→    const len = Math.max(1, Math.round(bars * (fractions[i] ?? 0.15)))
   417→   417→    const start = Math.min(cursor, bars - 1)
   418→   418→    const end = Math.min(cursor + len, bars)
   419→   419→    const stageTension =
   420→   420→      i === 3
   421→   421→        ? Math.min(1, baseTension + 0.4)
   422→   422→        : i === 5
   423→   423→          ? baseTension * 0.2
   424→   424→          : baseTension + i * 0.05
   425→   425→    const stageDensity = i === 0 ? 0.3 : i === 5 ? 0.4 : Math.min(0.9, 0.4 + i * 0.1)
   426→   426→    const stageRegister =
   427→   427→      i === 3 ? registerCenter + 4 : i === 5 ? registerCenter - 2 : registerCenter + (i - 2) * 2
   428→   428→    const label = labels[i] ?? 'DEVELOP'
   429→   429→    stages.push({
   430→   430→      stage: label,
   431→   431→      barRange: [start, end],
   432→   432→      density: stageDensity,
   433→   433→      register: stageRegister,
   434→   434→      tension: Math.max(0, Math.min(1, stageTension)),
   435→   435→    })
   436→   436→    tensionTrajectory.push(Math.max(0, Math.min(1, stageTension)))
   437→   437→    cursor = end
   438→   438→    // Don't break early — always add all 6 stages. For short phrases, stages
   439→   439→    // may share the same bar range, but all labels are present.
   440→   440→  }
   441→   441→  // Ensure the last stage is always CADENCE (even for 1-bar phrases).
   442→   442→  if (stages.length > 0 && stages[stages.length - 1]?.stage !== 'CADENCE') {
   443→   443→    stages.push({
   444→   444→      stage: 'CADENCE',
   445→   445→      barRange: [Math.max(0, bars - 1), bars],
   446→   446→      density: 0.4,
   447→   447→      register: registerCenter - 2,
   448→   448→      tension: Math.max(0, Math.min(1, baseTension * 0.2)),
   449→   449→    })
   450→   450→    tensionTrajectory.push(Math.max(0, Math.min(1, baseTension * 0.2)))
   451→   451→  }
   452→   452→  return { stages, focalBar, cadenceBar, tensionTrajectory }
   453→   453→}
   454→   454→
   455→   455→/** Get the arc stage active at a given bar index within the phrase. */
   456→   456→export function arcStageAt(arc: PhraseArc, bar: number): PhraseArcStage | null {
   457→   457→  for (const s of arc.stages) {
   458→   458→    if (bar >= s.barRange[0] && bar < s.barRange[1]) return s
   459→   459→  }
   460→   460→  return arc.stages[arc.stages.length - 1] ?? null
   461→   461→}

   466→   466→ * CONTINUE — return the material unchanged (lineage only).
   467→   467→ * The phrase reiterates the previous identity.
   468→   468→ */
   469→   469→export function continueMaterial(src: PhraseMaterial): PhraseMaterial {
   470→   470→  return cloneMaterial(src, `${src.motifId}:continue`, 'CONTINUE')
   471→   471→}

   473→   473→/**
   474→   474→ * DEVELOP — mutate the last interval so the contour evolves while keeping the
   475→   475→ * opening intact. Example: A B C D (intervals [x,y,z]) → A B D E where the
   476→   476→ * last step is displaced by one scale degree. Also nudges the second-to-last
   477→   477→ * note occasionally to keep the development from being a single-note change.
   478→   478→ *
   479→   479→ * This is real MOTIF DEVELOPMENT: the pitch contour changes, but the opening
   480→   480→ * shape is preserved so the result is recognizably the same motif.
   481→   481→ */
   482→   482→export function developMaterial(
   483→   483→  src: PhraseMaterial,
   484→   484→  ctx: MaterialTransformContext
   485→   485→): PhraseMaterial {
   486→   486→  const scale = ctx.scale ?? getScale(ctx.scaleName)
   487→   487→  if (src.pitchContour.length < 2 || !scale)
   488→   488→    return cloneMaterial(src, `${src.motifId}:develop`, 'DEVELOP')
   489→   489→  const pcs = scalePcs(ctx.tonic, scale)
   490→   490→  const out = cloneMaterial(src, `${src.motifId}:develop`, 'DEVELOP')
   491→   491→  const variation = ctx.variationAmount ?? 0.3
   492→   492→
   493→   493→  // Mutate the last interval: shift the last note by one scale degree in the
   494→   494→  // direction of the existing contour (or up if flat).
   495→   495→  const lastIdx = src.pitchContour.length - 1
   496→   496→  const prevMidi = src.pitchContour[lastIdx - 1] ?? ctx.rootMidi
   497→   497→  const curMidi = src.pitchContour[lastIdx] ?? ctx.rootMidi
   498→   498→  const dir = curMidi >= prevMidi ? 1 : -1
   499→   499→  // Pick the next scale degree in `dir`.
   500→   500→  const curPc = ((curMidi % 12) + 12) % 12
   501→   501→  const curDegIdx = pcs.indexOf(curPc)
   502→   502→  const nextDegIdx = curDegIdx >= 0 ? (curDegIdx + dir + pcs.length) % pcs.length : 0
   503→   503→  const nextPc = pcs[nextDegIdx] ?? curPc
   504→   504→  // Construct the new midi: keep the octave of curMidi, change the pc, nudge by dir*12 if variation is high.
   505→   505→  let newMidi = curMidi - curPc + nextPc
   506→   506→  if (ctx.rng.next() < variation * 0.5) newMidi += dir * 12
   507→   507→  newMidi = snapToScale(newMidi, ctx.tonic, scale)
   508→   508→  out.pitchContour[lastIdx] = newMidi
   509→   509→
   510→   510→  // Occasionally mutate the second-to-last note too (smaller displacement).
   511→   511→  if (src.pitchContour.length >= 3 && ctx.rng.next() < variation * 0.6) {
   512→   512→    const idx = src.pitchContour.length - 2
   513→   513→    const base = src.pitchContour[idx] ?? ctx.rootMidi
   514→   514→    const step = ctx.rng.pick([1, -1, 2, -2])
   515→   515→    out.pitchContour[idx] = snapToScale(base + step, ctx.tonic, scale)
   516→   516→  }
   517→   517→
   518→   518→  out.intervalSequence = intervalsOf(out.pitchContour)
   519→   519→  out.registerProfile = mean(out.pitchContour)
   520→   520→  out.harmonicTargets = pcsOf(out.pitchContour)
   521→   521→  return out
   522→   522→}
   523→   523→
   524→   524→/**
   525→   525→ * VARIATE — reorder notes (swap two adjacent positions) so the pitch sequence
   526→   526→ * shuffles while preserving the set of pitches and the rhythm. Example:
   527→   527→ * A B C D → A C B D. The contour changes but the material is the same set of
   528→   528→ * notes — a recognizable variant.
   529→   529→ */
   530→   530→export function variateMaterial(
   531→   531→  src: PhraseMaterial,
   532→   532→  ctx: MaterialTransformContext
   533→   533→): PhraseMaterial {
   534→   534→  if (src.pitchContour.length < 3) return cloneMaterial(src, `${src.motifId}:variate`, 'VARIATE')
   535→   535→  const out = cloneMaterial(src, `${src.motifId}:variate`, 'VARIATE')
   536→   536→  // Swap two adjacent notes (not the first — keep the opening stable).
   537→   537→  const i = 1 + Math.floor(ctx.rng.next() * (src.pitchContour.length - 2))
   538→   538→  const j = i + 1
   539→   539→  const tmp = out.pitchContour[i]
   540→   540→  out.pitchContour[i] = out.pitchContour[j] as number
   541→   541→  out.pitchContour[j] = tmp as number
   542→   542→  // Also swap the accent/duration so the rhythm+rhythm tags stay aligned to
   543→   543→  // the pitches (the rhythm PATTERN is preserved — only pitches reorder).
   544→   544→  const tmpV = out.accentPattern[i]
   545→   545→  out.accentPattern[i] = out.accentPattern[j] as number
   546→   546→  out.accentPattern[j] = tmpV as number
   547→   547→  const tmpD = out.noteDurations[i]
   548→   548→  out.noteDurations[i] = out.noteDurations[j] as number
   549→   549→  out.noteDurations[j] = tmpD as number
   550→   550→  out.intervalSequence = intervalsOf(out.pitchContour)
   551→   551→  out.registerProfile = mean(out.pitchContour)
   552→   552→  out.harmonicTargets = pcsOf(out.pitchContour)
   553→   553→  return out
   554→   554→}
   555→   555→
   556→   556→/**
   557→   557→ * ANSWER — contour inversion (mirror intervals around the first note). The
   558→   558→ * rhythm is preserved; the pitch contour flips direction. A rising motif
   559→   559→ * becomes a falling one — the classic "answer" to a "call".
   560→   560→ */
   561→   561→export function answerMaterial(src: PhraseMaterial, ctx: MaterialTransformContext): PhraseMaterial {
   562→   562→  const scale = ctx.scale ?? getScale(ctx.scaleName)
   563→   563→  if (src.pitchContour.length < 2 || !scale)
   564→   564→    return cloneMaterial(src, `${src.motifId}:answer`, 'ANSWER')
   565→   565→  const out = cloneMaterial(src, `${src.motifId}:answer`, 'ANSWER')
   566→   566→  const first = src.pitchContour[0] ?? ctx.rootMidi
   567→   567→  for (let i = 1; i < src.pitchContour.length; i++) {
   568→   568→    const offset = (src.pitchContour[i] ?? first) - first
   569→   569→    out.pitchContour[i] = snapToScale(first - offset, ctx.tonic, scale)
   570→   570→  }
   571→   571→  out.intervalSequence = intervalsOf(out.pitchContour)
   572→   572→  out.registerProfile = mean(out.pitchContour)
   573→   573→  out.harmonicTargets = pcsOf(out.pitchContour)
   574→   574→  return out
   575→   575→}
   576→   576→
   577→   577→/**
   578→   578→ * CONTRAST — introduce new material (new motifId) while staying in the same
   579→   579→ * scale. Generates a fresh short contour from scale degrees. The motifId is
   580→   580→ * NEW (lineage reset) because the material is intentionally different.
   581→   581→ */
   582→   582→export function contrastMaterial(
   583→   583→  src: PhraseMaterial,
   584→   584→  ctx: MaterialTransformContext
   585→   585→): PhraseMaterial {
   586→   586→  const scale = ctx.scale ?? getScale(ctx.scaleName)
   587→   587→  if (!scale) return cloneMaterial(src, `${src.motifId}:contrast`, 'CONTRAST')
   588→   588→  const pcs = scalePcs(ctx.tonic, scale)
   589→   589→  const len = Math.max(3, Math.min(6, src.pitchContour.length || 4))
   590→   590→  const pitchContour: number[] = []
   591→   591→  let cur = ctx.rootMidi
   592→   592→  for (let i = 0; i < len; i++) {
   593→   593→    const pc = ctx.rng.pick(pcs)
   594→   594→    const oct = 4 + (ctx.rng.next() < 0.3 ? 1 : 0)
   595→   595→    const candidate = degreeToMidi(
   596→   596→      ctx.tonic,
   597→   597→      scale,
   598→   598→      pcs.indexOf(pc) >= 0 ? pcs.indexOf(pc) : 0,
   599→   599→      oct
   600→   600→    )
   601→   601→    pitchContour.push(snapToScale(candidate, ctx.tonic, scale))
   602→   602→    cur = candidate
   603→   603→  }
   604→   604→  void cur
   605→   605→  const rhythmPattern: number[] = []
   606→   606→  const accentPattern: number[] = []
   607→   607→  const noteDurations: number[] = []
   608→   608→  let step = 0
   609→   609→  for (let i = 0; i < len; i++) {
   610→   610→    rhythmPattern.push(step % src.stepsPerBar)
   611→   611→    accentPattern.push(i === 0 ? 0.9 : 0.5 + ctx.rng.next() * 0.3)
   612→   612→    noteDurations.push(ctx.rng.pick([1, 1, 2]))
   613→   613→    step += ctx.rng.pick([1, 2, 2])
   614→   614→  }
   615→   615→  const intervals = intervalsOf(pitchContour)
   616→   616→  return {
   617→   617→    motifId: `${src.motifId}:contrast-${Date.now().toString(36)}`,
   618→   618→    pitchContour,
   619→   619→    intervalSequence: intervals,
   620→   620→    rhythmPattern,
   621→   621→    onsetPositions: rhythmPattern.slice(),
   622→   622→    accentPattern,
   623→   623→    noteDurations,
   624→   624→    registerProfile: mean(pitchContour),
   625→   625→    harmonicTargets: pcsOf(pitchContour),
   626→   626→    stepsPerBar: src.stepsPerBar,
   627→   627→    transformHistory: [...src.transformHistory, 'CONTRAST'],
   628→   628→    // F21 shape fields.
   629→   629→    rhythmicCell: deriveRhythmicCell(rhythmPattern),
   630→   630→    intervalCell: deriveIntervalCell(intervals),
   631→   631→    contour: classifyContour(intervals),
   632→   632→    accentShape: { weights: accentPattern.slice(), climaxPosition: findClimax(accentPattern) },
   633→   633→    densityShape: [Math.min(1, pitchContour.length / src.stepsPerBar)],
   634→   634→    registerShape: [mean(pitchContour)],
   635→   635→    harmonicTargetShape: pcsOf(pitchContour),
   636→   636→    cadenceTarget:
   637→   637→      pitchContour.length > 0
   638→   638→        ? ((pitchContour[pitchContour.length - 1] ?? 0 % 12) + 12) % 12
   639→   639→        : null,
   640→   640→    phraseArc: buildPhraseArc(1, mean(pitchContour), 0.3),
   641→   641→    developmentHistory: src.developmentHistory.slice(),
   642→   642→  }
   643→   643→}
   644→   644→
   645→   645→/**
   646→   646→ * INTENSIFY — rhythmic augmentation: subdivide longer notes into shorter ones
   647→   647→ * so density rises. Pitches are preserved (a note may repeat). The contour is
   648→   648→ * unchanged but the rhythm becomes busier — the musical "intensification".
   649→   649→ */
   650→   650→export function intensifyMaterial(
   651→   651→  src: PhraseMaterial,
   652→   652→  ctx: MaterialTransformContext
   653→   653→): PhraseMaterial {
   654→   654→  if (src.pitchContour.length === 0)
   655→   655→    return cloneMaterial(src, `${src.motifId}:intensify`, 'INTENSIFY')
   656→   656→  const out = cloneMaterial(src, `${src.motifId}:intensify`, 'INTENSIFY')
   657→   657→  const newPitches: number[] = []
   658→   658→  const newRhythm: number[] = []
   659→   659→  const newAccent: number[] = []
   660→   660→  const newDurations: number[] = []
   661→   661→  for (let i = 0; i < src.pitchContour.length; i++) {
   662→   662→    const midi = src.pitchContour[i] as number
   663→   663→    const step = src.rhythmPattern[i] as number
   664→   664→    const dur = src.noteDurations[i] as number
   665→   665→    const vel = src.accentPattern[i] as number
   666→   666→    newPitches.push(midi)
   667→   667→    newRhythm.push(step)
   668→   668→    newAccent.push(vel)
   669→   669→    newDurations.push(Math.max(1, Math.floor(dur / 2)))
   670→   670→    // If the original note was long enough, add a repeated note (rhythmic subdivision).
   671→   671→    if (dur >= 2 && ctx.rng.next() < (ctx.variationAmount ?? 0.5) + 0.3) {
   672→   672→      const subStep = (step + Math.floor(dur / 2)) % src.stepsPerBar
   673→   673→      newPitches.push(midi)
   674→   674→      newRhythm.push(subStep)
   675→   675→      newAccent.push(vel * 0.7)
   676→   676→      newDurations.push(Math.max(1, dur - Math.floor(dur / 2)))
   677→   677→    }
   678→   678→  }
   679→   679→  out.pitchContour = newPitches
   680→   680→  out.rhythmPattern = newRhythm
   681→   681→  out.onsetPositions = newRhythm.slice()
   682→   682→  out.accentPattern = newAccent
   683→   683→  out.noteDurations = newDurations
   684→   684→  out.intervalSequence = intervalsOf(newPitches)
   685→   685→  out.registerProfile = mean(newPitches)
   686→   686→  out.harmonicTargets = pcsOf(newPitches)
   687→   687→  return out
   688→   688→}

   690→   690→/**
   691→   691→ * REDUCE — fragment: keep only the first half of the material. Creates space.
   692→   692→ * Example: A B C D → A B. The opening identity is preserved.
   693→   693→ */
   694→   694→export function reduceMaterial(src: PhraseMaterial): PhraseMaterial {
   695→   695→  if (src.pitchContour.length <= 1) return cloneMaterial(src, `${src.motifId}:reduce`, 'REDUCE')
   696→   696→  const keep = Math.max(1, Math.ceil(src.pitchContour.length / 2))
   697→   697→  const out = cloneMaterial(src, `${src.motifId}:reduce`, 'REDUCE')
   698→   698→  out.pitchContour = src.pitchContour.slice(0, keep)
   699→   699→  out.rhythmPattern = src.rhythmPattern.slice(0, keep)
   700→   700→  out.onsetPositions = src.onsetPositions.slice(0, keep)
   701→   701→  out.accentPattern = src.accentPattern.slice(0, keep)
   702→   702→  out.noteDurations = src.noteDurations.slice(0, keep)
   703→   703→  out.intervalSequence = intervalsOf(out.pitchContour)
   704→   704→  out.registerProfile = mean(out.pitchContour)
   705→   705→  out.harmonicTargets = pcsOf(out.pitchContour)
   706→   706→  return out
   707→   707→}

   709→   709→/**
   710→   710→ * BREAK — strip to a single sustained note (the first pitch). The most
   711→   711→ * extreme reduction — used at breakdown points.
   712→   712→ */
   713→   713→export function breakMaterial(src: PhraseMaterial): PhraseMaterial {
   714→   714→  if (src.pitchContour.length === 0) return cloneMaterial(src, `${src.motifId}:break`, 'BREAK')
   715→   715→  const out = cloneMaterial(src, `${src.motifId}:break`, 'BREAK')
   716→   716→  out.pitchContour = [src.pitchContour[0] as number]
   717→   717→  out.rhythmPattern = [0]
   718→   718→  out.onsetPositions = [0]
   719→   719→  out.accentPattern = [0.9]
   720→   720→  out.noteDurations = [src.stepsPerBar]
   721→   721→  out.intervalSequence = []
   722→   722→  out.registerProfile = out.pitchContour[0] as number
   723→   723→  out.harmonicTargets = pcsOf(out.pitchContour)
   724→   724→  return out
   725→   725→}

   727→   727→/**
   728→   728→ * RESOLVE — append a stepwise descent to the cadence target pitch class.
   729→   729→ * The motif's identity is preserved (opening notes unchanged) but a resolving
   730→   730→ * tail is added that lands on the cadence target (root/third/fifth). This is
   731→   731→ * the musical "resolution" at a phrase cadence.
   732→   732→ */
   733→   733→export function resolveMaterial(
   734→   734→  src: PhraseMaterial,
   735→   735→  ctx: MaterialTransformContext
   736→   736→): PhraseMaterial {
   737→   737→  const scale = ctx.scale ?? getScale(ctx.scaleName)
   738→   738→  if (!scale || src.pitchContour.length === 0)
   739→   739→    return cloneMaterial(src, `${src.motifId}:resolve`, 'RESOLVE')
   740→   740→  const pcs = scalePcs(ctx.tonic, scale)
   741→   741→  const targetPc = ctx.cadenceTargetPc ?? pcs[0] ?? ctx.tonic
   742→   742→  const out = cloneMaterial(src, `${src.motifId}:resolve`, 'RESOLVE')
   743→   743→  // Start the descent from the last pitch.
   744→   744→  let cur = src.pitchContour[src.pitchContour.length - 1] ?? ctx.rootMidi
   745→   745→  let step = ((src.rhythmPattern[src.rhythmPattern.length - 1] ?? 0) + 2) % src.stepsPerBar
   746→   746→  // Walk down by scale degrees until we land on the target pc.
   747→   747→  let guard = 0
   748→   748→  while (((cur % 12) + 12) % 12 !== targetPc && guard < 8) {
   749→   749→    const curPc = ((cur % 12) + 12) % 12
   750→   750→    const curDegIdx = pcs.indexOf(curPc)
   751→   751→    const nextDegIdx = curDegIdx >= 0 ? (curDegIdx - 1 + pcs.length) % pcs.length : 0
   752→   752→    const nextPc = pcs[nextDegIdx] ?? targetPc
   753→   753→    cur = cur - curPc + nextPc
   754→   754→    cur = snapToScale(cur, ctx.tonic, scale)
   755→   755→    out.pitchContour.push(cur)
   756→   756→    out.rhythmPattern.push(step)
   757→   757→    out.onsetPositions.push(step)
   758→   758→    out.accentPattern.push(0.7)
   759→   759→    out.noteDurations.push(1)
   760→   760→    step = (step + 1) % src.stepsPerBar
   761→   761→    guard++
   762→   762→  }
   763→   763→  // Final cadence note — emphasize.
   764→   764→  out.pitchContour.push(cur)
   765→   765→  out.rhythmPattern.push(step)
   766→   766→  out.onsetPositions.push(step)
   767→   767→  out.accentPattern.push(1.0)
   768→   768→  out.noteDurations.push(2)
   769→   769→  out.intervalSequence = intervalsOf(out.pitchContour)
   770→   770→  out.registerProfile = mean(out.pitchContour)
   771→   771→  out.harmonicTargets = pcsOf(out.pitchContour)
   772→   772→  return out
   773→   773→}

   775→   775→/**
   776→   776→ * TRANSITION — bridge: generate a short connecting motif that walks from the
   777→   777→ * last pitch of the source toward the root. Used at section boundaries.
   778→   778→ */
   779→   779→export function transitionMaterial(
   780→   780→  src: PhraseMaterial,
   781→   781→  ctx: MaterialTransformContext
   782→   782→): PhraseMaterial {
   783→   783→  const scale = ctx.scale ?? getScale(ctx.scaleName)
   784→   784→  if (!scale) return cloneMaterial(src, `${src.motifId}:transition`, 'TRANSITION')
   785→   785→  const pcs = scalePcs(ctx.tonic, scale)
   786→   786→  const out = cloneMaterial(src, `${src.motifId}:transition`, 'TRANSITION')
   787→   787→  let cur = src.pitchContour[src.pitchContour.length - 1] ?? ctx.rootMidi
   788→   788→  let step = ((src.rhythmPattern[src.rhythmPattern.length - 1] ?? 0) + 2) % src.stepsPerBar
   789→   789→  // Walk by scale degrees toward the root pc.
   790→   790→  const rootPc = pcs[0] ?? ctx.tonic
   791→   791→  let guard = 0
   792→   792→  while (((cur % 12) + 12) % 12 !== rootPc && guard < 6) {
   793→   793→    const curPc = ((cur % 12) + 12) % 12
   794→   794→    const curDegIdx = pcs.indexOf(curPc)
   795→   795→    const dir = curDegIdx > 0 ? -1 : 0
   796→   796→    const nextDegIdx = curDegIdx >= 0 ? (curDegIdx + dir + pcs.length) % pcs.length : 0
   797→   797→    const nextPc = pcs[nextDegIdx] ?? rootPc
   798→   798→    cur = cur - curPc + nextPc
   799→   799→    cur = snapToScale(cur, ctx.tonic, scale)
   800→   800→    out.pitchContour.push(cur)
   801→   801→    out.rhythmPattern.push(step)
   802→   802→    out.onsetPositions.push(step)
   803→   803→    out.accentPattern.push(0.6)
   804→   804→    out.noteDurations.push(1)
   805→   805→    step = (step + 1) % src.stepsPerBar
   806→   806→    guard++
   807→   807→  }
   808→   808→  out.intervalSequence = intervalsOf(out.pitchContour)
   809→   809→  out.registerProfile = mean(out.pitchContour)
   810→   810→  out.harmonicTargets = pcsOf(out.pitchContour)
   811→   811→  return out
   812→   812→}

   814→   814→/**
   815→   815→ * Dispatch a development operator against a PhraseMaterial. At minimum
   816→   816→ * DEVELOP / VARIATE / ANSWER / RESOLVE transform real note material; the
   817→   817→ * remaining operators also transform material (CONTINUE is identity).
   818→   818→ */
   819→   819→export function applyOperatorToMaterial(
   820→   820→  op: DevelopmentOperator,
   821→   821→  src: PhraseMaterial,
   822→   822→  ctx: MaterialTransformContext
   823→   823→): PhraseMaterial {
   824→   824→  switch (op) {
   825→   825→    case 'CONTINUE':
   826→   826→      return continueMaterial(src)
   827→   827→    case 'DEVELOP':
   828→   828→      return developMaterial(src, ctx)
   829→   829→    case 'ANSWER':
   830→   830→      return answerMaterial(src, ctx)
   831→   831→    case 'CONTRAST':
   832→   832→      return contrastMaterial(src, ctx)
   833→   833→    case 'VARIATE':
   834→   834→      return variateMaterial(src, ctx)
   835→   835→    case 'INTENSIFY':
   836→   836→      return intensifyMaterial(src, ctx)
   837→   837→    case 'REDUCE':
   838→   838→      return reduceMaterial(src)
   839→   839→    case 'BREAK':
   840→   840→      return breakMaterial(src)
   841→   841→    case 'RESOLVE':
   842→   842→      return resolveMaterial(src, ctx)
   843→   843→    case 'TRANSITION':
   844→   844→      return transitionMaterial(src, ctx)
   845→   845→    default:
   846→   846→      return continueMaterial(src)
   847→   847→  }
   848→   848→}
   849→   849→
   850→   850→// ─────────────────────────── SIMILARITY ───────────────────────────
   851→   851→
   852→   852→/**
   853→   853→ * Structural similarity in [0, 1] between two PhraseMaterials. Combines:
   854→   854→ *  - contour direction agreement (fraction of matching interval signs)
   855→   855→ *  - interval-class agreement (fraction of matching signed interval classes mod 12)
   856→   856→ *  - pitch-class set overlap (Jaccard of harmonicTargets)
   857→   857→ *
   858→   858→ * Returns 1 for identical materials, lower for divergent. Used by the A/B
   859→   859→ * test suite to prove a transformed phrase is still recognizably related to
   860→   860→ * its source (motif identity is preserved through development).
   861→   861→ */
   862→   862→export function materialSimilarity(a: PhraseMaterial, b: PhraseMaterial): number {
   863→   863→  if (a.pitchContour.length === 0 && b.pitchContour.length === 0) return 1
   864→   864→  if (a.pitchContour.length === 0 || b.pitchContour.length === 0) return 0
   865→   865→
   866→   866→  // Contour direction agreement.
   867→   867→  const dirA = a.intervalSequence.map((iv) => (iv > 0 ? 1 : iv < 0 ? -1 : 0))
   868→   868→  const dirB = b.intervalSequence.map((iv) => (iv > 0 ? 1 : iv < 0 ? -1 : 0))
   869→   869→  const minLen = Math.min(dirA.length, dirB.length)
   870→   870→  let dirMatch = 0
   871→   871→  for (let i = 0; i < minLen; i++) {
   872→   872→    if (dirA[i] === dirB[i]) dirMatch++
   873→   873→  }
   874→   874→  const contourScore = minLen > 0 ? dirMatch / minLen : 0.5
   875→   875→
   876→   876→  // Interval-class agreement (signed mod 12).
   877→   877→  const icA = a.intervalSequence.map((iv) => {
   878→   878→    const cls = ((iv % 12) + 12) % 12
   879→   879→    return iv < 0 ? -cls : cls
   880→   880→  })
   881→   881→  const icB = b.intervalSequence.map((iv) => {
   882→   882→    const cls = ((iv % 12) + 12) % 12
   883→   883→    return iv < 0 ? -cls : cls
   884→   884→  })
   885→   885→  const minIc = Math.min(icA.length, icB.length)
   886→   886→  let icMatch = 0
   887→   887→  for (let i = 0; i < minIc; i++) {
   888→   888→    if (icA[i] === icB[i]) icMatch++
   889→   889→  }
   890→   890→  const intervalScore = minIc > 0 ? icMatch / minIc : 0.5
   891→   891→
   892→   892→  // Pitch-class set overlap (Jaccard).
   893→   893→  const setA = new Set(a.harmonicTargets)
   894→   894→  const setB = new Set(b.harmonicTargets)
   895→   895→  let inter = 0
   896→   896→  for (const pc of setA) if (setB.has(pc)) inter++
   897→   897→  const union = setA.size + setB.size - inter
   898→   898→  const pcScore = union > 0 ? inter / union : 1
   899→   899→
   900→   900→  return 0.4 * contourScore + 0.35 * intervalScore + 0.25 * pcScore
   901→   901→}
   902→   902→
   903→   903→/** True if `b` is recognizably derived from `a` (similarity above threshold). */
   904→   904→export function isMaterialRelated(a: PhraseMaterial, b: PhraseMaterial, threshold = 0.4): boolean {
   905→   905→  return materialSimilarity(a, b) >= threshold
   906→   906→}
