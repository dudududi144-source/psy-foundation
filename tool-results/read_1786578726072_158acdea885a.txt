     1→     1→/**
     2→     2→ * Forensic Offline Renderer — the deterministic render() function.
     3→     3→ *
     4→     4→ * render(seed, worldId, duration, options) => { samplesL, samplesR, events }
     5→     5→ *
     6→     6→ * This is the CRITICAL piece of the forensic infrastructure. It:
     7→     7→ *   1. Generates a musical timeline (events) from (seed, world)
     8→     8→ *   2. Renders events through the voice pool + bus/master/FX chain
     9→     9→ *   3. Returns stereo PCM as Float32Array
    10→    10→ *
    11→    11→ * DETERMINISM: Same seed + same world + same params => same output, always.
    12→    12→ * No Math.random(), no Date.now(), no performance.now(). Everything flows
    13→    13→ * from the seeded Rng.
    14→    14→ *
    15→    15→ * ISOMORPHIC: Runs in Node.js and browser. No Web Audio dependencies.
    16→    16→ *
    17→    17→ * ISOLATION: Supports rendering specific voices only (e.g., bass-only,
    18→    18→ * kick-only) for the bass isolation and kick/bass masking tests.
    19→    19→ */

   107→   107→// ─── Event generation (the musical brain) ──────────────────────────────────
   108→   108→
   109→   109→/**
   110→   110→ * Generate the complete musical timeline for a render.
   111→   111→ * (Standalone port of the original psy4LiveEngine step() logic.)
   112→   112→ */
   113→   113→export function generateEvents(
   114→   114→  seed: number,
   115→   115→  world: Psy4World,
   116→   116→  duration: number,
   117→   117→  macros: ForensicMacros = DEFAULT_FORENSIC_MACROS,
   118→   118→): MusicEvent[] {
   119→   119→  const events: MusicEvent[] = [];
   120→   120→  const s16 = 60 / world.bpm / 4; // seconds per 16th note
   121→   121→
   122→   122→  let sectionIdx = 0;
   123→   123→  let si = 0;       // step index within section
   124→   124→  let nextTime = 0; // absolute time of next step
   125→   125→
   126→   126→  // Build first section
   127→   127→  let S = buildSectionState(
   128→   128→    ARRANGEMENT[0], seed, sectionIdx,
   129→   129→    world.root, world.scale, world.acid, world.bass,
   130→   130→  );
   131→   131→
   132→   132→  const e = macros.energy;
   133→   133→  const psy = macros.psychedelia;
   134→   134→  const dens = macros.density;
   135→   135→  const sw = world.swing * macros.groove;
   136→   136→
   137→   137→  while (nextTime < duration) {
   138→   138→    const t = nextTime;
   139→   139→    const sb = si % 16;           // step within bar
   140→   140→    const bar = Math.floor(si / 16);
   141→   141→
   142→   142→    // ── Section automation ──
   143→   143→    const isPreDrop = S.label.includes('BUILD') && bar >= S.bars - 2;
   144→   144→    const isDropStart = S.bassOn && S.leadOn && bar === 0;
   145→   145→    const isPreTransition = bar === S.bars - 1;
   146→   146→    const isBuildClimax = S.label.includes('BUILD') && bar >= S.bars - 4;
   147→   147→
   148→   148→    // Riser before drop
   149→   149→    if (isPreDrop && sb === 0) {
   150→   150→      events.push({ time: t, voice: V_RISER, note: 0, velocity: 0.3, duration: s16 * 32, param: 0 });
   151→   151→    }
   152→   152→    if (isBuildClimax && bar === S.bars - 4 && sb === 0) {
   153→   153→      events.push({ time: t, voice: V_SWEEP, note: 0, velocity: 0.2, duration: s16 * 16, param: 0 });
   154→   154→    }
   155→   155→    if (isDropStart && sb === 0) {
   156→   156→      events.push({ time: t, voice: V_IMPACT, note: 0, velocity: 0.4, duration: 0.5, param: 0 });
   157→   157→      events.push({ time: t + s16 * 2, voice: V_IMPACT, note: 0, velocity: 0.3, duration: 0.5, param: 0 });
   158→   158→    }
   159→   159→    if (!S.bassOn && bar === 0 && sb === 0) {
   160→   160→      events.push({ time: t, voice: V_SWEEP, note: 0, velocity: 0.2, duration: s16 * 32, param: 0 });
   161→   161→    }
   162→   162→    if (isPreTransition && sb === 12 && S.bassOn) {
   163→   163→      const sweepLen = S.label.includes('BUILD') ? s16 * 8 : s16 * 4;
   164→   164→      events.push({ time: t, voice: V_SWEEP, note: 0, velocity: 0.2, duration: sweepLen, param: 0 });
   165→   165→    }
   166→   166→    if (isDropStart && sb === 4) {
   167→   167→      events.push({ time: t, voice: V_DOWNLIFTER, note: 0, velocity: 0.2, duration: 0.1 + macros.energy * 0.05, param: 0 });
   168→   168→    }
   169→   169→    const hatsMuted = (S.label.includes('BUILD') && bar >= S.bars - 2) ||
   170→   170→                      (S.bassOn && S.leadOn && bar >= S.bars - 2 && S.label !== 'FINAL DROP');
   171→   171→
   172→   172→    // ── PAD (chord progression, every 2 bars) ──
   173→   173→    if (sb === 0 && bar % 2 === 0) {
   174→   174→      const progs = CHORD_PROGRESSIONS[world.scale] || CHORD_PROGRESSIONS.minor;
   175→   175→      const chord = progs[(bar / 2) % progs.length];
   176→   176→      const padAmp = 0.12 * (0.5 + e * 0.5) * (!S.bassOn ? 1.5 : 0.8);
   177→   177→      // Play each chord tone
   178→   178→      for (const degree of chord) {
   179→   179→        const note = scaleNote(world.root - 12, world.scale, degree);
   180→   180→        events.push({ time: t, voice: V_PAD, note, velocity: padAmp, duration: s16 * 32, param: 0 });
   181→   181→      }
   182→   182→    }
   183→   183→
   184→   184→    // ── TEXTURE ──
   185→   185→    if (sb === 0 && bar % 4 === 0 && S.bassOn || (sb === 0 && bar % 4 === 0 && S.leadOn)) {
   186→   186→      events.push({ time: t, voice: V_TEXTURE, note: 0, velocity: world.textureLevel * (0.5 + psy * 0.5), duration: s16 * 64, param: 0 });
   187→   187→    }
   188→   188→    if (sb === 0 && bar % 2 === 1 && (S.bassOn && S.leadOn)) {
   189→   189→      events.push({ time: t, voice: V_TEXTURE, note: 0, velocity: world.textureLevel * 0.4 * psy, duration: s16 * 32, param: 0 });
   190→   190→    }
   191→   191→
   192→   192→    // ── KICK (4 on floor) ──
   193→   193→    if (sb % 4 === 0) {
   194→   194→      const isDownbeat = sb === 0;
   195→   195→      const kickVel = isDownbeat ? 0.5 + e * 0.05 : 0.42 + e * 0.08;
   196→   196→      events.push({ time: t, voice: V_KICK, note: 0, velocity: kickVel, duration: 0.3, param: 0 });
   197→   197→    }
   198→   198→    if (S.bassOn && S.leadOn && sb === 14 && S.rng.chance(0.3 * dens)) {
   199→   199→      events.push({ time: t, voice: V_KICK, note: 0, velocity: 0.15, duration: 0.2, param: 0 });
   200→   200→    }
   201→   201→
   202→   202→    // ── BASS ──
   203→   203→    const isOff = sb % 2 === 1;
   204→   204→    const bt = isOff ? t + sw * s16 : t;
   205→   205→    const bassPatterns = BASS_PATTERNS[world.bass] || BASS_PATTERNS.off;
   206→   206→    const bassPatternRotIdx = (S.bassPatternIdx + Math.floor(bar / 4)) % bassPatterns.length;
   207→   207→    const bassPattern = bassPatterns[bassPatternRotIdx];
   208→   208→    const patternStep = Math.floor(sb / 2) % bassPattern.steps.length;
   209→   209→    let bassDegree = bassPattern.steps[patternStep];
   210→   210→    let bassAccent = bassPattern.accents[patternStep];
   211→   211→
   212→   212→    if (bar % 2 === 1 && sb === 6 && S.rng.chance(0.4)) {
   213→   213→      bassDegree = S.rng.pick([2, 4, 7]);
   214→   214→      bassAccent = 0.6;
   215→   215→    }
   216→   216→    if (bar % 4 === 3 && sb === 14) {
   217→   217→      bassDegree = 7;
   218→   218→      bassAccent = 0.8;
   219→   219→    }
   220→   220→
   221→   221→    let bassOn = bassDegree >= 0 && bassAccent > 0;
   222→   222→    if (world.bass === 'roll') bassOn = bassOn && (isOff || sb % 4 === 0);
   223→   223→    else if (world.bass === 'off') bassOn = bassOn && sb % 4 === 2;
   224→   224→    else if (world.bass === 'acid') bassOn = bassOn && (isOff || sb === 0);
   225→   225→
   226→   226→    if (isPreDrop) bassOn = false;
   227→   227→    if (!S.bassOn) bassOn = false;
   228→   228→
   229→   229→    if (bassOn) {
   230→   230→      const bassNote = scaleNote(world.root, world.scale, bassDegree);
   231→   231→      let bassVel = bassAccent * (0.35 + e * 0.15);
   232→   232→      let bassDur = s16 * 0.9;
   233→   233→      const isGhost = (bar % 2 === 1 && sb === 0 && S.rng.chance(0.3));
   234→   234→      if (isGhost) { bassVel = 0.2; bassDur = s16 * 0.4; }
   235→   235→      if (bar % 8 === 7 && sb === 0) {
   236→   236→        bassDur = s16 * 2.5;
   237→   237→        bassVel *= 1.1;
   238→   238→      }
   239→   239→      events.push({ time: bt, voice: V_BASS, note: bassNote, velocity: bassVel, duration: bassDur, param: world.acid ? 1 : 0 });
   240→   240→    }
   241→   241→
   242→   242→    // ── ACID LINE ──
   243→   243→    if (world.acid && S.bassOn && S.leadOn && sb % 2 === 0 && S.rng.chance(0.5 * psy)) {
   244→   244→      if (S.acidPattern) {
   245→   245→        const acidNote = S.acidPattern.next();
   246→   246→        if (acidNote !== null) {
   247→   247→          events.push({ time: t, voice: V_ACID, note: acidNote, velocity: 0.15 + psy * 0.1, duration: s16 * 1.5, param: 0 });
   248→   248→        }
   249→   249→      }
   250→   250→    }
   251→   251→
   252→   252→    // ── HATS ──
   253→   253→    if (!hatsMuted && world.hatPattern[sb] === 'x') {
   254→   254→      const beatPos = sb % 4;
   255→   255→      let hatVel: number;
   256→   256→      if (beatPos === 0) hatVel = 0.25;
   257→   257→      else if (beatPos === 2) hatVel = 0.18;
   258→   258→      else hatVel = 0.12;
   259→   259→      if (bar % 4 === 3) hatVel *= 1.2;
   260→   260→      hatVel *= (0.5 + dens * 0.5);
   261→   261→      events.push({ time: t + (sb % 4 === 2 ? sw * s16 : 0), voice: V_HAT, note: 0, velocity: hatVel, duration: 0.05, param: 0 });
   262→   262→    }
   263→   263→    if (!hatsMuted && world.hatPattern[sb] === '.' && sb % 2 === 0 && S.rng.chance(0.15 * dens) && S.bassOn) {
   264→   264→      events.push({ time: t, voice: V_HAT, note: 0, velocity: 0.04 + dens * 0.02, duration: 0.04, param: 0 });
   265→   265→    }
   266→   266→
   267→   267→    // ── PERC ──
   268→   268→    if (world.percPattern[sb] === 'x' && S.rng.chance(S.percDensity * dens)) {
   269→   269→      events.push({ time: t, voice: V_PERC, note: 400 + S.rng.int(0, 200), velocity: 0.2, duration: 0.1, param: 0 });
   270→   270→    }
   271→   271→
   272→   272→    // ── LEAD ──
   273→   273→    if (S.leadOn && S.leadMotif && sb % 2 === 0) {
   274→   274→      const result = S.leadMotif.nextNote(sb, bar, S.energy, S.rng);
   275→   275→      if (result) {
   276→   276→        events.push({ time: t, voice: V_LEAD, note: result.note, velocity: result.velocity * 0.5, duration: s16 * 1.5, param: 0 });
   277→   277→      }
   278→   278→    }
   279→   279→
   280→   280→    // ── CLAP (on 2 and 4 in drops) ──
   281→   281→    if (S.bassOn && S.leadOn && (sb === 4 || sb === 12)) {
   282→   282→      events.push({ time: t, voice: V_CLAP, note: 0, velocity: 0.25, duration: 0.15, param: 0 });
   283→   283→    }
   284→   284→
   285→   285→    // ── SHAKER (offbeat, in drops) ──
   286→   286→    if (S.bassOn && S.leadOn && sb % 2 === 1 && S.rng.chance(0.6)) {
   287→   287→      events.push({ time: t, voice: V_SHAKER, note: 0, velocity: 0.12, duration: 0.05, param: 0 });
   288→   288→    }
   289→   289→
   290→   290→    // Advance step
   291→   291→    si++;
   292→   292→    nextTime += s16;
   293→   293→    if (si >= S.bars * 16) {
   294→   294→      sectionIdx++;
   295→   295→      const nextArr = ARRANGEMENT[sectionIdx % ARRANGEMENT.length];
   296→   296→      S = buildSectionState(
   297→   297→        nextArr, seed, sectionIdx,
   298→   298→        world.root, world.scale, world.acid, world.bass,
   299→   299→      );
   300→   300→      si = 0;
   301→   301→    }
   302→   302→  }
   303→   303→
   304→   304→  return events;
   305→   305→}

   307→   307→// ─── Render (process events through DSP) ───────────────────────────────────
   308→   308→
   309→   309→export function render(
   310→   310→  seed: number,
   311→   311→  worldId: string,
   312→   312→  duration: number,
   313→   313→  options: RenderOptions = {},
   314→   314→): RenderResult {
   315→   315→  const world: Psy4World = {
   316→   316→    ...FORENSIC_WORLDS[worldId],
   317→   317→    ...options.paramOverrides,
   318→   318→  };
   319→   319→  const macros = { ...DEFAULT_FORENSIC_MACROS, ...options.macros };
   320→   320→  const sr = SR;
   321→   321→  const totalSamples = Math.floor(duration * sr);
   322→   322→
   323→   323→  // 1. Generate events
   324→   324→  const events = generateEvents(seed, world, duration, macros);
   325→   325→
   326→   326→  // 2. Create voice pools (all deterministic)
   327→   327→  const poolRng = new Rng(seed ^ 0xDEAD);
   328→   328→  const pool = createVoicePool(poolRng);
   329→   329→
   330→   330→  // 3. Create mixing infrastructure
   331→   331→  const drumConfig: Partial<BusConfig> = {
   332→   332→    hpFreq: 0, compThr: 0.5, compRatio: 3, compAtt: 0.002, compRel: 0.08,
   333→   333→    compMakeup: 1.4, drive: 1.4, gain: 1.0,
   334→   334→  };
   335→   335→  const bassConfig: Partial<BusConfig> = {
   336→   336→    hpFreq: 40, compThr: 0.4, compRatio: 2, compAtt: 0.005, compRel: 0.12,
   337→   337→    compMakeup: 1.2, drive: 1.2, gain: 1.0,
   338→   338→  };
   339→   339→  const musicConfig: Partial<BusConfig> = {
   340→   340→    hpFreq: 80, compThr: 0.45, compRatio: 2, compAtt: 0.01, compRel: 0.15,
   341→   341→    compMakeup: 1.1, drive: 1.15, gain: 1.0,
   342→   342→  };
   343→   343→  const atmosConfig: Partial<BusConfig> = {
   344→   344→    hpFreq: 60, compThr: 0, drive: 1.0, gain: 1.0,
   345→   345→  };
   346→   346→  const fxConfig: Partial<BusConfig> = {
   347→   347→    hpFreq: 40, compThr: 0.35, compRatio: 2.5, compAtt: 0.003, compRel: 0.1,
   348→   348→    compMakeup: 1.2, drive: 1.2, gain: 1.0,
   349→   349→  };
   350→   350→
   351→   351→  const drumProcL = new BusProcessor(drumConfig);
   352→   352→  const drumProcR = new BusProcessor(drumConfig);
   353→   353→  const bassProcL = new BusProcessor(bassConfig);
   354→   354→  const bassProcR = new BusProcessor(bassConfig);
   355→   355→  const musicProcL = new BusProcessor(musicConfig);
   356→   356→  const musicProcR = new BusProcessor(musicConfig);
   357→   357→  const atmosProcL = new BusProcessor(atmosConfig);
   358→   358→  const atmosProcR = new BusProcessor(atmosConfig);
   359→   359→  const fxProcL = new BusProcessor(fxConfig);
   360→   360→  const fxProcR = new BusProcessor(fxConfig);
   361→   361→
   362→   362→  const masterL = new MasterChain();
   363→   363→  const masterR = new MasterChain();
   364→   364→
   365→   365→  const reverb = new SchroederReverb();
   366→   366→  const delay = new StereoDelay();
   367→   367→
   368→   368→  // Section-aware FX sends
   369→   369→  let reverbSends = [0.08, 0.02, 0.25, 0.40, 0.30];
   370→   370→  let delaySends = [0.05, 0.0, 0.20, 0.10, 0.15];
   371→   371→
   372→   372→  // Bus gains — REBALANCED + now optimizable via level parameters
   373→   373→  // Base gains provide good balance; level params scale them
   374→   374→  const baseBusGains = [0.45, 1.8, 1.5, 0.85, 0.65];
   375→   375→  const busGains = [
   376→   376→    baseBusGains[0] * (world.kickLevel ?? 1.0),   // drum bus (kick, hat, clap, perc)
   377→   377→    baseBusGains[1] * (world.bassLevel ?? 1.0),   // bass bus
   378→   378→    baseBusGains[2] * (world.leadLevel ?? 1.0),   // music bus (lead, acid)
   379→   379→    baseBusGains[3],                                // atmos bus (pad, texture)
   380→   380→    baseBusGains[4],                                // fx bus
   381→   381→  ];
   382→   382→  const masterLevel = world.masterLevel ?? 0.85;
   383→   383→
   384→   384→  // Sidechain
   385→   385→  let duckEnv = 1.0;
   386→   386→  const duckDepth = world.duck;
   387→   387→  // FIX: was 0.08 (80ms) — too fast, bass plays on offbeat (117ms after kick)
   388→   388→  // so the duck recovered before the bass played, making duck a DEAD parameter.
   389→   389→  // 250ms release keeps the bass ducked through the offbeat.
   390→   390→  const duckRelease = 0.25;
   391→   391→
   392→   392→  // Lead Haas delay buffer
   393→   393→  const leadDelayBuf = new Float32Array(18);
   394→   394→  let leadDelayIdx = 0;
   395→   395→
   396→   396→  // Only-voices filter (for isolation rendering)
   397→   397→  const onlyVoices = options.onlyVoices ? new Set(options.onlyVoices) : null;
   398→   398→
   399→   399→  // 4. Process events: sort by time, trigger voices
   400→   400→  //    We'll process events inline during sample rendering
   401→   401→  let eventIdx = 0;
   402→   402→  const sortedEvents = [...events].sort((a, b) => a.time - b.time);
   403→   403→
   404→   404→  function triggerVoice(ev: MusicEvent): void {
   405→   405→    if (onlyVoices && !onlyVoices.has(ev.voice)) return;
   406→   406→    const sr = SR;
   407→   407→    const wp = world;
   408→   408→    const mc = macros;
   409→   409→
   410→   410→    switch (ev.voice) {
   411→   411→      case V_KICK: {
   412→   412→        const v = getFreeVoice(pool.kick);
   413→   413→        v.trigger(0, ev.velocity, wp.kickFundamental, wp.kickDecay, sr);
   414→   414→        // Trigger sidechain
   415→   415→        duckEnv = 1 - duckDepth * 0.7 * (0.5 + mc.aggression * 0.5);
   416→   416→        break;
   417→   417→      }
   418→   418→      case V_BASS: {
   419→   419→        const v = getFreeVoice(pool.bass);
   420→   420→        const acid = ev.param >= 0.5;
   421→   421→        const freq = mtof(ev.note);
   422→   422→        const params: BassParams = {
   423→   423→          cutoffStart: Math.min(2000, wp.bassCutoff * 4),
   424→   424→          cutoffEnd: wp.bassCutoff,
   425→   425→          resonance: wp.bassResonance,
   426→   426→        };
   427→   427→        v.trigger(0, freq, ev.duration, ev.velocity, acid, sr, params);
   428→   428→        break;
   429→   429→      }
   430→   430→      case V_LEAD: {
   431→   431→        const v = getFreeVoice(pool.lead);
   432→   432→        const freq = mtof(ev.note);
   433→   433→        const params: LeadParams = {
   434→   434→          cutoff: wp.leadCutoff * (0.7 + mc.brightness * 0.6),
   435→   435→          detune: wp.leadDetune * (0.5 + mc.psychedelia),
   436→   436→          resonance: 2 + mc.psychedelia * 3,
   437→   437→          lfoRate: 0.5 + mc.psychedelia * 3,
   438→   438→          lfoDepth: mc.psychedelia * 0.3,
   439→   439→        };
   440→   440→        v.trigger(0, freq, ev.duration, ev.velocity, sr, params);
   441→   441→        break;
   442→   442→      }
   443→   443→      case V_ACID: {
   444→   444→        const v = getFreeVoice(pool.acid);
   445→   445→        const freq = mtof(ev.note);
   446→   446→        v.trigger(0, freq, ev.duration, ev.velocity, sr);
   447→   447→        break;
   448→   448→      }
   449→   449→      case V_PAD: {
   450→   450→        const v = getFreeVoice(pool.pad);
   451→   451→        const freq = mtof(ev.note);
   452→   452→        const params: PadParams = {
   453→   453→          cutoff: wp.padCutoff, attack: 0.5, detune: 7, evolveRate: 0.1,
   454→   454→        };
   455→   455→        v.trigger(0, freq, ev.duration, ev.velocity, sr, params);
   456→   456→        break;
   457→   457→      }
   458→   458→      case V_HAT: {
   459→   459→        const v = getFreeVoice(pool.hat);
   460→   460→        v.trigger(0, false, ev.velocity, sr);
   461→   461→        break;
   462→   462→      }
   463→   463→      case V_CLAP: {
   464→   464→        const v = getFreeVoice(pool.clap);
   465→   465→        v.trigger(0, ev.velocity, sr);
   466→   466→        break;
   467→   467→      }
   468→   468→      case V_PERC: {
   469→   469→        const v = getFreeVoice(pool.perc);
   470→   470→        v.trigger(0, ev.note, ev.velocity, sr);
   471→   471→        break;
   472→   472→      }
   473→   473→      case V_SHAKER: {
   474→   474→        const v = getFreeVoice(pool.shaker);
   475→   475→        v.trigger(0, ev.velocity, sr);
   476→   476→        break;
   477→   477→      }
   478→   478→      case V_TEXTURE: {
   479→   479→        const v = getFreeVoice(pool.texture);
   480→   480→        v.trigger(0, ev.duration, ev.velocity, world.textureType, sr);
   481→   481→        break;
   482→   482→      }
   483→   483→      case V_RISER: case V_IMPACT: case V_SWEEP: case V_DOWNLIFTER: {
   484→   484→        const v = getFreeVoice(pool.fx);
   485→   485→        v.trigger(ev.voice, 0, ev.duration, ev.velocity, sr);
   486→   486→        break;
   487→   487→      }
   488→   488→    }
   489→   489→  }
   490→   490→
   491→   491→  // 5. Render sample by sample
   492→   492→  const samplesL = new Float32Array(totalSamples);
   493→   493→  const samplesR = new Float32Array(totalSamples);
   494→   494→  const dt = 1 / sr;
   495→   495→
   496→   496→  for (let i = 0; i < totalSamples; i++) {
   497→   497→    const currentTime = i / sr;
   498→   498→
   499→   499→    // Trigger any events that are due
   500→   500→    while (eventIdx < sortedEvents.length && sortedEvents[eventIdx].time <= currentTime) {
   501→   501→      triggerVoice(sortedEvents[eventIdx]);
   502→   502→      eventIdx++;
   503→   503→    }
   504→   504→
   505→   505→    // Sidechain recovery
   506→   506→    if (duckEnv < 1) {
   507→   507→      duckEnv += (1 - duckEnv) * (dt / duckRelease);
   508→   508→    }
   509→   509→
   510→   510→    // Mix voices into buses
   511→   511→    let drumBusL = 0, drumBusR = 0;
   512→   512→    let bassBusL = 0, bassBusR = 0;
   513→   513→    let musicBusL = 0, musicBusR = 0;
   514→   514→    let atmosBusL = 0, atmosBusR = 0;
   515→   515→    let fxBusL = 0, fxBusR = 0;
   516→   516→
   517→   517→    // Kick → drum bus (mono)
   518→   518→    for (const v of pool.kick) {
   519→   519→      if (v.active) {
   520→   520→        const [s] = v.render();
   521→   521→        drumBusL += s; drumBusR += s;
   522→   522→      }
   523→   523→    }
   524→   524→    // Hat → drum bus (mono)
   525→   525→    for (const v of pool.hat) {
   526→   526→      if (v.active) {
   527→   527→        const [s] = v.render();
   528→   528→        drumBusL += s; drumBusR += s;
   529→   529→      }
   530→   530→    }
   531→   531→    // Clap → drum bus (mono)
   532→   532→    for (const v of pool.clap) {
   533→   533→      if (v.active) {
   534→   534→        const [s] = v.render();
   535→   535→        drumBusL += s; drumBusR += s;
   536→   536→      }
   537→   537→    }
   538→   538→    // Perc → drum bus (mono)
   539→   539→    for (const v of pool.perc) {
   540→   540→      if (v.active) {
   541→   541→        const [s] = v.render();
   542→   542→        drumBusL += s; drumBusR += s;
   543→   543→      }
   544→   544→    }
   545→   545→    // Shaker → drum bus (mono)
   546→   546→    for (const v of pool.shaker) {
   547→   547→      if (v.active) {
   548→   548→        const [s] = v.render();
   549→   549→        drumBusL += s; drumBusR += s;
   550→   550→      }
   551→   551→    }
   552→   552→
   553→   553→    // Bass → bass bus (mono, sidechain ducked)
   554→   554→    for (const v of pool.bass) {
   555→   555→      if (v.active) {
   556→   556→        const [s] = v.render();
   557→   557→        const ducked = s * duckEnv;
   558→   558→        bassBusL += ducked; bassBusR += ducked;
   559→   559→      }
   560→   560→    }
   561→   561→
   562→   562→    // Lead → music bus (Haas stereo)
   563→   563→    for (const v of pool.lead) {
   564→   564→      if (v.active) {
   565→   565→        const [s] = v.render();
   566→   566→        musicBusL += s;
   567→   567→        const delayed = leadDelayBuf[leadDelayIdx];
   568→   568→        leadDelayBuf[leadDelayIdx] = s;
   569→   569→        leadDelayIdx = (leadDelayIdx + 1) % 18;
   570→   570→        musicBusR += delayed;
   571→   571→      }
   572→   572→    }
   573→   573→    // Acid → music bus (mono)
   574→   574→    for (const v of pool.acid) {
   575→   575→      if (v.active) {
   576→   576→        const [s] = v.render();
   577→   577→        musicBusL += s; musicBusR += s;
   578→   578→      }
   579→   579→    }
   580→   580→
   581→   581→    // Pad → atmos bus (stereo amplitude LFO)
   582→   582→    for (const v of pool.pad) {
   583→   583→      if (v.active) {
   584→   584→        const [s] = v.render();
   585→   585→        const lfo = Math.sin(i * 0.0008);
   586→   586→        atmosBusL += s * (0.85 + lfo * 0.15);
   587→   587→        atmosBusR += s * (0.85 - lfo * 0.15);
   588→   588→      }
   589→   589→    }
   590→   590→    // Texture → atmos bus (stereo pan)
   591→   591→    for (const v of pool.texture) {
   592→   592→      if (v.active) {
   593→   593→        const [s] = v.render();
   594→   594→        const pan = Math.sin(i * 0.0005);
   595→   595→        atmosBusL += s * (0.5 - pan * 0.3);
   596→   596→        atmosBusR += s * (0.5 + pan * 0.3);
   597→   597→      }
   598→   598→    }
   599→   599→
   600→   600→    // FX → fx bus (mono)
   601→   601→    for (const v of pool.fx) {
   602→   602→      if (v.active) {
   603→   603→        const [s] = v.render();
   604→   604→        fxBusL += s; fxBusR += s;
   605→   605→      }
   606→   606→    }
   607→   607→
   608→   608→    // Bus processing (separate L/R) — call .process on the BusProcessor
   609→   609→    // instances, NOT on the numeric accumulator variables.
   610→   610→    drumBusL = drumProcL.process(drumBusL, sr);
   611→   611→    drumBusR = drumProcR.process(drumBusR, sr);
   612→   612→    bassBusL = bassProcL.process(bassBusL, sr);
   613→   613→    bassBusR = bassProcR.process(bassBusR, sr);
   614→   614→    musicBusL = musicProcL.process(musicBusL, sr);
   615→   615→    musicBusR = musicProcR.process(musicBusR, sr);
   616→   616→    atmosBusL = atmosProcL.process(atmosBusL, sr);
   617→   617→    atmosBusR = atmosProcR.process(atmosBusR, sr);
   618→   618→    fxBusL = fxProcL.process(fxBusL, sr);
   619→   619→    fxBusR = fxProcR.process(fxBusR, sr);
   620→   620→
   621→   621→    // Sum buses
   622→   622→    let mixL = drumBusL * busGains[0] + bassBusL * busGains[1] + musicBusL * busGains[2] + atmosBusL * busGains[3] + fxBusL * busGains[4];
   623→   623→    let mixR = drumBusR * busGains[0] + bassBusR * busGains[1] + musicBusR * busGains[2] + atmosBusR * busGains[3] + fxBusR * busGains[4];
   624→   624→
   625→   625→    // FX sends
   626→   626→    if (!options.dry) {
   627→   627→      const reverbInput = (drumBusL + drumBusR) * 0.5 * reverbSends[0]
   628→   628→                        + (bassBusL + bassBusR) * 0.5 * reverbSends[1]
   629→   629→                        + (musicBusL + musicBusR) * 0.5 * reverbSends[2]
   630→   630→                        + (atmosBusL + atmosBusR) * 0.5 * reverbSends[3]
   631→   631→                        + (fxBusL + fxBusR) * 0.5 * reverbSends[4];
   632→   632→      const [revL, revR] = reverb.process(reverbInput, sr);
   633→   633→
   634→   634→      const delayInputL = drumBusL * delaySends[0] + bassBusL * delaySends[1] + musicBusL * delaySends[2] + atmosBusL * delaySends[3] + fxBusL * delaySends[4];
   635→   635→      const delayInputR = drumBusR * delaySends[0] + bassBusR * delaySends[1] + musicBusR * delaySends[2] + atmosBusR * delaySends[3] + fxBusR * delaySends[4];
   636→   636→      const [delL, delR] = delay.process(delayInputL, delayInputR, sr);
   637→   637→
   638→   638→      mixL += revL + delL;
   639→   639→      mixR += revR + delR;
   640→   640→    }
   641→   641→
   642→   642→    // Guard against NaN/Infinity from feedback loops (reverb, delay, filter resonance)
   643→   643→    if (!isFinite(mixL)) mixL = 0;
   644→   644→    if (!isFinite(mixR)) mixR = 0;
   645→   645→
   646→   646→    // Apply master level (optimizable)
   647→   647→    mixL *= masterLevel;
   648→   648→    mixR *= masterLevel;
   649→   649→
   650→   650→    // Master chain (separate L/R)
   651→   651→    mixL = masterL.process(mixL, sr);
   652→   652→    mixR = masterR.process(mixR, sr);
   653→   653→
   654→   654→    // Final safety clamp
   655→   655→    samplesL[i] = isFinite(mixL) ? Math.max(-1, Math.min(1, mixL)) : 0;
   656→   656→    samplesR[i] = isFinite(mixR) ? Math.max(-1, Math.min(1, mixR)) : 0;
   657→   657→  }
   658→   658→
   659→   659→  return {
   660→   660→    samplesL,
   661→   661→    samplesR,
   662→   662→    sampleRate: sr,
   663→   663→    duration,
   664→   664→    events,
   665→   665→    world,
   666→   666→    seed,
   667→   667→    options,
   668→   668→  };
   669→   669→}

   671→   671→// ─── WAV encoding (for downloads / API responses) ──────────────────────────
   672→   672→
   673→   673→export function encodeWav(samplesL: Float32Array, samplesR: Float32Array, sr: number): ArrayBuffer {
   674→   674→  const numSamples = samplesL.length;
   675→   675→  const buffer = new ArrayBuffer(44 + numSamples * 4);
   676→   676→  const view = new DataView(buffer);
   677→   677→
   678→   678→  const writeString = (offset: number, str: string) => {
   679→   679→    for (let i = 0; i < str.length; i++) view.setUint8(offset + i, str.charCodeAt(i));
   680→   680→  };
   681→   681→
   682→   682→  writeString(0, 'RIFF');
   683→   683→  view.setUint32(4, 36 + numSamples * 4, true);
   684→   684→  writeString(8, 'WAVE');
   685→   685→  writeString(12, 'fmt ');
   686→   686→  view.setUint32(16, 16, true);
   687→   687→  view.setUint16(20, 1, true);    // PCM
   688→   688→  view.setUint16(22, 2, true);    // stereo
   689→   689→  view.setUint32(24, sr, true);
   690→   690→  view.setUint32(28, sr * 4, true);
   691→   691→  view.setUint16(32, 4, true);
   692→   692→  view.setUint16(34, 16, true);
   693→   693→  writeString(36, 'data');
   694→   694→  view.setUint32(40, numSamples * 4, true);
   695→   695→
   696→   696→  let offset = 44;
   697→   697→  for (let i = 0; i < numSamples; i++) {
   698→   698→    const l = Math.max(-1, Math.min(1, samplesL[i]));
   699→   699→    const r = Math.max(-1, Math.min(1, samplesR[i]));
   700→   700→    view.setInt16(offset, l < 0 ? l * 0x8000 : l * 0x7fff, true);
   701→   701→    view.setInt16(offset + 2, r < 0 ? r * 0x8000 : r * 0x7fff, true);
   702→   702→    offset += 4;
   703→   703→  }
   704→   704→
   705→   705→  return buffer;
   706→   706→}

   708→   708→// ─── Mono downmix ──────────────────────────────────────────────────────────
   709→   709→
   710→   710→export function downmixToMono(samplesL: Float32Array, samplesR: Float32Array): Float32Array {
   711→   711→  const n = samplesL.length;
   712→   712→  const mono = new Float32Array(n);
   713→   713→  for (let i = 0; i < n; i++) {
   714→   714→    mono[i] = (samplesL[i] + samplesR[i]) * 0.5;
   715→   715→  }
   716→   716→  return mono;
   717→   717→}
