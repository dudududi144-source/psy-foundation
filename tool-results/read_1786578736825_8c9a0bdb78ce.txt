    36→export type Category = 'drum' | 'bass' | 'lead' | 'pad' | 'pluck' | 'arp' | 'fx' | 'texture';
    37→
    38→export interface SoundPreset {
    39→  id: string;
    40→  name: string;
    41→  genre: Genre;
    42→  cat: Category;
    43→  engine: EngineType;
    44→
    45→  // Drum-specific
    46→  drumType?: DrumType;
    47→  tune?: number;
    48→  decay?: number;
    49→  tone?: number;
    50→  punch?: number;
    51→
    52→  // Synth oscillators
    53→  wave1?: OscillatorType;
    54→  wave2?: OscillatorType;
    55→  oct2?: number;       // octave offset for osc2
    56→  detune?: number;     // cents
    57→  fmAmount?: number;   // for FM engine
    58→  fmRatio?: number;    // carrier:modulator ratio
    59→
    60→  // Filter
    61→  fType?: FilterType;
    62→  cutoff?: number;
    63→  res?: number;
    64→  fEnvAmt?: number;    // filter envelope amount
    65→  fDecay?: number;     // filter envelope decay
    66→
    67→  // Amp envelope (ADSR)
    68→  atk?: number;
    69→  dec?: number;
    70→  sus?: number;
    71→  rel?: number;
    72→  gate?: number;       // note gate (multiplier of step duration)
    73→
    74→  // LFO
    75→  lfoRate?: number;
    76→  lfoDepth?: number;
    77→  lfoDest?: LFODest;
    78→  lfoShape?: OscillatorType;
    79→
    80→  // Effects sends
    81→  sendDelay?: number;  // 0-1
    82→  sendReverb?: number; // 0-1
    83→
    84→  // Polyphony
    85→  poly?: number;
    86→
    87→  // Velocity sensitivity
    88→  velSens?: number;    // 0-1
    89→
    90→  // Musical context (for learning system auto-selection)
    91→  scaleDegrees?: number[]; // which scale degrees this fits (e.g., [0, 4, 7] = root, 3rd, 5th)
    92→  energyLevel?: number;    // 0-1, how energetic
    93→  moodTags?: string[];     // ['dark', 'hypnotic', 'uplifting', 'aggressive', 'ethereal']
    94→}
