    13→ *   - Filter (type, cutoff, resonance, envelope)

    15→ *   - Effects sends (delay, reverb)
    16→ *   - Genre tags (for auto-selection by learning system)

    19→// ─── Types ─────────────────────────────────────────────────────────────────
    20→export type EngineType = 'DRUM' | 'SYNTH' | 'FM' | 'NOISE' | 'WAVETABLE';
    21→export type DrumType = 'kick' | 'snare' | 'clap' | 'hatC' | 'hatO' | 'tom' | 'rim' | 'glitch' | 'shaker' | 'riser' | 'impact' | 'downlifter';
    22→export type FilterType = 'lowpass' | 'highpass' | 'bandpass' | 'notch' | 'allpass';
    23→export type Category = 'drum' | 'bass' | 'lead' | 'pad' | 'pluck' | 'arp' | 'fx' | 'texture';
    24→export type Genre = 'PSYTRANCE' | 'TECHNO' | 'TRANCE' | 'PROGRESSIVE' | 'DARK-PSY' | 'GOA' | 'ANY';
    25→export type LFODest = 'off' | 'cutoff' | 'pitch' | 'gain' | 'pan';
    26→
    27→export interface SoundPreset {
    28→  id: string;
    29→  name: string;
    30→  genre: Genre;
    31→  cat: Category;
    32→  engine: EngineType;
    33→
    34→  // Drum-specific
    35→  drumType?: DrumType;
    36→  tune?: number;
    37→  decay?: number;
    38→  tone?: number;
    39→  punch?: number;
    40→
    41→  // Synth oscillators
    42→  wave1?: OscillatorType;
    43→  wave2?: OscillatorType;
    44→  oct2?: number;       // octave offset for osc2
    45→  detune?: number;     // cents
    46→  fmAmount?: number;   // for FM engine
    47→  fmRatio?: number;    // carrier:modulator ratio
    48→
    49→  // Filter
    50→  fType?: FilterType;
    51→  cutoff?: number;
    52→  res?: number;
    53→  fEnvAmt?: number;    // filter envelope amount
    54→  fDecay?: number;     // filter envelope decay
    55→
    56→  // Amp envelope (ADSR)
    57→  atk?: number;
    58→  dec?: number;
    59→  sus?: number;
    60→  rel?: number;
    61→  gate?: number;       // note gate (multiplier of step duration)
    62→
    63→  // LFO
    64→  lfoRate?: number;
    65→  lfoDepth?: number;
    66→  lfoDest?: LFODest;
    67→  lfoShape?: OscillatorType;
    68→
    69→  // Effects sends
    70→  sendDelay?: number;  // 0-1
    71→  sendReverb?: number; // 0-1
    72→
    73→  // Polyphony
    74→  poly?: number;
    75→
    76→  // Velocity sensitivity
    77→  velSens?: number;    // 0-1
    78→
    79→  // Musical context (for learning system auto-selection)
    80→  scaleDegrees?: number[]; // which scale degrees this fits (e.g., [0, 4, 7] = root, 3rd, 5th)
    81→  energyLevel?: number;    // 0-1, how energetic
    82→  moodTags?: string[];     // ['dark', 'hypnotic', 'uplifting', 'aggressive', 'ethereal']
    83→}

   518→// ─── Query helpers ────────────────────────────────────────────────────────
   519→export function getByCategory(cat: Category): SoundPreset[] {
   520→  return SOUND_BANK.filter(p => p.cat === cat);
   521→}
   522→
   523→export function getByGenre(genre: Genre): SoundPreset[] {
   524→  return SOUND_BANK.filter(p => p.genre === genre || p.genre === 'ANY');
   525→}
   526→
   527→export function getById(id: string): SoundPreset | null {
   528→  return SOUND_BANK.find(p => p.id === id) || null;
   529→}
   530→
   531→export function getByMood(mood: string): SoundPreset[] {
   532→  return SOUND_BANK.filter(p => p.moodTags?.includes(mood));
   533→}
   534→
   535→export function getByEnergyLevel(min: number, max: number): SoundPreset[] {
   536→  return SOUND_BANK.filter(p => (p.energyLevel || 0) >= min && (p.energyLevel || 0) <= max);
   537→}
   538→
   539→/**
   540→ * Auto-select presets by detected scale + genre + energy.
   541→ * Used by the learning system to pick the best sounds for the
   542→ * current musical context.
   543→ */
   544→export function autoSelectPresets(
   545→  genre: Genre,
   546→  scaleDegrees: number[],
   547→  energyLevel: number,
   548→): Record<Category, SoundPreset[]> {
   549→  const candidates = getByGenre(genre).filter(p => {
   550→    const energyMatch = Math.abs((p.energyLevel || 0) - energyLevel) < 0.3;
   551→    const scaleMatch = (p.scaleDegrees || []).some(d => scaleDegrees.includes(d));
   552→    return energyMatch || scaleMatch;
   553→  });
   554→
   555→  const result: Record<Category, SoundPreset[]> = {
   556→    drum: [], bass: [], lead: [], pad: [], pluck: [], arp: [], fx: [], texture: [],
   557→  };
   558→
   559→  candidates.forEach(p => {
   560→    if (result[p.cat].length < 5) {
   561→      result[p.cat].push(p);
   562→    }
   563→  });
   564→
   565→  return result;
   566→}

   568→export function bankStats() {
   569→  const cats: Record<string, number> = {};
   570→  const genres: Record<string, number> = {};
   571→  SOUND_BANK.forEach(p => {
   572→    cats[p.cat] = (cats[p.cat] || 0) + 1;
   573→    genres[p.genre] = (genres[p.genre] || 0) + 1;
   574→  });
   575→  return { total: SOUND_BANK.length, cats, genres };
   576→}
