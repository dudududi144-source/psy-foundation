     1→/**
     2→ * CompositionEngine: the single authoritative composer.
     3→ *
     4→ * The engine composes whole phrases, not isolated bars. It follows a strict
     5→ * hierarchy:
     6→ *   1. Build GroovePlan (kick skeleton, subdivision, accents)
     7→ *   2. Determine harmonic plan (chord tones, tension notes)
     8→ *   3. Compose BASS against groove (bass knows where kick is)
     9→ *   4. Compose LEAD against bass+harmony (lead leaves space for bass,
    10→ *      stays in tessitura)
    11→ *   5. Arrange parts (some roles OFF in BREAK / INTRO / OUTRO sections)
    12→ *
    13→ * Bass composition rules:
    14→ *   - Bass ALWAYS places a ROOT on step 0 (beat 1) — the LOCKED invariant
    15→ *     that keeps bass-kick alignment high.
    16→ *   - In LOCKED mode: bass hits every kick step with ROOT.
    17→ *   - In COMPLEMENTARY mode: bass hits the gaps between kicks with
    18→ *     FIFTH / OCTAVE.
    19→ *   - Bass register: octave 2 (MIDI 36-59).
    20→ *   - Last bar of a phrase: CADENCE walk (fifth → root).
    21→ *
    22→ * Lead composition rules:
    23→ *   - Lead MUST NOT overlap bass register (lead clamped to MIDI 60-84).
    24→ *   - Lead respects harmonic context (chord tones preferred).
    25→ *   - Lead max leap: grammar.maxLeap (default 7 semitones).
    26→ *   - Lead tessitura center: grammar.tessituraCenter.
    27→ *   - Lead uses motifs from memory, transformed purposefully.
    28→ *   - Lead has call/response structure: first half = call (small
    29→ *     transpositions of the phrase motif), second half = response
    30→ *     (callResponse transform).
    31→ */

   103→export interface CompositionEngineOptions {
   104→  memory?: MotifMemory
   105→  seed: number
   106→  context: MusicalContext
   107→  /** Optional explicit grammar; defaults to context.sectionRole lookup. */
   108→  grammar?: StyleGrammar
   109→  /** Optional learned preference function (0..1). When provided, motif selection is biased. */
   110→  preferenceFor?: (motif: Motif) => number

   112→  learnedContext?: LearnedMusicalContext

   119→export class CompositionEngine {
   120→  private memory: MotifMemory
   121→  private seed: number
   122→  private context: MusicalContext
   123→  private grammar: StyleGrammar
   124→  private preferenceFor: ((motif: Motif) => number) | null
   125→  private learned: LearnedMusicalContext
   126→
   127→  constructor(opts: CompositionEngineOptions) {
   128→    this.memory = opts.memory ?? new MotifMemory()
   129→    this.seed = opts.seed
   130→    this.context = opts.context
   131→    this.grammar = opts.grammar ?? getStyleGrammar(opts.context.sectionRole || 'full-on')
   132→    this.preferenceFor = opts.preferenceFor ?? null
   133→    this.learned = opts.learnedContext ?? createEmptyLearnedContext()
   134→    this.seedMemory()
   135→  }

   137→  /** Compose a single phrase as ONE musical object. */
   138→  composePhrase(opts: {
   139→    bars: number
   140→    arrangementState: ArrangementState
   141→    groove: GroovePlan
   142→    harmonicContext: number[]
   143→    previousPhrase?: ComposedPhrase
   144→    /** If set, the phrase uses this motif (callback) instead of generating fresh. */
   145→    callbackMotif?: Motif
   146→    /** Optional adaptation intent — changes what is composed (not just filters). */
   147→    intent?: AdaptedCompositionIntent
   148→  }): ComposedPhrase {
   149→    const { bars, arrangementState, groove, harmonicContext } = opts
   150→    const intent = opts.intent
   151→    // Per-phrase rng, derived deterministically from the engine seed + state.
   152→    const rng = new Rng(
   153→      (this.seed * 7919 + bars * 31 + arrangementState.length * 17 + harmonicContext.length * 7) >>>
   154→        0
   155→    )
   156→    const roles = { ...ARRANGEMENT_ROLE_MAP[arrangementState] }
   157→
   158→    // Apply intent to roles (adaptation changes WHO plays)
   159→    if (intent) {
   160→      if (intent.bassPressure < 0.3) roles.bass = false
   161→      if (intent.leadPressure < 0.3) roles.lead = false
   162→      if (intent.groovePressure < 0.3) {
   163→        roles.kick = false
   164→        roles.hats = false
   165→      }
   166→      if (intent.texturePressure > 0.6) roles.hats = true
   167→    }
   168→
   169→    const phraseMotif = opts.callbackMotif ?? this.choosePhraseMotif(rng, opts.previousPhrase)
   170→    const callbackTo = opts.callbackMotif?.id
   171→
   172→    const half = Math.floor(bars / 2)
   173→    const composedBars: ComposedBar[] = []
   174→
   175→    for (let bar = 0; bar < bars; bar++) {
   176→      const isLast = bar === bars - 1
   177→      const isResponse = bar >= half
   178→      // Per-bar bass + lead rngs (independent streams).
   179→      const bassRng = new Rng((this.seed * 131 + bar * 53 + 11) >>> 0)
   180→      const leadRng = new Rng((this.seed * 191 + bar * 71 + 23) >>> 0)
   181→
   182→      // 1. Kick — from groove, modified by learned rhythm if available
   183→      let kickNotes: number[] = []
   184→      if (roles.kick) {
   185→        kickNotes = Array.from(new Set(groove.kickSteps)).sort((a, b) => a - b)
   186→        // Wire learned kick grammar: if confidence is high, generate original
   187→        // kick patterns from learned statistics (not copying, but same language)
   188→        const learnedKick = this.learned.rhythm.kickGrammar
   189→        const kickConf = this.learned.meta.confidence
   190→        if (kickConf > 0.3 && learnedKick.some((v: number) => v > 0.1)) {
   191→          // Generate original kick pattern from learned probabilities
   192→          const kickRng = new Rng((this.seed * 211 + bar * 43 + 7) >>> 0)
   193→          const generated: number[] = []
   194→          for (let step = 0; step < groove.stepsPerBar; step++) {
   195→            const prob = learnedKick[step % 16] ?? 0
   196→            // Blend: 60% learned probability, 40% style groove
   197→            const styleHas = groove.kickSteps.includes(step)
   198→            const blended = styleHas ? 0.8 : prob * 0.6
   199→            if (kickRng.next() < blended) {
   200→              generated.push(step)
   201→            }
   202→          }
   203→          // Always keep beat 1 (LOCKED invariant)
   204→          if (!generated.includes(0)) generated.unshift(0)
   205→          kickNotes = generated.sort((a, b) => a - b)
   206→        }
   207→      }
   208→
   209→      // 2. Hats — from groove, modified by learned hat grammar if available
   210→      let hatNotes: number[] = []
   211→      if (roles.hats) {
   212→        hatNotes = Array.from(new Set(groove.hatSteps)).sort((a, b) => a - b)
   213→        // Wire learned hat grammar
   214→        const learnedHats = this.learned.rhythm.hatGrammar
   215→        if (this.learned.meta.confidence > 0.3 && learnedHats.some((v: number) => v > 0.1)) {
   216→          const hatRng = new Rng((this.seed * 223 + bar * 47 + 13) >>> 0)
   217→          const generated: number[] = []
   218→          for (let step = 0; step < groove.stepsPerBar; step++) {
   219→            const prob = learnedHats[step % 16] ?? 0
   220→            const styleHas = groove.hatSteps.includes(step)
   221→            const blended = styleHas ? 0.7 : prob * 0.5
   222→            if (hatRng.next() < blended) {
   223→              generated.push(step)
   224→            }
   225→          }
   226→          hatNotes = generated.length > 0 ? generated.sort((a, b) => a - b) : hatNotes
   227→        }
   228→      }
   229→
   230→      // 3. Bass — only if roles.bass. Apply intent to reduce bass density.
   231→      let bassNotes = roles.bass
   232→        ? this.composeBass(bar, groove, harmonicContext, bassRng, isLast)
   233→        : []
   234→
   235→      // 4. Lead — only if roles.lead. Apply intent to reduce lead density.
   236→      let leadNotes = roles.lead
   237→        ? this.composeLead(bar, bars, phraseMotif, harmonicContext, leadRng, isResponse)
   238→        : []
   239→
   240→      // Apply adaptation intent to change DENSITY (not just on/off)
   241→      if (intent) {
   242→        // Reduce bass notes proportionally to bassPressure
   243→        if (intent.bassPressure < 0.6 && bassNotes.length > 1) {
   244→          const keepRatio = Math.max(0.3, intent.bassPressure)
   245→          const keepCount = Math.max(1, Math.floor(bassNotes.length * keepRatio))
   246→          bassNotes = bassNotes.slice(0, keepCount)
   247→        }
   248→        // Reduce lead notes proportionally to leadPressure
   249→        if (intent.leadPressure < 0.6 && leadNotes.length > 1) {
   250→          const keepRatio = Math.max(0.2, intent.leadPressure)
   251→          const keepCount = Math.max(1, Math.floor(leadNotes.length * keepRatio))
   252→          leadNotes = leadNotes.slice(0, keepCount)
   253→        }
   254→        // Register shift: shift lead notes
   255→        if (intent.registerShift !== 0 && leadNotes.length > 0) {
   256→          const shift = intent.registerShift * 12
   257→          leadNotes = leadNotes.map((n) => ({
   258→            ...n,
   259→            midi: Math.max(LEAD_MIN_MIDI, Math.min(LEAD_MAX_MIDI, n.midi + shift)),
   260→          }))
   261→        }
   262→        // Rest pressure: silence some bars entirely
   263→        if (intent.restPressure > 0.5 && bar % 4 === 2) {
   264→          leadNotes = []
   265→        }
   266→        // High texture: add hats if sparse
   267→        if (intent.texturePressure > 0.6 && hatNotes.length === 0 && roles.hats) {
   268→          hatNotes = [2, 6, 10, 14]
   269→        }
   270→      }
   271→
   272→      composedBars.push({
   273→        barIndex: bar,
   274→        arrangementState,
   275→        groove,
   276→        kickNotes,
   277→        bassNotes,
   278→        leadNotes,
   279→        hatNotes,
   280→        harmonicContext: harmonicContext.slice(),
   281→        roles: { ...roles },
   282→        timbreIntent:
   283→          this.learned.meta.confidence > 0.3
   284→            ? {
   285→                brightness: this.learned.timbre.brightness,
   286→                harmonicity: this.learned.timbre.harmonicity,
   287→                noisiness: this.learned.timbre.noisiness,
   288→                attack: this.learned.timbre.attack,
   289→                subEnergy: this.learned.timbre.subEnergy,
   290→              }
   291→            : undefined,
   292→      })
   293→    }
   294→
   295→    return {
   296→      bars: composedBars,
   297→      phraseArc: { opening: 0, peak: half, resolution: Math.max(0, bars - 1) },
   298→      motifIds: [phraseMotif.id],
   299→      callbackTo,
   300→      seed: this.seed,
   301→    }
   302→  }
   303→
   304→  /** Compose a full section (32-256 bars). */
   305→  composeSection(opts: { bars: number }): ComposedSection {
   306→    const groove = buildGroovePlan({
   307→      context: this.context,
   308→      seed: this.seed,
   309→      bars: opts.bars,
   310→      grammar: this.grammar,
   311→    })
   312→    const arrangement = planArrangement({
   313→      bars: opts.bars,
   314→      seed: this.seed,
   315→      context: this.context,
   316→    })
   317→
   318→    // Slice the arrangement into phrases at state boundaries so each phrase
   319→    // has a single ArrangementState. This keeps composePhrase's
   320→    // role-activation logic correct (an INTRO phrase has no kick; a GROOVE
   321→    // phrase has kick) while still respecting the arrangement's per-bar
   322→    // state changes.
   323→    const groups: { state: ArrangementState; start: number; length: number }[] = []
   324→    let curState: ArrangementState | null = null
   325→    let curStart = 0
   326→    for (let bar = 0; bar < opts.bars; bar++) {
   327→      const slot = arrangement.slots[bar]
   328→      const state: ArrangementState = slot?.state ?? 'GROOVE'
   329→      if (curState === null) {
   330→        curState = state
   331→        curStart = bar
   332→      } else if (state !== curState) {
   333→        groups.push({ state: curState, start: curStart, length: bar - curStart })
   334→        curState = state
   335→        curStart = bar
   336→      }
   337→    }
   338→    if (curState !== null) {
   339→      groups.push({
   340→        state: curState,
   341→        start: curStart,
   342→        length: opts.bars - curStart,
   343→      })
   344→    }
   345→    const totalPhrases = groups.length
   346→
   347→    const phrases: ComposedPhrase[] = []
   348→    const bars: ComposedBar[] = []
   349→    let prev: ComposedPhrase | undefined
   350→    let firstPhraseMotif: Motif | undefined
   351→
   352→    for (let phraseIdx = 0; phraseIdx < groups.length; phraseIdx++) {
   353→      const group = groups[phraseIdx] as { state: ArrangementState; start: number; length: number }
   354→      const isLastPhrase = phraseIdx === totalPhrases - 1
   355→      const harmonicContext = this.chooseHarmonicForPhrase(phraseIdx)
   356→      const callbackMotif = isLastPhrase && firstPhraseMotif ? firstPhraseMotif : undefined
   357→
   358→      const phrase = this.composePhrase({
   359→        bars: group.length,
   360→        arrangementState: group.state,
   361→        groove,
   362→        harmonicContext,
   363→        previousPhrase: prev,
   364→        callbackMotif,
   365→      })
   366→
   367→      // Re-index bars to absolute bar indices and sync per-bar roles from
   368→      // the arrangement (in case the arrangement's slot has slightly
   369→      // different roles than the state's default — e.g., due to jitter).
   370→      for (const b of phrase.bars) {
   371→        const absoluteBar = b.barIndex + group.start
   372→        const absSlot = arrangement.slots.find((s) => s.barIndex === absoluteBar)
   373→        const out: ComposedBar = { ...b, barIndex: absoluteBar }
   374→        if (absSlot) {
   375→          out.roles = { ...absSlot.roles }
   376→        }
   377→        bars.push(out)
   378→      }
   379→      phrases.push(phrase)
   380→
   381→      // Capture the first phrase's motif for the final callback.
   382→      if (phraseIdx === 0) {
   383→        const id = phrase.motifIds[0]
   384→        if (id) {
   385→          const entry = this.memory.retrieve(id)
   386→          if (entry) firstPhraseMotif = entry.motif
   387→        }
   388→      }
   389→      prev = phrase
   390→    }
   391→
   392→    return { bars, phrases, arrangement, groove, seed: this.seed }
   393→  }
   394→
   395→  /** Get all notes from a composed section as flat arrays per part. */
   396→  renderNotes(section: ComposedSection): {
   397→    kick: { step: number; bar: number }[]
   398→    bass: { midi: number; step: number; bar: number; function: string }[]
   399→    lead: { midi: number; step: number; bar: number; velocity: number }[]
   400→    hats: { step: number; bar: number }[]
   401→  } {
   402→    const kick: { step: number; bar: number }[] = []
   403→    const bass: { midi: number; step: number; bar: number; function: string }[] = []
   404→    const lead: { midi: number; step: number; bar: number; velocity: number }[] = []
   405→    const hats: { step: number; bar: number }[] = []
   406→    for (const bar of section.bars) {
   407→      for (const step of bar.kickNotes) kick.push({ step, bar: bar.barIndex })
   408→      for (const n of bar.bassNotes) {
   409→        bass.push({ midi: n.midi, step: n.step, bar: bar.barIndex, function: n.function })
   410→      }
   411→      for (const n of bar.leadNotes) {
   412→        lead.push({ midi: n.midi, step: n.step, bar: bar.barIndex, velocity: n.velocity })
   413→      }
   414→      for (const step of bar.hatNotes) hats.push({ step, bar: bar.barIndex })
   415→    }
   416→    return { kick, bass, lead, hats }
   417→  }

   421→  /** Seed memory with 2-3 motifs so the first phrase has material to draw on. */
   422→  private seedMemory(): void {
   423→    if (this.memory.size > 0) return
   424→    const baseMotif = generateMotifV2(
   425→      { ...this.context, octave: 4 },
   426→      (this.seed * 7 + 1) >>> 0,
   427→      'lead'
   428→    )
   429→    this.memory.ingest(baseMotif, 0, { salience: 0.7, role: 'lead' })
   430→    const altMotif = generateMotifV2(
   431→      { ...this.context, octave: 4 },
   432→      (this.seed * 13 + 31) >>> 0,
   433→      'lead'
   434→    )
   435→    this.memory.ingest(altMotif, 0, { salience: 0.6, role: 'lead' })
   436→  }
   437→
   438→  /**
   439→   * Choose the phrase motif. 30% of the time (deterministic via rng), pull
   440→   * a motif from memory referenced by the previous phrase — a cross-phrase
   441→   * callback. Otherwise generate a fresh motif. When a learned preference
   442→   * function is provided, candidates are scored and the highest-scoring
   443→   * motif is selected with 70% probability (30% exploration).
   444→   */
   445→  private choosePhraseMotif(rng: Rng, previousPhrase?: ComposedPhrase): Motif {
   446→    // 30% chance: callback to previous phrase motif
   447→    if (previousPhrase && rng.next() < 0.3) {
   448→      const prevId = previousPhrase.motifIds[0]
   449→      if (prevId) {
   450→        const entry = this.memory.retrieve(prevId)
   451→        if (entry) return entry.motif
   452→      }
   453→    }
   454→
   455→    // Generate 3 candidate motifs
   456→    const candidates: Motif[] = []
   457→    for (let i = 0; i < 3; i++) {
   458→      const freshSeed = (this.seed * 101 + rng.int(1, 1_000_000) + i * 7919) >>> 0
   459→      candidates.push(generateMotifV2({ ...this.context, octave: 4 }, freshSeed, 'lead'))
   460→    }
   461→
   462→    // If learning is wired, score candidates and prefer the best (70% exploit, 30% explore)
   463→    if (this.preferenceFor) {
   464→      const prefFn = this.preferenceFor
   465→      const scored = candidates.map((m) => ({ motif: m, score: prefFn ? prefFn(m) : 0.5 }))
   466→      scored.sort((a, b) => b.score - a.score)
   467→      if (rng.next() < 0.7) {
   468→        // Exploit: pick highest-scoring
   469→        const best = scored[0] as { motif: Motif; score: number }
   470→        this.memory.ingest(best.motif, 0, { salience: 0.6, role: 'lead' })
   471→        return best.motif
   472→      }
   473→      // Explore: pick a random candidate
   474→      const pick = candidates[rng.int(0, candidates.length - 1)] as Motif
   475→      this.memory.ingest(pick, 0, { salience: 0.6, role: 'lead' })
   476→      return pick
   477→    }
   478→
   479→    // No learning: pick first candidate (deterministic)
   480→    const motif = candidates[0] as Motif
   481→    this.memory.ingest(motif, 0, { salience: 0.6, role: 'lead' })
   482→    return motif
   483→  }

   485→  /**
   486→   * Choose chord pitch classes for a phrase. Rotates through tonic /
   487→   * subdominant / tonic / dominant across phrases so harmonic rhythm is
   488→   * non-zero across a section.
   489→   */
   490→  private chooseHarmonicForPhrase(phraseIndex: number): number[] {
   491→    const scale = getScale(this.context.scaleName)
   492→    if (!scale) return [this.context.tonic]
   493→    const pcs = scalePcs(this.context.tonic, scale)
   494→    if (pcs.length < 5) return [pcs[0] ?? this.context.tonic]
   495→
   496→    // Wire learned harmony: if pitch-class profile has data, use it to
   497→    // influence which chord tones are selected (not copying, but preferring
   498→    // pitch classes that were prominent in the source)
   499→    const pcProfile = this.learned.harmony.pitchClassProfile
   500→    const hasLearnedHarmony =
   501→      this.learned.meta.confidence > 0.3 && pcProfile.some((v: number) => v > 0.05)
   502→
   503→    const idx = ((phraseIndex % 4) + 4) % 4
   504→
   505→    // Base progression: tonic / subdominant / tonic / dominant
   506→    let baseChord: number[]
   507→    if (idx === 0) {
   508→      baseChord = [pcs[0] ?? 0, pcs[2] ?? 0, pcs[4] ?? 0]
   509→    } else if (idx === 1) {
   510→      baseChord = [pcs[3] ?? 0, pcs[5] ?? 0, pcs[7 % pcs.length] ?? 0]
   511→    } else if (idx === 2) {
   512→      baseChord = [pcs[0] ?? 0, pcs[2] ?? 0, pcs[4] ?? 0]
   513→    } else {
   514→      baseChord = [pcs[4] ?? 0, pcs[6 % pcs.length] ?? 0, pcs[8 % pcs.length] ?? 0]
   515→    }
   516→
   517→    if (hasLearnedHarmony) {
   518→      // Blend: keep the progression structure but weight chord tones by
   519→      // learned pitch-class profile. Replace one chord tone with a
   520→      // high-weight pc from the learned profile if it's in the scale.
   521→      const sortedPcs = pcProfile
   522→        .map((weight: number, pc: number) => ({ pc, weight }))
   523→        .filter(
   524→          (x: { pc: number; weight: number }) => pcs.includes(x.pc) && !baseChord.includes(x.pc)
   525→        )
   526→        .sort((a: { weight: number }, b: { weight: number }) => b.weight - a.weight)
   527→
   528→      if (sortedPcs.length > 0 && sortedPcs[0]) {
   529→        const top = sortedPcs[0] as { pc: number; weight: number }
   530→        // Replace the third chord tone with the learned-preferred pc
   531→        baseChord = [baseChord[0] ?? 0, baseChord[1] ?? 0, top.pc]
   532→      }
   533→    }
   534→
   535→    return baseChord
   536→  }
   537→
   538→  /**
   539→   * Compose the bass for one bar against the groove.
   540→   *
   541→   * LOCKED mode: bass hits every kick step with ROOT (so bass+kick are
   542→   * rhythmically locked).
   543→   * COMPLEMENTARY mode: bass hits the gaps between kicks with FIFTH/OCTAVE
   544→   * (so bass+kick interlock without colliding).
   545→   *
   546→   * Either way, beat 1 (step 0) ALWAYS gets a ROOT — the invariant that
   547→   * keeps bass-kick alignment high.
   548→   */
   549→  private composeBass(
   550→    bar: number,
   551→    groove: GroovePlan,
   552→    _harmonicContext: number[],
   553→    rng: Rng,
   554→    isLast: boolean
   555→  ): { midi: number; step: number; durationSteps: number; function: string }[] {
   556→    const scale = getScale(this.context.scaleName)
   557→    if (!scale) return []
   558→
   559→    const rootMidi = degreeToMidi(this.context.tonic, scale, 0, BASS_OCTAVE)
   560→    const fifthMidi = degreeToMidi(this.context.tonic, scale, 4, BASS_OCTAVE)
   561→    const octaveMidi = degreeToMidi(this.context.tonic, scale, 0, BASS_OCTAVE + 1)
   562→    const thirdMidi = degreeToMidi(this.context.tonic, scale, 2, BASS_OCTAVE)
   563→    const seventhMidi = degreeToMidi(this.context.tonic, scale, 6, BASS_OCTAVE)
   564→    const beat = Math.max(1, Math.round(groove.stepsPerBar / 4))
   565→    const tension = this.grammar.tensionPreference
   566→
   567→    // Learned bass grammar: degree preferences from radio observation
   568→    const bassDegPrefs = this.learned.bass.degreePreferences
   569→    const hasLearnedBass =
   570→      Object.keys(bassDegPrefs).length > 0 && this.learned.meta.confidence > 0.3
   571→
   572→    /** Choose a bass pitch using learned preferences or fallback to style */
   573→    const chooseBassPitch = (rng: Rng): { midi: number; fn: string } => {
   574→      if (hasLearnedBass) {
   575→        // Weighted selection from learned degree preferences
   576→        const degrees = Object.entries(bassDegPrefs)
   577→        const totalWeight = degrees.reduce((s, [, w]) => s + w, 0)
   578→        let r = rng.next() * totalWeight
   579→        for (const [degStr, weight] of degrees) {
   580→          r -= weight
   581→          if (r <= 0) {
   582→            const deg = Number.parseInt(degStr, 10)
   583→            const midi = degreeToMidi(this.context.tonic, scale, deg, BASS_OCTAVE)
   584→            const fns = ['ROOT', 'THIRD', 'FIFTH', 'SEVENTH', 'OCTAVE']
   585→            return { midi, fn: fns[deg] ?? 'PASSING' }
   586→          }
   587→        }
   588→      }
   589→      // Fallback: style-based selection
   590→      const useFifth = rng.next() < 0.5
   591→      return useFifth ? { midi: fifthMidi, fn: 'FIFTH' } : { midi: octaveMidi, fn: 'OCTAVE' }
   592→    }
   593→
   594→    const notes: { midi: number; step: number; durationSteps: number; function: string }[] = []
   595→
   596→    // Beat 1: ALWAYS ROOT (LOCKED invariant — bass-kick alignment).
   597→    notes.push({ midi: rootMidi, step: 0, durationSteps: 2, function: 'ROOT' })
   598→
   599→    if (groove.bassKickAlignment === 'LOCKED') {
   600→      for (const step of groove.kickSteps) {
   601→        if (step === 0) continue
   602→        if (rng.next() < 0.3 + tension * 0.2) {
   603→          const pitch = chooseBassPitch(rng)
   604→          const dur = rng.next() < 0.5 ? 1 : 2
   605→          notes.push({ midi: pitch.midi, step, durationSteps: dur, function: pitch.fn })
   606→        } else {
   607→          notes.push({ midi: rootMidi, step, durationSteps: 2, function: 'ROOT' })
   608→        }
   609→      }
   610→    } else {
   611→      const half = Math.max(1, Math.round(groove.stepsPerBar / 8))
   612→      const complementary = [half, beat + half, beat * 2 + half, beat * 3 + half]
   613→      for (const step of complementary) {
   614→        if (groove.kickSteps.includes(step) || step === 0) continue
   615→        const pitch = chooseBassPitch(rng)
   616→        notes.push({ midi: pitch.midi, step, durationSteps: 1, function: pitch.fn })
   617→      }
   618→    }
   619→
   620→    // Offbeat passing/approach tones — vary per bar
   621→    const offbeatVariation = (bar * 7 + this.seed * 13) % 4
   622→    const offbeatTargets =
   623→      offbeatVariation === 0
   624→        ? [beat * 2, beat * 3]
   625→        : offbeatVariation === 1
   626→          ? [beat + 1, beat * 3 + 1]
   627→          : offbeatVariation === 2
   628→            ? [beat * 2 + 1]
   629→            : [beat * 3, beat * 3 + 2]
   630→
   631→    for (const step of offbeatTargets) {
   632→      if (notes.some((n) => n.step === step)) continue
   633→      if (step >= groove.stepsPerBar) continue
   634→      if (rng.next() < 0.2 + groove.syncopationBudget * 0.3 + tension * 0.2) {
   635→        const useApproach = rng.next() < 0.3
   636→        if (useApproach) {
   637→          const approachMidi = rootMidi + (rng.next() < 0.5 ? -1 : 1)
   638→          notes.push({ midi: approachMidi, step, durationSteps: 1, function: 'APPROACH' })
   639→        } else {
   640→          const useThird = rng.next() < 0.5
   641→          notes.push({
   642→            midi: useThird ? thirdMidi : seventhMidi,
   643→            step,
   644→            durationSteps: 1,
   645→            function: useThird ? 'THIRD' : 'PASSING',
   646→          })
   647→        }
   648→      }
   649→    }
   650→
   651→    // Octave jump for energy on some bars
   652→    if (rng.next() < 0.15 + tension * 0.15) {
   653→      const jumpStep = beat * 2
   654→      if (!notes.some((n) => n.step === jumpStep)) {
   655→        notes.push({ midi: octaveMidi, step: jumpStep, durationSteps: 1, function: 'OCTAVE' })
   656→      }
   657→    }
   658→
   659→    // Cadence on the last bar
   660→    if (isLast) {
   661→      const lastStep = groove.stepsPerBar - 2
   662→      const filtered = notes.filter((n) => n.step < lastStep)
   663→      notes.length = 0
   664→      notes.push(...filtered)
   665→      notes.push({ midi: fifthMidi, step: lastStep, durationSteps: 1, function: 'CADENCE' })
   666→      notes.push({ midi: rootMidi, step: lastStep + 1, durationSteps: 2, function: 'CADENCE' })
   667→    }
   668→
   669→    // Clamp to octave 2 (MIDI 36-59)
   670→    for (const n of notes) {
   671→      while (n.midi < 36) n.midi += 12
   672→      while (n.midi > 59) n.midi -= 12
   673→    }
   674→
   675→    notes.sort((a, b) => a.step - b.step)
   676→    const deduped: typeof notes = []
   677→    for (const n of notes) {
   678→      if (!deduped.some((d) => d.step === n.step)) deduped.push(n)
   679→    }
   680→    return deduped
   681→  }
   682→
   683→  /**
   684→   * Compose the lead for one bar against the bass+harmony.
   685→   *
   686→   * Call/response structure:
   687→   *   - Bars in the first half (call): the phrase motif, with small
   688→   *     transpositions for variety.
   689→   *   - Bars in the second half (response): the callResponse transform of
   690→   *     the phrase motif (shifts to the fifth, resolves the last note to
   691→   *     the root).
   692→   *
   693→   * The rendered notes are clamped to tessitura (MIDI 60-84) and the max
   694→   * leap is enforced by clamping consecutive intervals to ±maxLeap.
   695→   */
   696→  private composeLead(
   697→    bar: number,
   698→    _bars: number,
   699→    phraseMotif: Motif,
   700→    harmonicContext: number[],
   701→    rng: Rng,
   702→    isResponse: boolean
   703→  ): { midi: number; step: number; durationSteps: number; velocity: number }[] {
   704→    const scale = getScale(this.context.scaleName)
   705→    if (!scale || phraseMotif.notes.length === 0) return []
   706→
   707→    let motif = phraseMotif
   708→    if (isResponse) {
   709→      motif = callResponse(phraseMotif, this.context.tonic, scale, (this.seed + bar * 17) >>> 0)
   710→    } else if (bar > 0) {
   711→      // More aggressive per-bar variation (prevents 90% repetition)
   712→      const variationChoice = (bar * 3 + this.seed) % 5
   713→      if (variationChoice === 0) {
   714→        // Transpose ±2 or ±3
   715→        const t = rng.pick([-3, -2, 2, 3])
   716→        motif = transposeMotif(phraseMotif, t, this.context.tonic, scale)
   717→      } else if (variationChoice === 1) {
   718→        // Invert (with 30% probability)
   719→        if (rng.next() < 0.3) {
   720→          motif = invertMotif(phraseMotif, this.context.tonic, scale)
   721→        } else {
   722→          const t = rng.pick([-2, 0, 2])
   723→          if (t !== 0) motif = transposeMotif(phraseMotif, t, this.context.tonic, scale)
   724→        }
   725→      } else if (variationChoice === 2) {
   726→        // Retrograde (with 20% probability)
   727→        if (rng.next() < 0.2) {
   728→          motif = retrogradeMotif(phraseMotif)
   729→        }
   730→      } else if (variationChoice === 3) {
   731→        // Fragment (first half only, with 25% probability)
   732→        if (rng.next() < 0.25) {
   733→          const halfNotes = phraseMotif.notes.slice(0, Math.ceil(phraseMotif.notes.length / 2))
   734→          if (halfNotes.length > 0) {
   735→            motif = { ...phraseMotif, notes: halfNotes }
   736→          }
   737→        }
   738→      }
   739→      // variationChoice === 4: no change (original motif)
   740→    }
   741→
   742→    // Render motif notes, snapping to tessitura (MIDI 60-84).
   743→    const notes = motif.notes.map((n) => {
   744→      let midi = n.midi
   745→      while (midi < LEAD_MIN_MIDI) midi += 12
   746→      while (midi > LEAD_MAX_MIDI) midi -= 12
   747→      return {
   748→        midi,
   749→        step: n.step,
   750→        durationSteps: n.durationSteps,
   751→        velocity: n.velocity,
   752→      }
   753→    })
   754→
   755→    // Enforce max leap: clamp consecutive intervals to ±maxLeap.
   756→    const maxLeap = this.grammar.maxLeap
   757→    for (let i = 1; i < notes.length; i++) {
   758→      const prev = notes[i - 1]
   759→      const cur = notes[i]
   760→      if (!prev || !cur) continue
   761→      const interval = cur.midi - prev.midi
   762→      if (Math.abs(interval) > maxLeap) {
   763→        const sign = Math.sign(interval)
   764→        cur.midi = prev.midi + sign * maxLeap
   765→      }
   766→    }
   767→
   768→    // Snap to chord tones with some probability — keeps HARMONY_IGNORED away
   769→    // while preserving the motif's overall shape.
   770→    const chordPcs = new Set(harmonicContext)
   771→    if (chordPcs.size > 0) {
   772→      for (const n of notes) {
   773→        const pc = ((n.midi % 12) + 12) % 12
   774→        if (!chordPcs.has(pc) && rng.next() < 0.5) {
   775→          for (let off = 1; off <= 2; off++) {
   776→            const upPc = (((n.midi + off) % 12) + 12) % 12
   777→            if (chordPcs.has(upPc)) {
   778→              n.midi += off
   779→              break
   780→            }
   781→            const downPc = (((n.midi - off) % 12) + 12) % 12
   782→            if (chordPcs.has(downPc)) {
   783→              n.midi -= off
   784→              break
   785→            }
   786→          }
   787→        }
   788→      }
   789→    }
   790→
   791→    return notes
   792→  }
   793→}

   797→/**
   798→ * Pure pitch inversion (no scale snapping). Mirrors each note around the
   799→ * first note's MIDI value. This is a TRUE involution: applying it twice
   800→ * returns the original motif exactly. Useful for property-based tests.
   801→ */
   802→export function invertPitchPure(motif: Motif): Motif {
   803→  if (motif.notes.length === 0) {
   804→    return createMotif([], {
   805→      id: `${motif.id}:pureInvert`,
   806→      rootPc: motif.rootPc,
   807→      scaleName: motif.scaleName,
   808→      steps: motif.steps,
   809→      role: motif.role,
   810→      sourceMotifId: motif.id,
   811→      transformHistory: [...motif.transformHistory, 'invertPitchPure'],
   812→    })
   813→  }
   814→  const first = motif.notes[0] as MotifNote
   815→  const notes = motif.notes.map((n) => {
   816→    if (n === first) return { ...n }
   817→    const offset = n.midi - first.midi
   818→    return { ...n, midi: first.midi - offset }
   819→  })
   820→  return createMotif(notes, {
   821→    id: `${motif.id}:pureInvert`,
   822→    rootPc: motif.rootPc,
   823→    scaleName: motif.scaleName,
   824→    steps: motif.steps,
   825→    role: motif.role,
   826→    sourceMotifId: motif.id,
   827→    transformHistory: [...motif.transformHistory, 'invertPitchPure'],
   828→  })
   829→}

   831→/**
   832→ * Retrograde that is a TRUE involution: reversing note content while
   833→ * preserving step positions. Wraps the existing transformation.retrograde
   834→ * so callers can verify the involution property via the composition module.
   835→ */
   836→export function retrogradePure(motif: Motif): Motif {
   837→  return retrogradeMotif(motif)
   838→}

   840→/** Convenience: how tightly the bass locks with the kick on beat 1. */
   841→export function measureBassKickAlignment(
   842→  bassNotes: { step: number; bar: number }[],
   843→  kickNotes: { step: number; bar: number }[],
   844→  bars: number
   845→): number {
   846→  if (bars === 0) return 0
   847→  let aligned = 0
   848→  for (let bar = 0; bar < bars; bar++) {
   849→    const bassOnBeat1 = bassNotes.some((n) => n.bar === bar && n.step === 0)
   850→    const kickOnBeat1 = kickNotes.some((n) => n.bar === bar && n.step === 0)
   851→    if (bassOnBeat1 && kickOnBeat1) aligned++
   852→  }
   853→  return aligned / bars
   854→}

   856→/** Convenience: clamp a MIDI note into a register range. */
   857→export function clampToRegister(midi: number, minMidi: number, maxMidi: number): number {
   858→  let m = midi
   859→  while (m < minMidi) m += 12
   860→  while (m > maxMidi) m -= 12
   861→  return m
   862→}
