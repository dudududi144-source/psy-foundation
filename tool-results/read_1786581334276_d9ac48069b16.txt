     1→     1→/**
     2→     2→ * CompositionEngine: the single authoritative composer.
     3→     3→ *
     4→     4→ * The engine composes whole phrases, not isolated bars. It follows a strict
     5→     5→ * hierarchy:
     6→     6→ *   1. Build GroovePlan (kick skeleton, subdivision, accents)
     7→     7→ *   2. Determine harmonic plan (chord tones, tension notes)
     8→     8→ *   3. Compose BASS against groove (bass knows where kick is)
     9→     9→ *   4. Compose LEAD against bass+harmony (lead leaves space for bass,
    10→    10→ *      stays in tessitura)
    11→    11→ *   5. Arrange parts (some roles OFF in BREAK / INTRO / OUTRO sections)
    12→    12→ *
    13→    13→ * Bass composition rules:
    14→    14→ *   - Bass ALWAYS places a ROOT on step 0 (beat 1) — the LOCKED invariant
    15→    15→ *     that keeps bass-kick alignment high.
    16→    16→ *   - In LOCKED mode: bass hits every kick step with ROOT.
    17→    17→ *   - In COMPLEMENTARY mode: bass hits the gaps between kicks with
    18→    18→ *     FIFTH / OCTAVE.
    19→    19→ *   - Bass register: octave 2 (MIDI 36-59).
    20→    20→ *   - Last bar of a phrase: CADENCE walk (fifth → root).
    21→    21→ *
    22→    22→ * Lead composition rules:
    23→    23→ *   - Lead MUST NOT overlap bass register (lead clamped to MIDI 60-84).
    24→    24→ *   - Lead respects harmonic context (chord tones preferred).
    25→    25→ *   - Lead max leap: grammar.maxLeap (default 7 semitones).
    26→    26→ *   - Lead tessitura center: grammar.tessituraCenter.
    27→    27→ *   - Lead uses motifs from memory, transformed purposefully.
    28→    28→ *   - Lead has call/response structure: first half = call (small
    29→    29→ *     transpositions of the phrase motif), second half = response
    30→    30→ *     (callResponse transform).
    31→    31→ */

   205→   205→export class CompositionEngine {
   206→   206→  private memory: MotifMemory
   207→   207→  private seed: number
   208→   208→  private context: MusicalContext
   209→   209→  private grammar: StyleGrammar
   210→   210→  private preferenceFor: ((motif: Motif) => number) | null
   211→   211→  private learned: LearnedMusicalContext
   212→   212→  private interactionGrammar: InteractionGrammar
   213→   213→  private previousPhraseMaterial: PhraseMaterial | null
   214→   214→  private developmentOperator: DevelopmentOperator | null
   215→   215→  private relationalOff: boolean
   216→   216→  private identity: LearnedIdentity | null
   217→   217→
   218→   218→  constructor(opts: CompositionEngineOptions) {
   219→   219→    this.memory = opts.memory ?? new MotifMemory()
   220→   220→    this.seed = opts.seed
   221→   221→    this.context = opts.context
   222→   222→    this.grammar = opts.grammar ?? getStyleGrammar(opts.context.sectionRole || 'full-on')
   223→   223→    this.preferenceFor = opts.preferenceFor ?? null
   224→   224→    this.learned = opts.learnedContext ?? opts.identity?.learned ?? createEmptyLearnedContext()
   225→   225→    this.interactionGrammar =
   226→   226→      opts.interactionGrammar ?? opts.identity?.grammar ?? createEmptyInteractionGrammar()
   227→   227→    this.previousPhraseMaterial = opts.previousPhraseMaterial ?? null
   228→   228→    this.developmentOperator = opts.developmentOperator ?? null
   229→   229→    this.relationalOff = opts.relationalGenerationOff ?? false
   230→   230→    this.identity = opts.identity ?? null
   231→   231→    // If an identity is provided, bake its energy/tension/syncopation into the context.
   232→   232→    if (this.identity) {
   233→   233→      this.context = {
   234→   234→        ...this.context,
   235→   235→        energy: this.identity.energy,
   236→   236→        tension: this.identity.tension,
   237→   237→        density: 0.3 + this.identity.energy * 0.4,
   238→   238→      }
   239→   239→    }
   240→   240→    this.seedMemory()
   241→   241→  }
   242→   242→
   243→   243→  /** Compose a single phrase as ONE musical object. */
   244→   244→  composePhrase(opts: {
   245→   245→    bars: number
   246→   246→    arrangementState: ArrangementState
   247→   247→    groove: GroovePlan
   248→   248→    harmonicContext: number[]
   249→   249→    previousPhrase?: ComposedPhrase
   250→   250→    /** If set, the phrase uses this motif (callback) instead of generating fresh. */
   251→   251→    callbackMotif?: Motif
   252→   252→    /** Optional adaptation intent — changes what is composed (not just filters). */
   253→   253→    intent?: AdaptedCompositionIntent
   254→   254→    /** F20: absolute bar index where this phrase starts (for HarmonicPlan). */
   255→   255→    startBar?: number
   256→   256→    /** F20: phrase index within the section (for harmonic function + cadence). */
   257→   257→    phraseIndex?: number
   258→   258→    /** F20: is this the last phrase in the section? (drives cadence target). */
   259→   259→    isLastPhrase?: boolean
   260→   260→    /** F20: phrase role (STATEMENT/RESPONSE/RESOLUTION/...) — drives cadence target. */
   261→   261→    phraseRole?: string
   262→   262→  }): ComposedPhrase {
   263→   263→    const { bars, arrangementState, groove, harmonicContext } = opts
   264→   264→    const intent = opts.intent
   265→   265→    // Per-phrase rng, derived deterministically from the engine seed + state.
   266→   266→    const rng = new Rng(
   267→   267→      (this.seed * 7919 + bars * 31 + arrangementState.length * 17 + harmonicContext.length * 7) >>>
   268→   268→        0
   269→   269→    )
   270→   270→    const roles = { ...ARRANGEMENT_ROLE_MAP[arrangementState] }
   271→   271→
   272→   272→    // Apply intent to roles (adaptation changes WHO plays)
   273→   273→    if (intent) {
   274→   274→      if (intent.bassPressure < 0.3) roles.bass = false
   275→   275→      if (intent.leadPressure < 0.3) roles.lead = false
   276→   276→      if (intent.groovePressure < 0.3) {
   277→   277→        roles.kick = false
   278→   278→        roles.hats = false
   279→   279→      }
   280→   280→      if (intent.texturePressure > 0.6) roles.hats = true
   281→   281→    }
   282→   282→
   283→   283→    const phraseMotif = opts.callbackMotif ?? this.choosePhraseMotif(rng, opts.previousPhrase)
   284→   284→    const callbackTo = opts.callbackMotif?.id
   285→   285→
   286→   286→    // ── F20 STEP 1: HarmonicPlan ──
   287→   287→    // Build a real harmonic plan for the phrase. The lead will consult this
   288→   288→    // for current chord, next chord, cadence target, and harmonic function.
   289→   289→    const harmonicPlan = buildHarmonicPlan({
   290→   290→      bars,
   291→   291→      startBar: opts.startBar ?? 0,
   292→   292→      tonic: this.context.tonic,
   293→   293→      scaleName: this.context.scaleName,
   294→   294→      phraseIndex: opts.phraseIndex ?? 0,
   295→   295→      isLastPhrase: opts.isLastPhrase ?? false,
   296→   296→      phraseRole: opts.phraseRole,
   297→   297→      chordChangeRate: this.grammar.chordChangeRate,
   298→   298→      tensionPreference: this.grammar.tensionPreference,
   299→   299→      learnedPcProfile: this.learned.harmony.pitchClassProfile,
   300→   300→      learnedConfidence: this.learned.meta.confidence,
   301→   301→    })
   302→   302→
   303→   303→    // ── F20 STEP 2: PhraseMaterial ──
   304→   304→    // Derive the phrase's musical material from the previous phrase's material
   305→   305→    // via the development operator. When no previous material exists (first
   306→   306→    // phrase), extract material from the chosen phrase motif.
   307→   307→    let phraseMaterial: PhraseMaterial
   308→   308→    if (this.previousPhraseMaterial && this.developmentOperator) {
   309→   309→      const scale = getScale(this.context.scaleName)
   310→   310→      const fallbackScale = getScale('phrygian-dominant')
   311→   311→      const rootMidi = degreeToMidi(
   312→   312→        this.context.tonic,
   313→   313→        scale ?? fallbackScale ?? (getScale('minor') as NonNullable<typeof scale>),
   314→   314→        0,
   315→   315→        4
   316→   316→      )
   317→   317→      phraseMaterial = applyOperatorToMaterial(
   318→   318→        this.developmentOperator,
   319→   319→        this.previousPhraseMaterial,
   320→   320→        {
   321→   321→          tonic: this.context.tonic,
   322→   322→          scaleName: this.context.scaleName,
   323→   323→          scale: scale ?? undefined,
   324→   324→          rootMidi,
   325→   325→          rng: rng as unknown as {
   326→   326→            next: () => number
   327→   327→            pick: <T>(a: T[]) => T
   328→   328→            int: (a: number, b: number) => number
   329→   329→          },
   330→   330→          variationAmount: 0.4,
   331→   331→          cadenceTargetPc: harmonicPlan.cadenceTarget.pc,
   332→   332→        }
   333→   333→      )
   334→   334→    } else {
   335→   335→      phraseMaterial = motifToPhraseMaterial(phraseMotif, groove.stepsPerBar)
   336→   336→    }
   337→   337→
   338→   338→    const half = Math.floor(bars / 2)
   339→   339→    const composedBars: ComposedBar[] = []
   340→   340→    const grammarConfidence = this.interactionGrammar.confidence
   341→   341→    // F21: derive tension dimensions for this phrase (used by SoundDNA + lead).
   342→   342→    const tensionDims = deriveTensionDimensions(this.context.tension)
   343→   343→
   344→   344→    for (let bar = 0; bar < bars; bar++) {
   345→   345→      const isLast = bar === bars - 1
   346→   346→      const isResponse = bar >= half
   347→   347→      // Per-bar bass + lead rngs (independent streams).
   348→   348→      const bassRng = new Rng((this.seed * 131 + bar * 53 + 11) >>> 0)
   349→   349→      const leadRng = new Rng((this.seed * 191 + bar * 71 + 23) >>> 0)
   350→   350→
   351→   351→      // Active chord at this bar (absolute bar index within the phrase).
   352→   352→      const absoluteBar = (opts.startBar ?? 0) + bar
   353→   353→      const activeChord = chordAtBar(harmonicPlan, absoluteBar)
   354→   354→      const barChordTones = activeChord?.chordTones ?? harmonicContext
   355→   355→      const nextChord = nextChordAfterBar(harmonicPlan, absoluteBar)
   356→   356→      const isAnticipation = isAnticipationBar(harmonicPlan, absoluteBar)
   357→   357→
   358→   358→      // ── F20 STEP 3: KickPlan ──
   359→   359→      // Kick is generated from the groove skeleton + learned kick grammar.
   360→   360→      // The KickPlan object is emitted so the bass and lead can consume the
   361→   361→      // ACTUAL kick onsets (not the style skeleton).
   362→   362→      const kickPlan = this.composeKickPlan(bar, groove, rng)
   363→   363→      const kickNotes = kickPlan.onsets
   364→   364→
   365→   365→      // ── F20 STEP 4: Hats ── (unchanged logic, kept for backward compat)
   366→   366→      let hatNotes: number[] = []
   367→   367→      if (roles.hats) {
   368→   368→        hatNotes = Array.from(new Set(groove.hatSteps)).sort((a, b) => a - b)
   369→   369→        const learnedHats = this.learned.rhythm.hatGrammar
   370→   370→        if (this.learned.meta.confidence > 0.3 && learnedHats.some((v: number) => v > 0.1)) {
   371→   371→          const hatRng = new Rng((this.seed * 223 + bar * 47 + 13) >>> 0)
   372→   372→          const generated: number[] = []
   373→   373→          for (let step = 0; step < groove.stepsPerBar; step++) {
   374→   374→            const prob = learnedHats[step % 16] ?? 0
   375→   375→            const styleHas = groove.hatSteps.includes(step)
   376→   376→            const blended = styleHas ? 0.7 : prob * 0.5
   377→   377→            if (hatRng.next() < blended) {
   378→   378→              generated.push(step)
   379→   379→            }
   380→   380→          }
   381→   381→          hatNotes = generated.length > 0 ? generated.sort((a, b) => a - b) : hatNotes
   382→   382→        }
   383→   383→      }
   384→   384→
   385→   385→      // ── F20/F21 STEP 5: BassPlan ──
   386→   386→      // Bass is composed AGAINST the actual KickPlan + HarmonicPlan.
   387→   387→      // F21: when a learned identity is present, dispatch to the vocabulary-
   388→   388→      // based generator (ROLLING/SYNCOPATED/MELODIC/ACID/SPARSE/TENSION) —
   389→   389→      // each mode alters the GENERATION PROCEDURE, not a step array.
   390→   390→      // F20: otherwise use the grammar-driven composeBassPlan.
   391→   391→      let bassPlan: BassPlan = emptyBassPlan()
   392→   392→      if (roles.bass) {
   393→   393→        if (this.relationalOff) {
   394→   394→          // A/B test fallback: old deaf bass (groove skeleton only, no grammar).
   395→   395→          bassPlan = this.composeBassLegacy(bar, groove, bassRng, isLast)
   396→   396→        } else if (this.identity) {
   397→   397→          // F21: vocabulary-based bass generation.
   398→   398→          const bassScale = getScale(this.context.scaleName)
   399→   399→          if (bassScale) {
   400→   400→            const vocabNotes = generateBassByVocabulary(this.identity.bassVocabulary, {
   401→   401→              bar,
   402→   402→              groove,
   403→   403→              kickPlan,
   404→   404→              chordTones: barChordTones,
   405→   405→              tonic: this.context.tonic,
   406→   406→              scaleName: this.context.scaleName,
   407→   407→              scale: bassScale,
   408→   408→              isLast,
   409→   409→              isAnticipationBar: isAnticipation,
   410→   410→              rng: bassRng,
   411→   411→              intervalWidth: this.identity.bassIntervalWidth,
   412→   412→              syncopation: this.identity.syncopation,
   413→   413→              bassOctave: BASS_OCTAVE,
   414→   414→              tension: this.context.tension,
   415→   415→            })
   416→   416→            bassPlan = { notes: vocabNotes, onsets: vocabNotes.map((n) => n.step) }
   417→   417→          }
   418→   418→        } else {
   419→   419→          bassPlan = this.composeBassPlan({
   420→   420→            bar,
   421→   421→            groove,
   422→   422→            kickPlan,
   423→   423→            chordTones: barChordTones,
   424→   424→            harmonicFunction: activeChord?.function ?? 'TONIC',
   425→   425→            isLast,
   426→   426→            rng: bassRng,
   427→   427→            grammarConfidence,
   428→   428→            isAnticipationBar: isAnticipation,
   429→   429→          })
   430→   430→        }
   431→   431→      }
   432→   432→      const bassNotes: { midi: number; step: number; durationSteps: number; function: string }[] =
   433→   433→        bassPlan.notes.map((n) => ({
   434→   434→          midi: n.midi,
   435→   435→          step: n.step,
   436→   436→          durationSteps: n.durationSteps,
   437→   437→          function: n.function,
   438→   438→        }))
   439→   439→
   440→   440→      // ── F20 STEP 6: RhythmicSpaceMap ──
   441→   441→      // Derived from the actual kick + bass plans. The lead consults this to
   442→   442→      // decide where to play, where to rest, where to answer bass.
   443→   443→      const spaceMap = buildRhythmicSpaceMap({
   444→   444→        stepsPerBar: groove.stepsPerBar,
   445→   445→        kickOnsets: kickPlan.onsets,
   446→   446→        bassOnsets: bassPlan.onsets,
   447→   447→        accentSteps: groove.accentSteps,
   448→   448→        hatOnsets: hatNotes,
   449→   449→        harmonicChangeSteps: isAnticipation ? [groove.stepsPerBar - 1] : [],
   450→   450→        bassAnticipationSteps: bassPlan.notes.filter((n) => n.isAnticipation).map((n) => n.step),
   451→   451→      })
   452→   452→
   453→   453→      // ── F20 STEP 7: LeadPlan ──
   454→   454→      // Lead is composed AGAINST everything: harmonic plan, groove, kick plan,
   455→   455→      // bass plan, space map, phrase material, phrase role, tension state,
   456→   456→      // and the learned interaction grammar. The lead explicitly decides per
   457→   457→      // step where to play / rest / answer / anticipate / cadence.
   458→   458→      let leadPlan: LeadPlan = emptyLeadPlan()
   459→   459→      if (roles.lead) {
   460→   460→        if (this.relationalOff) {
   461→   461→          // A/B test fallback: old deaf lead (motif + chord snapping only).
   462→   462→          const legacyNotes = this.composeLeadLegacy(
   463→   463→            bar,
   464→   464→            bars,
   465→   465→            phraseMotif,
   466→   466→            barChordTones,
   467→   467→            leadRng,
   468→   468→            isResponse
   469→   469→          )
   470→   470→          leadPlan = { notes: legacyNotes.map((n) => ({ ...n, role: 'CONTINUATION' as const })) }
   471→   471→        } else {
   472→   472→          leadPlan = this.composeLeadPlan({
   473→   473→            bar,
   474→   474→            bars,
   475→   475→            groove,
   476→   476→            kickPlan,
   477→   477→            bassPlan,
   478→   478→            spaceMap,
   479→   479→            harmonicPlan,
   480→   480→            activeChord: activeChord ?? null,
   481→   481→            nextChord: nextChord,
   482→   482→            isAnticipation,
   483→   483→            isLast,
   484→   484→            isResponse,
   485→   485→            phraseMaterial,
   486→   486→            phraseRole: opts.phraseRole ?? 'STATEMENT',
   487→   487→            rng: leadRng,
   488→   488→            grammarConfidence,
   489→   489→          })
   490→   490→        }
   491→   491→      }
   492→   492→      const leadNotes: { midi: number; step: number; durationSteps: number; velocity: number }[] =
   493→   493→        leadPlan.notes.map((n) => ({
   494→   494→          midi: n.midi,
   495→   495→          step: n.step,
   496→   496→          durationSteps: n.durationSteps,
   497→   497→          velocity: n.velocity,
   498→   498→        }))
   499→   499→
   500→   500→      // Apply adaptation intent to change DENSITY (not just on/off)
   501→   501→      let adjustedBass = bassNotes
   502→   502→      let adjustedLead = leadNotes
   503→   503→      if (intent) {
   504→   504→        // Reduce bass notes proportionally to bassPressure
   505→   505→        if (intent.bassPressure < 0.6 && adjustedBass.length > 1) {
   506→   506→          const keepRatio = Math.max(0.3, intent.bassPressure)
   507→   507→          const keepCount = Math.max(1, Math.floor(adjustedBass.length * keepRatio))
   508→   508→          adjustedBass = adjustedBass.slice(0, keepCount)
   509→   509→        }
   510→   510→        // Reduce lead notes proportionally to leadPressure
   511→   511→        if (intent.leadPressure < 0.6 && adjustedLead.length > 1) {
   512→   512→          const keepRatio = Math.max(0.2, intent.leadPressure)
   513→   513→          const keepCount = Math.max(1, Math.floor(adjustedLead.length * keepRatio))
   514→   514→          adjustedLead = adjustedLead.slice(0, keepCount)
   515→   515→        }
   516→   516→        // Register shift: shift lead notes
   517→   517→        if (intent.registerShift !== 0 && adjustedLead.length > 0) {
   518→   518→          const shift = intent.registerShift * 12
   519→   519→          adjustedLead = adjustedLead.map((n) => ({
   520→   520→            ...n,
   521→   521→            midi: Math.max(LEAD_MIN_MIDI, Math.min(LEAD_MAX_MIDI, n.midi + shift)),
   522→   522→          }))
   523→   523→        }
   524→   524→        // Rest pressure: silence some bars entirely
   525→   525→        if (intent.restPressure > 0.5 && bar % 4 === 2) {
   526→   526→          adjustedLead = []
   527→   527→        }
   528→   528→        // High texture: add hats if sparse
   529→   529→        if (intent.texturePressure > 0.6 && hatNotes.length === 0 && roles.hats) {
   530→   530→          hatNotes.push(2, 6, 10, 14)
   531→   531→        }
   532→   532→      }
   533→   533→
   534→   534→      // F21: render per-role SynthRecipes from SoundDNA. The SoundDNA is
   535→   535→      // derived from the learned TimbreProfile and tension dimensions. The
   536→   536→      // recipes specify oscillator type, layer count, filter topology,
   537→   537→      // envelope, saturation, modulation, and stereo — they reach the real
   538→   538→      // synthesis graph (consumed by PSY4's DSP layer).
   539→   539→      const soundDNA = timbreToSoundDNA(this.learned.timbre)
   540→   540→      // Apply spectral tension: high tension → brighter.
   541→   541→      soundDNA.brightness = Math.max(
   542→   542→        0,
   543→   543→        Math.min(1, soundDNA.brightness * 0.5 + tensionDims.spectral * 0.5)
   544→   544→      )
   545→   545→      const synthRecipes = {
   546→   546→        kick: renderSynthRecipe(soundDNA, 'kick'),
   547→   547→        bass: renderSynthRecipe(soundDNA, 'bass'),
   548→   548→        lead: renderSynthRecipe(soundDNA, 'lead'),
   549→   549→        hats: renderSynthRecipe(soundDNA, 'hats'),
   550→   550→      }
   551→   551→
   552→   552→      composedBars.push({
   553→   553→        barIndex: bar,
   554→   554→        arrangementState,
   555→   555→        groove,
   556→   556→        kickNotes,
   557→   557→        bassNotes: adjustedBass,
   558→   558→        leadNotes: adjustedLead,
   559→   559→        hatNotes,
   560→   560→        harmonicContext: barChordTones.slice(),
   561→   561→        roles: { ...roles },
   562→   562→        timbreIntent:
   563→   563→          this.learned.meta.confidence > 0.3
   564→   564→            ? {
   565→   565→                brightness: this.learned.timbre.brightness,
   566→   566→                harmonicity: this.learned.timbre.harmonicity,
   567→   567→                noisiness: this.learned.timbre.noisiness,
   568→   568→                attack: this.learned.timbre.attack,
   569→   569→                subEnergy: this.learned.timbre.subEnergy,
   570→   570→              }
   571→   571→            : undefined,
   572→   572→        // F20 plan inspection fields.
   573→   573→        harmonicPlan,
   574→   574→        activeChord: activeChord ?? undefined,
   575→   575→        spaceMap,
   576→   576→        kickPlan,
   577→   577→        bassPlan,
   578→   578→        leadPlan,
   579→   579→        // F21 SoundDNA fields.
   580→   580→        synthRecipes,
   581→   581→        soundDNA,
   582→   582→      })
   583→   583→    }
   584→   584→
   585→   585→    return {
   586→   586→      bars: composedBars,
   587→   587→      phraseArc: { opening: 0, peak: half, resolution: Math.max(0, bars - 1) },
   588→   588→      motifIds: [phraseMotif.id],
   589→   589→      callbackTo,
   590→   590→      seed: this.seed,
   591→   591→      phraseMaterial,
   592→   592→      developmentOperator: this.developmentOperator ?? undefined,
   593→   593→      harmonicPlan,
   594→   594→    }
   595→   595→  }

   597→   597→  /** Compose a full section (32-256 bars). */
   598→   598→  composeSection(opts: { bars: number }): ComposedSection {
   599→   599→    const groove = buildGroovePlan({
   600→   600→      context: this.context,
   601→   601→      seed: this.seed,
   602→   602→      bars: opts.bars,
   603→   603→      grammar: this.grammar,
   604→   604→    })
   605→   605→    const arrangement = planArrangement({
   606→   606→      bars: opts.bars,
   607→   607→      seed: this.seed,
   608→   608→      context: this.context,
   609→   609→    })
   610→   610→
   611→   611→    // Slice the arrangement into phrases at state boundaries so each phrase
   612→   612→    // has a single ArrangementState. This keeps composePhrase's
   613→   613→    // role-activation logic correct (an INTRO phrase has no kick; a GROOVE
   614→   614→    // phrase has kick) while still respecting the arrangement's per-bar
   615→   615→    // state changes.
   616→   616→    const groups: { state: ArrangementState; start: number; length: number }[] = []
   617→   617→    let curState: ArrangementState | null = null
   618→   618→    let curStart = 0
   619→   619→    for (let bar = 0; bar < opts.bars; bar++) {
   620→   620→      const slot = arrangement.slots[bar]
   621→   621→      const state: ArrangementState = slot?.state ?? 'GROOVE'
   622→   622→      if (curState === null) {
   623→   623→        curState = state
   624→   624→        curStart = bar
   625→   625→      } else if (state !== curState) {
   626→   626→        groups.push({ state: curState, start: curStart, length: bar - curStart })
   627→   627→        curState = state
   628→   628→        curStart = bar
   629→   629→      }
   630→   630→    }
   631→   631→    if (curState !== null) {
   632→   632→      groups.push({
   633→   633→        state: curState,
   634→   634→        start: curStart,
   635→   635→        length: opts.bars - curStart,
   636→   636→      })
   637→   637→    }
   638→   638→    const totalPhrases = groups.length
   639→   639→
   640→   640→    const phrases: ComposedPhrase[] = []
   641→   641→    const bars: ComposedBar[] = []
   642→   642→    let prev: ComposedPhrase | undefined
   643→   643→    let firstPhraseMotif: Motif | undefined
   644→   644→
   645→   645→    for (let phraseIdx = 0; phraseIdx < groups.length; phraseIdx++) {
   646→   646→      const group = groups[phraseIdx] as { state: ArrangementState; start: number; length: number }
   647→   647→      const isLastPhrase = phraseIdx === totalPhrases - 1
   648→   648→      const harmonicContext = this.chooseHarmonicForPhrase(phraseIdx)
   649→   649→      const callbackMotif = isLastPhrase && firstPhraseMotif ? firstPhraseMotif : undefined
   650→   650→
   651→   651→      // F20: derive a phrase role from the arrangement state for cadence targeting.
   652→   652→      const phraseRole = this.derivePhraseRole(group.state, phraseIdx, isLastPhrase)
   653→   653→
   654→   654→      const phrase = this.composePhrase({
   655→   655→        bars: group.length,
   656→   656→        arrangementState: group.state,
   657→   657→        groove,
   658→   658→        harmonicContext,
   659→   659→        previousPhrase: prev,
   660→   660→        callbackMotif,
   661→   661→        startBar: group.start,
   662→   662→        phraseIndex: phraseIdx,
   663→   663→        isLastPhrase,
   664→   664→        phraseRole,
   665→   665→      })
   666→   666→
   667→   667→      // Re-index bars to absolute bar indices and sync per-bar roles from
   668→   668→      // the arrangement (in case the arrangement's slot has slightly
   669→   669→      // different roles than the state's default — e.g., due to jitter).
   670→   670→      for (const b of phrase.bars) {
   671→   671→        const absoluteBar = b.barIndex + group.start
   672→   672→        const absSlot = arrangement.slots.find((s) => s.barIndex === absoluteBar)
   673→   673→        const out: ComposedBar = { ...b, barIndex: absoluteBar }
   674→   674→        if (absSlot) {
   675→   675→          out.roles = { ...absSlot.roles }
   676→   676→        }
   677→   677→        bars.push(out)
   678→   678→      }
   679→   679→      phrases.push(phrase)
   680→   680→
   681→   681→      // Capture the first phrase's motif for the final callback.
   682→   682→      if (phraseIdx === 0) {
   683→   683→        const id = phrase.motifIds[0]
   684→   684→        if (id) {
   685→   685→          const entry = this.memory.retrieve(id)
   686→   686→          if (entry) firstPhraseMotif = entry.motif
   687→   687→        }
   688→   688→      }
   689→   689→      prev = phrase
   690→   690→    }
   691→   691→
   692→   692→    return { bars, phrases, arrangement, groove, seed: this.seed }
   693→   693→  }

   695→   695→  /** Get all notes from a composed section as flat arrays per part. */
   696→   696→  renderNotes(section: ComposedSection): {
   697→   697→    kick: { step: number; bar: number }[]
   698→   698→    bass: { midi: number; step: number; bar: number; function: string }[]
   699→   699→    lead: { midi: number; step: number; bar: number; velocity: number }[]
   700→   700→    hats: { step: number; bar: number }[]
   701→   701→  } {
   702→   702→    const kick: { step: number; bar: number }[] = []
   703→   703→    const bass: { midi: number; step: number; bar: number; function: string }[] = []
   704→   704→    const lead: { midi: number; step: number; bar: number; velocity: number }[] = []
   705→   705→    const hats: { step: number; bar: number }[] = []
   706→   706→    for (const bar of section.bars) {
   707→   707→      for (const step of bar.kickNotes) kick.push({ step, bar: bar.barIndex })
   708→   708→      for (const n of bar.bassNotes) {
   709→   709→        bass.push({ midi: n.midi, step: n.step, bar: bar.barIndex, function: n.function })
   710→   710→      }
   711→   711→      for (const n of bar.leadNotes) {
   712→   712→        lead.push({ midi: n.midi, step: n.step, bar: bar.barIndex, velocity: n.velocity })
   713→   713→      }
   714→   714→      for (const step of bar.hatNotes) hats.push({ step, bar: bar.barIndex })
   715→   715→    }
   716→   716→    return { kick, bass, lead, hats }
   717→   717→  }

   721→   721→  /** Seed memory with 2-3 motifs so the first phrase has material to draw on. */
   722→   722→  private seedMemory(): void {
   723→   723→    if (this.memory.size > 0) return
   724→   724→    const baseMotif = generateMotifV2(
   725→   725→      { ...this.context, octave: 4 },
   726→   726→      (this.seed * 7 + 1) >>> 0,
   727→   727→      'lead'
   728→   728→    )
   729→   729→    this.memory.ingest(baseMotif, 0, { salience: 0.7, role: 'lead' })
   730→   730→    const altMotif = generateMotifV2(
   731→   731→      { ...this.context, octave: 4 },
   732→   732→      (this.seed * 13 + 31) >>> 0,
   733→   733→      'lead'
   734→   734→    )
   735→   735→    this.memory.ingest(altMotif, 0, { salience: 0.6, role: 'lead' })
   736→   736→  }
   737→   737→
   738→   738→  /**
   739→   739→   * Choose the phrase motif. 30% of the time (deterministic via rng), pull
   740→   740→   * a motif from memory referenced by the previous phrase — a cross-phrase
   741→   741→   * callback. Otherwise generate a fresh motif. When a learned preference
   742→   742→   * function is provided, candidates are scored and the highest-scoring
   743→   743→   * motif is selected with 70% probability (30% exploration).
   744→   744→   */
   745→   745→  private choosePhraseMotif(rng: Rng, previousPhrase?: ComposedPhrase): Motif {
   746→   746→    // 30% chance: callback to previous phrase motif
   747→   747→    if (previousPhrase && rng.next() < 0.3) {
   748→   748→      const prevId = previousPhrase.motifIds[0]
   749→   749→      if (prevId) {
   750→   750→        const entry = this.memory.retrieve(prevId)
   751→   751→        if (entry) return entry.motif
   752→   752→      }
   753→   753→    }
   754→   754→
   755→   755→    // Generate 3 candidate motifs
   756→   756→    const candidates: Motif[] = []
   757→   757→    for (let i = 0; i < 3; i++) {
   758→   758→      const freshSeed = (this.seed * 101 + rng.int(1, 1_000_000) + i * 7919) >>> 0
   759→   759→      candidates.push(generateMotifV2({ ...this.context, octave: 4 }, freshSeed, 'lead'))
   760→   760→    }
   761→   761→
   762→   762→    // If learning is wired, score candidates and prefer the best (70% exploit, 30% explore)
   763→   763→    if (this.preferenceFor) {
   764→   764→      const prefFn = this.preferenceFor
   765→   765→      const scored = candidates.map((m) => ({ motif: m, score: prefFn ? prefFn(m) : 0.5 }))
   766→   766→      scored.sort((a, b) => b.score - a.score)
   767→   767→      if (rng.next() < 0.7) {
   768→   768→        // Exploit: pick highest-scoring
   769→   769→        const best = scored[0] as { motif: Motif; score: number }
   770→   770→        this.memory.ingest(best.motif, 0, { salience: 0.6, role: 'lead' })
   771→   771→        return best.motif
   772→   772→      }
   773→   773→      // Explore: pick a random candidate
   774→   774→      const pick = candidates[rng.int(0, candidates.length - 1)] as Motif
   775→   775→      this.memory.ingest(pick, 0, { salience: 0.6, role: 'lead' })
   776→   776→      return pick
   777→   777→    }
   778→   778→
   779→   779→    // No learning: pick first candidate (deterministic)
   780→   780→    const motif = candidates[0] as Motif
   781→   781→    this.memory.ingest(motif, 0, { salience: 0.6, role: 'lead' })
   782→   782→    return motif
   783→   783→  }
   784→   784→
   785→   785→  /**
   786→   786→   * Choose chord pitch classes for a phrase. Rotates through tonic /
   787→   787→   * subdominant / tonic / dominant across phrases so harmonic rhythm is
   788→   788→   * non-zero across a section.
   789→   789→   */
   790→   790→  private chooseHarmonicForPhrase(phraseIndex: number): number[] {
   791→   791→    const scale = getScale(this.context.scaleName)
   792→   792→    if (!scale) return [this.context.tonic]
   793→   793→    const pcs = scalePcs(this.context.tonic, scale)
   794→   794→    if (pcs.length < 5) return [pcs[0] ?? this.context.tonic]
   795→   795→
   796→   796→    // Wire learned harmony: if pitch-class profile has data, use it to
   797→   797→    // influence which chord tones are selected (not copying, but preferring
   798→   798→    // pitch classes that were prominent in the source)
   799→   799→    const pcProfile = this.learned.harmony.pitchClassProfile
   800→   800→    const hasLearnedHarmony =
   801→   801→      this.learned.meta.confidence > 0.3 && pcProfile.some((v: number) => v > 0.05)
   802→   802→
   803→   803→    const idx = ((phraseIndex % 4) + 4) % 4
   804→   804→
   805→   805→    // Base progression: tonic / subdominant / tonic / dominant
   806→   806→    let baseChord: number[]
   807→   807→    if (idx === 0) {
   808→   808→      baseChord = [pcs[0] ?? 0, pcs[2] ?? 0, pcs[4] ?? 0]
   809→   809→    } else if (idx === 1) {
   810→   810→      baseChord = [pcs[3] ?? 0, pcs[5] ?? 0, pcs[7 % pcs.length] ?? 0]
   811→   811→    } else if (idx === 2) {
   812→   812→      baseChord = [pcs[0] ?? 0, pcs[2] ?? 0, pcs[4] ?? 0]
   813→   813→    } else {
   814→   814→      baseChord = [pcs[4] ?? 0, pcs[6 % pcs.length] ?? 0, pcs[8 % pcs.length] ?? 0]
   815→   815→    }
   816→   816→
   817→   817→    if (hasLearnedHarmony) {
   818→   818→      // Blend: keep the progression structure but weight chord tones by
   819→   819→      // learned pitch-class profile. Replace one chord tone with a
   820→   820→      // high-weight pc from the learned profile if it's in the scale.
   821→   821→      const sortedPcs = pcProfile
   822→   822→        .map((weight: number, pc: number) => ({ pc, weight }))
   823→   823→        .filter(
   824→   824→          (x: { pc: number; weight: number }) => pcs.includes(x.pc) && !baseChord.includes(x.pc)
   825→   825→        )
   826→   826→        .sort((a: { weight: number }, b: { weight: number }) => b.weight - a.weight)
   827→   827→
   828→   828→      if (sortedPcs.length > 0 && sortedPcs[0]) {
   829→   829→        const top = sortedPcs[0] as { pc: number; weight: number }
   830→   830→        // Replace the third chord tone with the learned-preferred pc
   831→   831→        baseChord = [baseChord[0] ?? 0, baseChord[1] ?? 0, top.pc]
   832→   832→      }
   833→   833→    }
   834→   834→
   835→   835→    return baseChord
   836→   836→  }
   837→   837→
   838→   838→  /**
   839→   839→   * F20: derive a phrase role from the arrangement state + position. Drives
   840→   840→   * the cadence target (root / third / fifth) in the HarmonicPlan.
   841→   841→   */
   842→   842→  private derivePhraseRole(
   843→   843→    state: ArrangementState,
   844→   844→    phraseIdx: number,
   845→   845→    isLastPhrase: boolean
   846→   846→  ): string {
   847→   847→    if (isLastPhrase) return 'RESOLUTION'
   848→   848→    if (state === 'INTRO' || state === 'BUILD') return 'INTRO'
   849→   849→    if (state === 'BREAK' || state === 'DROP') return 'BUILD'
   850→   850→    if (state === 'OUTRO') return 'RESOLUTION'
   851→   851→    if (phraseIdx % 2 === 1) return 'RESPONSE'
   852→   852→    return 'STATEMENT'
   853→   853→  }
   854→   854→
   855→   855→  /**
   856→   856→   * F20: compose a KickPlan for one bar. Emits a first-class KickPlan object
   857→   857→   * (onsets + velocities) so the bass and lead generators can consume the
   858→   858→   * ACTUAL kick onsets, not the style skeleton.
   859→   859→   *
   860→   860→   * When learned kick grammar is available (confidence > 0.3), the kick
   861→   861→   * pattern is regenerated from learned probabilities blended with the style
   862→   862→   * groove. Beat 1 (step 0) is always kept — the LOCKED invariant.
   863→   863→   */
   864→   864→  private composeKickPlan(bar: number, groove: GroovePlan, _phraseRng: Rng): KickPlan {
   865→   865→    let onsets: number[] = Array.from(new Set(groove.kickSteps)).sort((a, b) => a - b)
   866→   866→    const learnedKick = this.learned.rhythm.kickGrammar
   867→   867→    const kickConf = this.learned.meta.confidence
   868→   868→    if (kickConf > 0.3 && learnedKick.some((v: number) => v > 0.1)) {
   869→   869→      const kickRng = new Rng((this.seed * 211 + bar * 43 + 7) >>> 0)
   870→   870→      const generated: number[] = []
   871→   871→      for (let step = 0; step < groove.stepsPerBar; step++) {
   872→   872→        const prob = learnedKick[step % 16] ?? 0
   873→   873→        const styleHas = groove.kickSteps.includes(step)
   874→   874→        const blended = styleHas ? 0.8 : prob * 0.6
   875→   875→        if (kickRng.next() < blended) {
   876→   876→          generated.push(step)
   877→   877→        }
   878→   878→      }
   879→   879→      if (!generated.includes(0)) generated.unshift(0)
   880→   880→      onsets = generated.sort((a, b) => a - b)
   881→   881→    }
   882→   882→    const velocities = onsets.map((s) => (s === 0 ? 1.0 : 0.7 + groove.accent[s] * 0.2))
   883→   883→    return { onsets, velocities }
   884→   884→  }
   885→   885→
   886→   886→  /**
   887→   887→   * F20: compose a BassPlan against the ACTUAL KickPlan + HarmonicPlan, and
   888→   888→   * consume the learned KICK→BASS grammar (bassOnsetProbability) and bass
   889→   889→   * transition grammar (pickNextBassDegree).
   890→   890→   *
   891→   891→   * Causal paths:
   892→   892→   *   KICK → BASS: bassOnsetProbability(step, kickHas) decides each onset
   893→   893→   *   BASS TRANSITIONS: pickNextBassDegree picks the next degree from the
   894→   894→   *   learned Markov chain
   895→   895→   *
   896→   896→   * The bass reads `kickPlan.onsets` (the generated kick), NOT
   897→   897→   * `groove.kickSteps` (the style skeleton) — so bass-kick alignment is real,
   898→   898→   * not vestigial.
   899→   899→   */
   900→   900→  private composeBassPlan(opts: {
   901→   901→    bar: number
   902→   902→    groove: GroovePlan
   903→   903→    kickPlan: KickPlan
   904→   904→    chordTones: number[]
   905→   905→    harmonicFunction: HarmonicFunction
   906→   906→    isLast: boolean
   907→   907→    rng: Rng
   908→   908→    grammarConfidence: number
   909→   909→    isAnticipationBar: boolean
   910→   910→  }): BassPlan {
   911→   911→    const scale = getScale(this.context.scaleName)
   912→   912→    if (!scale) return emptyBassPlan()
   913→   913→    const { groove, kickPlan, rng, isLast, grammarConfidence } = opts
   914→   914→
   915→   915→    const rootMidi = degreeToMidi(this.context.tonic, scale, 0, BASS_OCTAVE)
   916→   916→    const fifthMidi = degreeToMidi(this.context.tonic, scale, 4, BASS_OCTAVE)
   917→   917→    const octaveMidi = degreeToMidi(this.context.tonic, scale, 0, BASS_OCTAVE + 1)
   918→   918→    const thirdMidi = degreeToMidi(this.context.tonic, scale, 2, BASS_OCTAVE)
   919→   919→    const seventhMidi = degreeToMidi(this.context.tonic, scale, 6, BASS_OCTAVE)
   920→   920→    const tension = this.grammar.tensionPreference
   921→   921→
   922→   922→    const bassDegPrefs = this.learned.bass.degreePreferences
   923→   923→    const hasLearnedBass =
   924→   924→      Object.keys(bassDegPrefs).length > 0 && this.learned.meta.confidence > 0.3
   925→   925→
   926→   926→    const notes: BassPlanNote[] = []
   927→   927→    let prevDegree = 0
   928→   928→
   929→   929→    /** Weighted selection from learned degree preferences (F15 compat). */
   930→   930→    const pickDegreeByPreferences = (): number => {
   931→   931→      if (hasLearnedBass) {
   932→   932→        const degrees = Object.entries(bassDegPrefs)
   933→   933→        const totalWeight = degrees.reduce((s, [, w]) => s + (w as number), 0)
   934→   934→        let r = rng.next() * totalWeight
   935→   935→        for (const [degStr, weight] of degrees) {
   936→   936→          r -= weight as number
   937→   937→          if (r <= 0) return Number.parseInt(degStr, 10)
   938→   938→        }
   939→   939→      }
   940→   940→      return [0, 4, 2, 6][Math.floor(rng.next() * 4)] ?? 0
   941→   941→    }
   942→   942→
   943→   943→    /** Choose the next bass degree using the learned transition Markov chain,
   944→   944→     * falling back to degreePreferences when the grammar is untrained. */
   945→   945→    const chooseNextDegree = (candidates: number[]): number => {
   946→   946→      const row = this.interactionGrammar.bassTransitions.transitions[prevDegree]
   947→   947→      if (hasLearnedBass && grammarConfidence > 0.1 && row && Object.keys(row).length > 0) {
   948→   948→        return pickNextBassDegree({
   949→   949→          fromDegree: prevDegree,
   950→   950→          candidates,
   951→   951→          grammar: this.interactionGrammar,
   952→   952→          confidence: grammarConfidence,
   953→   953→          rng,
   954→   954→        })
   955→   955→      }
   956→   956→      // F15 compat: weighted selection from learned degree preferences.
   957→   957→      if (hasLearnedBass) {
   958→   958→        return pickDegreeByPreferences()
   959→   959→      }
   960→   960→      return candidates[Math.floor(rng.next() * candidates.length)] ?? 0
   961→   961→    }
   962→   962→
   963→   963→    /** Map a degree to a MIDI + function label. */
   964→   964→    const degreeToNote = (deg: number): { midi: number; fn: BassPlanNote['function'] } => {
   965→   965→      const midi = degreeToMidi(this.context.tonic, scale, deg, BASS_OCTAVE)
   966→   966→      const fns: BassPlanNote['function'][] = ['ROOT', 'THIRD', 'FIFTH', 'SEVENTH', 'OCTAVE']
   967→   967→      return { midi, fn: fns[deg] ?? 'PASSING' }
   968→   968→    }
   969→   969→
   970→   970→    // Beat 1: ALWAYS ROOT (LOCKED invariant — bass-kick alignment).
   971→   971→    notes.push({
   972→   972→      midi: rootMidi,
   973→   973→      step: 0,
   974→   974→      durationSteps: 2,
   975→   975→      function: 'ROOT',
   976→   976→      isAnticipation: false,
   977→   977→    })
   978→   978→    prevDegree = 0
   979→   979→
   980→   980→    // LOCKED mode: bass hits EVERY kick step (the invariant that keeps
   981→   981→    // bass-kick alignment high). The grammar's bassOnsetProbability is
   982→   982→    // consumed for NON-kick steps (the off-kick syncopation decisions).
   983→   983→    const kickSet = new Set(kickPlan.onsets)
   984→   984→    if (groove.bassKickAlignment === 'LOCKED') {
   985→   985→      for (const step of kickPlan.onsets) {
   986→   986→        if (step === 0) continue
   987→   987→        // LOCKED: bass always hits the kick step. Vary the pitch via the
   988→   988→        // learned transition chain / degree preferences (causal).
   989→   989→        if (rng.next() < 0.3 + tension * 0.2) {
   990→   990→          const deg = chooseNextDegree([0, 4, 2, 6])
   991→   991→          const { midi, fn } = degreeToNote(deg)
   992→   992→          const dur = rng.next() < 0.5 ? 1 : 2
   993→   993→          notes.push({ midi, step, durationSteps: dur, function: fn, isAnticipation: false })
   994→   994→          prevDegree = deg
   995→   995→        } else {
   996→   996→          notes.push({
   997→   997→            midi: rootMidi,
   998→   998→            step,
   999→   999→            durationSteps: 2,
  1000→  1000→            function: 'ROOT',
  1001→  1001→            isAnticipation: false,
  1002→  1002→          })
  1003→  1003→        }
  1004→  1004→      }
  1005→  1005→    }
  1006→  1006→
  1007→  1007→    // Walk the NON-kick steps. At each, decide whether to place a bass note
  1008→  1008→    // using the learned KICK→BASS probability (bassOnsetProbability). This is
  1009→  1009→    // the causal path: grammar.bassOffKickProb[step] → probability → note.
  1010→  1010→    for (let step = 1; step < groove.stepsPerBar; step++) {
  1011→  1011→      if (kickSet.has(step)) continue // already handled in LOCKED pass
  1012→  1012→      const kickHas = false // off-kick step
  1013→  1013→      const prob = bassOnsetProbability({
  1014→  1014→        step,
  1015→  1015→        kickHas,
  1016→  1016→        grammar: this.interactionGrammar,
  1017→  1017→        confidence: grammarConfidence,
  1018→  1018→        defaultOnKick: 0.3 + tension * 0.2,
  1019→  1019→        defaultOffKick: 0.1 + groove.syncopationBudget * 0.2,
  1020→  1020→      })
  1021→  1021→      if (rng.next() < prob) {
  1022→  1022→        const deg = chooseNextDegree([0, 4, 2, 6])
  1023→  1023→        const { midi, fn } = degreeToNote(deg)
  1024→  1024→        const isAnticipation = opts.isAnticipationBar && step >= groove.stepsPerBar - 2
  1025→  1025→        const dur = rng.next() < 0.5 ? 1 : 2
  1026→  1026→        notes.push({
  1027→  1027→          midi,
  1028→  1028→          step,
  1029→  1029→          durationSteps: dur,
  1030→  1030→          function: isAnticipation ? 'ANTICIPATION' : fn,
  1031→  1031→          isAnticipation,
  1032→  1032→        })
  1033→  1033→        prevDegree = deg
  1034→  1034→      }
  1035→  1035→    }
  1036→  1036→
  1037→  1037→    // Offbeat passing/approach tones — vary per bar (kept from legacy for
  1038→  1038→    // musical richness; these do not depend on grammar).
  1039→  1039→    const beat = Math.max(1, Math.round(groove.stepsPerBar / 4))
  1040→  1040→    const offbeatVariation = (opts.bar * 7 + this.seed * 13) % 4
  1041→  1041→    const offbeatTargets =
  1042→  1042→      offbeatVariation === 0
  1043→  1043→        ? [beat * 2, beat * 3]
  1044→  1044→        : offbeatVariation === 1
  1045→  1045→          ? [beat + 1, beat * 3 + 1]
  1046→  1046→          : offbeatVariation === 2
  1047→  1047→            ? [beat * 2 + 1]
  1048→  1048→            : [beat * 3, beat * 3 + 2]
  1049→  1049→    for (const step of offbeatTargets) {
  1050→  1050→      if (notes.some((n) => n.step === step)) continue
  1051→  1051→      if (step >= groove.stepsPerBar) continue
  1052→  1052→      if (rng.next() < 0.2 + groove.syncopationBudget * 0.3 + tension * 0.2) {
  1053→  1053→        const useApproach = rng.next() < 0.3
  1054→  1054→        if (useApproach) {
  1055→  1055→          const approachMidi = rootMidi + (rng.next() < 0.5 ? -1 : 1)
  1056→  1056→          notes.push({
  1057→  1057→            midi: approachMidi,
  1058→  1058→            step,
  1059→  1059→            durationSteps: 1,
  1060→  1060→            function: 'APPROACH',
  1061→  1061→            isAnticipation: false,
  1062→  1062→          })
  1063→  1063→        } else {
  1064→  1064→          const useThird = rng.next() < 0.5
  1065→  1065→          notes.push({
  1066→  1066→            midi: useThird ? thirdMidi : seventhMidi,
  1067→  1067→            step,
  1068→  1068→            durationSteps: 1,
  1069→  1069→            function: useThird ? 'THIRD' : 'PASSING',
  1070→  1070→            isAnticipation: false,
  1071→  1071→          })
  1072→  1072→        }
  1073→  1073→      }
  1074→  1074→    }
  1075→  1075→
  1076→  1076→    // Octave jump for energy on some bars.
  1077→  1077→    if (rng.next() < 0.15 + tension * 0.15) {
  1078→  1078→      const jumpStep = beat * 2
  1079→  1079→      if (!notes.some((n) => n.step === jumpStep)) {
  1080→  1080→        notes.push({
  1081→  1081→          midi: octaveMidi,
  1082→  1082→          step: jumpStep,
  1083→  1083→          durationSteps: 1,
  1084→  1084→          function: 'OCTAVE',
  1085→  1085→          isAnticipation: false,
  1086→  1086→        })
  1087→  1087→      }
  1088→  1088→    }
  1089→  1089→
  1090→  1090→    // Cadence on the last bar — fifth → root walk.
  1091→  1091→    if (isLast) {
  1092→  1092→      const lastStep = groove.stepsPerBar - 2
  1093→  1093→      const filtered = notes.filter((n) => n.step < lastStep)
  1094→  1094→      notes.length = 0
  1095→  1095→      notes.push(...filtered)
  1096→  1096→      notes.push({
  1097→  1097→        midi: fifthMidi,
  1098→  1098→        step: lastStep,
  1099→  1099→        durationSteps: 1,
  1100→  1100→        function: 'CADENCE',
  1101→  1101→        isAnticipation: false,
  1102→  1102→      })
  1103→  1103→      notes.push({
  1104→  1104→        midi: rootMidi,
  1105→  1105→        step: lastStep + 1,
  1106→  1106→        durationSteps: 2,
  1107→  1107→        function: 'CADENCE',
  1108→  1108→        isAnticipation: false,
  1109→  1109→      })
  1110→  1110→    }
  1111→  1111→
  1112→  1112→    // Clamp to octave 2 (MIDI 36-59).
  1113→  1113→    for (const n of notes) {
  1114→  1114→      while (n.midi < 36) n.midi += 12
  1115→  1115→      while (n.midi > 59) n.midi -= 12
  1116→  1116→    }
  1117→  1117→
  1118→  1118→    notes.sort((a, b) => a.step - b.step)
  1119→  1119→    const deduped: BassPlanNote[] = []
  1120→  1120→    for (const n of notes) {
  1121→  1121→      if (!deduped.some((d) => d.step === n.step)) deduped.push(n)
  1122→  1122→    }
  1123→  1123→    return { notes: deduped, onsets: deduped.map((n) => n.step) }
  1124→  1124→  }
  1125→  1125→
  1126→  1126→  /**
  1127→  1127→   * F21: compose a LeadPlan PHRASE-FIRST.
  1128→  1128→   *
  1129→  1129→   * The generation pipeline is:
  1130→  1130→   *   PhraseContext → PhraseIntent → Contour → Harmonic targets
  1131→  1131→   *     → Rhythmic cell → Bass/Kick relational constraints
  1132→  1132→   *     → Phrase transformation → LeadPlan → Scheduled notes
  1133→  1133→   *
  1134→  1134→   * The 16-step grid is the REALIZATION layer — it is NOT the musical
  1135→  1135→   * decision layer. The contour and rhythmic cell are designed at the phrase
  1136→  1136→   * level (driven by the learned identity's lead vocabulary), then placed
  1137→  1137→   * onto the per-bar grid respecting the RhythmicSpaceMap.
  1138→  1138→   *
  1139→  1139→   * Causal paths (all live):
  1140→  1140→   *   LEARNED IDENTITY → contour shape (ascending/descending/arch/wave)
  1141→  1141→   *   LEARNED IDENTITY → interval width (narrow vs wide)
  1142→  1142→   *   HARMONY → LEAD: chord-tone targeting + cadence resolution
  1143→  1143→   *   BASS → LEAD: response placement (leadResponseBoost)
  1144→  1144→   *   ENERGY → DENSITY: note count target
  1145→  1145→   *   TENSION → REGISTER: register center + expansion
  1146→  1146→   *   TENSION → MELODIC: max interval size
  1147→  1147→   *   TENSION → RHYTHMIC: syncopation level
  1148→  1148→   */
  1149→  1149→  private composeLeadPlan(opts: {
  1150→  1150→    bar: number
  1151→  1151→    bars: number
  1152→  1152→    groove: GroovePlan
  1153→  1153→    kickPlan: KickPlan
  1154→  1154→    bassPlan: BassPlan
  1155→  1155→    spaceMap: RhythmicSpaceMap
  1156→  1156→    harmonicPlan: HarmonicPlan
  1157→  1157→    activeChord: HarmonicChord | null
  1158→  1158→    nextChord: HarmonicChord | null
  1159→  1159→    isAnticipation: boolean
  1160→  1160→    isLast: boolean
  1161→  1161→    isResponse: boolean
  1162→  1162→    phraseMaterial: PhraseMaterial
  1163→  1163→    phraseRole: string
  1164→  1164→    rng: Rng
  1165→  1165→    grammarConfidence: number
  1166→  1166→  }): LeadPlan {
  1167→  1167→    const scale = getScale(this.context.scaleName)
  1168→  1168→    if (!scale) return emptyLeadPlan()
  1169→  1169→    const {
  1170→  1170→      groove,
  1171→  1171→      bassPlan,
  1172→  1172→      spaceMap,
  1173→  1173→      harmonicPlan,
  1174→  1174→      activeChord,
  1175→  1175→      nextChord,
  1176→  1176→      isLast,
  1177→  1177→      rng,
  1178→  1178→      grammarConfidence,
  1179→  1179→      phraseMaterial,
  1180→  1180→    } = opts
  1181→  1181→
  1182→  1182→    // ── PHRASE CONTEXT ──
  1183→  1183→    // Gather the phrase-level state: arc stage, tension, learned identity.
  1184→  1184→    const arcStage = arcStageAt(phraseMaterial.phraseArc, opts.bar)
  1185→  1185→    const tensionDims = deriveTensionDimensions(this.context.tension)
  1186→  1186→
  1187→  1187→    // ── PHRASE INTENT ──
  1188→  1188→    // Decide what this bar wants to do within the phrase arc.
  1189→  1189→    const stageLabel = arcStage?.stage ?? 'DEVELOP'
  1190→  1190→    const isFocalBar = stageLabel === 'FOCAL'
  1191→  1191→    const isOpenBar = stageLabel === 'OPEN'
  1192→  1192→    const isCadenceStage = stageLabel === 'CADENCE'
  1193→  1193→    // When no learned identity is present, dampen the arc's effect so bars
  1194→  1194→    // within the same phrase retain motif recurrence (backward compat).
  1195→  1195→    const arcStrength = this.identity ? 1.0 : 0.3
  1196→  1196→
  1197→  1197→    // ── TARGETS from learned grammar + tension dimensions (causal). ──
  1198→  1198→    // ENERGY → DENSITY (with tension density multiplier).
  1199→  1199→    const baseDensity = 0.3 + this.context.energy * 0.4
  1200→  1200→    const learnedDensity = densityForEnergy({
  1201→  1201→      energy: this.context.energy,
  1202→  1202→      grammar: this.interactionGrammar,
  1203→  1203→      confidence: grammarConfidence,
  1204→  1204→      defaultDensity: baseDensity,
  1205→  1205→    })
  1206→  1206→    const targetDensity = applyDensityTension(learnedDensity, this.context.tension)
  1207→  1207→    // TENSION → REGISTER (with tension register expansion).
  1208→  1208→    const baseRegister = this.grammar.tessituraCenter + (this.context.tension - 0.5) * 6
  1209→  1209→    const learnedRegister = registerForTension({
  1210→  1210→      tension: this.context.tension,
  1211→  1211→      grammar: this.interactionGrammar,
  1212→  1212→      confidence: grammarConfidence,
  1213→  1213→      defaultRegister: baseRegister,
  1214→  1214→    })
  1215→  1215→    const targetRegister = applyRegisterTension(learnedRegister, this.context.tension)
  1216→  1216→    // TENSION → MELODIC: max interval size.
  1217→  1217→    const maxInterval = tensionDims.melodic
  1218→  1218→
  1219→  1219→    // ── CONTOUR DESIGN (phrase-level, driven by learned identity) ──
  1220→  1220→    // The contour shape comes from the learned identity's lead vocabulary.
  1221→  1221→    // If no identity, derive from the phrase material's existing contour.
  1222→  1222→    let contourShape: ContourShape = phraseMaterial.contour
  1223→  1223→    if (this.identity) {
  1224→  1224→      switch (this.identity.leadVocabulary) {
  1225→  1225→        case 'DESCENDING_NARROW':
  1226→  1226→          contourShape = opts.bar < opts.bars / 2 ? 'arch' : 'descending'
  1227→  1227→          break
  1228→  1228→        case 'ASCENDING_WIDE':
  1229→  1229→          contourShape = opts.bar < opts.bars / 2 ? 'ascending' : 'arch'
  1230→  1230→          break
  1231→  1231→        case 'ARCH_BALANCED':
  1232→  1232→          contourShape = isFocalBar ? 'arch' : 'flat'
  1233→  1233→          break
  1234→  1234→        case 'WAVE_SYNCOPATED':
  1235→  1235→          contourShape = 'wave'
  1236→  1236→          break
  1237→  1237→      }
  1238→  1238→    }
  1239→  1239→
  1240→  1240→    // ── HARMONIC TARGETS (phrase-level) ──
  1241→  1241→    // Sequence the chord tones to target across the bar, ordered by contour.
  1242→  1242→    const chordPcs = activeChord?.chordTones ?? []
  1243→  1243→    const cadenceTarget = harmonicPlan.cadenceTarget
  1244→  1244→    const rootPc = activeChord?.rootPc ?? this.context.tonic
  1245→  1245→    // Apply harmonic tension: at high tension, add extensions.
  1246→  1246→    const pcs = scalePcs(this.context.tonic, scale)
  1247→  1247→    const extendedChordTones = applyHarmonicTension(chordPcs, this.context.tension, pcs)
  1248→  1248→
  1249→  1249→    // ── RHYTHMIC CELL (phrase-level, driven by learned identity) ──
  1250→  1250→    // The rhythmic cell is the repeating onset pattern. It comes from the
  1251→  1251→    // phrase material's rhythmicCell, or is designed from the space map.
  1252→  1252→    let rhythmicCell = phraseMaterial.rhythmicCell
  1253→  1253→    if (rhythmicCell.length === 0) {
  1254→  1254→      // Design a cell from the space map: pick the 3-4 most open steps.
  1255→  1255→      const openSteps = spaceMap.cells
  1256→  1256→        .filter((c) => c.open)
  1257→  1257→        .sort((a, b) => b.preferredLead - a.preferredLead)
  1258→  1258→        .slice(0, 4)
  1259→  1259→        .map((c) => c.step)
  1260→  1260→        .sort((a, b) => a - b)
  1261→  1261→      rhythmicCell = openSteps.length > 0 ? openSteps : [2, 6, 10, 14]
  1262→  1262→    }
  1263→  1263→
  1264→  1264→    // ── BASS/KICK RELATIONAL CONSTRAINTS ──
  1265→  1265→    // Build the per-step "lead desire" score using the space map + response
  1266→  1266→    // boost + arc stage. This shapes WHERE the rhythmic cell lands.
  1267→  1267→    const bassOnsets = bassPlan.onsets
  1268→  1268→    const leadDesire: number[] = new Array(groove.stepsPerBar).fill(0)
  1269→  1269→    for (let step = 0; step < groove.stepsPerBar; step++) {
  1270→  1270→      const cell = cellAt(spaceMap, step)
  1271→  1271→      let desire = cell.preferredLead
  1272→  1272→      desire += leadResponseBoost({
  1273→  1273→        step,
  1274→  1274→        bassOnsets,
  1275→  1275→        grammar: this.interactionGrammar,
  1276→  1276→        confidence: grammarConfidence,
  1277→  1277→      })
  1278→  1278→      // Focal bars get higher desire across all open steps (climax energy).
  1279→  1279→      if (isFocalBar && cell.open) desire += 0.3 * arcStrength
  1280→  1280→      // Open bars get lower desire (sparse, leaving space).
  1281→  1281→      if (isOpenBar) desire *= 1 - 0.4 * arcStrength
  1282→  1282→      // Anticipation: the step before a chord change is a good lead spot.
  1283→  1283→      if (opts.isAnticipation && step >= groove.stepsPerBar - 2) desire += 0.4
  1284→  1284→      // Cadence: on the last bar, the final steps are reserved.
  1285→  1285→      if (isLast && step >= groove.stepsPerBar - 2) desire += 0.5
  1286→  1286→      leadDesire[step] = desire
  1287→  1287→    }
  1288→  1288→
  1289→  1289→    // Pick the top-N steps by lead desire (N = target density * stepsPerBar).
  1290→  1290→    // This is the REALIZATION layer — the musical decision (contour + cell)
  1291→  1291→    // was already made above; now we place it on the grid.
  1292→  1292→    const targetCount = Math.max(1, Math.round(targetDensity * groove.stepsPerBar))
  1293→  1293→    const stepOrder = leadDesire
  1294→  1294→      .map((desire, step) => ({ step, desire }))
  1295→  1295→      .sort((a, b) => b.desire - a.desire)
  1296→  1296→      .slice(0, targetCount)
  1297→  1297→      .map((x) => x.step)
  1298→  1298→      .sort((a, b) => a - b)
  1299→  1299→
  1300→  1300→    // ── PITCH POOL (from phrase material, transposed to target register) ──
  1301→  1301→    let pitchPool: number[] = phraseMaterial.pitchContour.slice()
  1302→  1302→    if (pitchPool.length === 0) {
  1303→  1303→      pitchPool = pcs.map((pc) => degreeToMidi(this.context.tonic, scale, pcs.indexOf(pc), 4))
  1304→  1304→    }
  1305→  1305→    // Transpose the pool so its mean matches the target register (causal:
  1306→  1306→    // TENSION → REGISTER shifts the actual pitches).
  1307→  1307→    if (pitchPool.length > 0) {
  1308→  1308→      const poolMean = pitchPool.reduce((s, m) => s + m, 0) / pitchPool.length
  1309→  1309→      const offset = Math.round(targetRegister - poolMean)
  1310→  1310→      pitchPool = pitchPool.map((m) => {
  1311→  1311→        const shifted = m + offset
  1312→  1312→        for (let off = 0; off <= 2; off++) {
  1313→  1313→          if (isInScale(this.context.tonic, scale, shifted + off)) return shifted + off
  1314→  1314→          if (isInScale(this.context.tonic, scale, shifted - off)) return shifted - off
  1315→  1315→        }
  1316→  1316→        return shifted
  1317→  1317→      })
  1318→  1318→    }
  1319→  1319→    pitchPool = pitchPool.map((m) => {
  1320→  1320→      let midi = m
  1321→  1321→      while (midi < LEAD_MIN_MIDI) midi += 12
  1322→  1322→      while (midi > LEAD_MAX_MIDI) midi -= 12
  1323→  1323→      return midi
  1324→  1324→    })
  1325→  1325→
  1326→  1326→    // ── REALIZE THE PHRASE ONTO THE GRID ──
  1327→  1327→    // For each chosen step, select a pitch that follows the contour shape and
  1328→  1328→    // targets chord tones. The contour drives the pitch SEQUENCE; the space
  1329→  1329→    // map drives the ONSET positions.
  1330→  1330→    const notes: LeadPlanNote[] = []
  1331→  1331→    let prevMidi: number | null = null
  1332→  1332→
  1333→  1333→    // Build a contour-driven pitch sequence from the pool + chord tones.
  1334→  1334→    // When a learned identity is present, use chord tones as the primary pool
  1335→  1335→    // (identity-driven vocabulary). When no identity, use the phrase material's
  1336→  1336→    // pitchContour as the primary pool (preserves motif recurrence).
  1337→  1337→    const chordToneMidis = extendedChordTones.map((pc) => {
  1338→  1338→      let midi = pc + 60
  1339→  1339→      while (midi < targetRegister - 6) midi += 12
  1340→  1340→      while (midi > targetRegister + 6) midi -= 12
  1341→  1341→      while (midi < LEAD_MIN_MIDI) midi += 12
  1342→  1342→      while (midi > LEAD_MAX_MIDI) midi -= 12
  1343→  1343→      return midi
  1344→  1344→    })
  1345→  1345→    const primaryPool = this.identity
  1346→  1346→      ? chordToneMidis.length > 0
  1347→  1347→        ? chordToneMidis
  1348→  1348→        : pitchPool
  1349→  1349→      : pitchPool
  1350→  1350→
  1351→  1351→    for (let i = 0; i < stepOrder.length; i++) {
  1352→  1352→      const step = stepOrder[i] as number
  1353→  1353→      const isCadenceStep = (isLast || isCadenceStage) && step >= groove.stepsPerBar - 2
  1354→  1354→      const isResponseStep =
  1355→  1355→        leadResponseBoost({
  1356→  1356→          step,
  1357→  1357→          bassOnsets,
  1358→  1358→          grammar: this.interactionGrammar,
  1359→  1359→          confidence: grammarConfidence,
  1360→  1360→        }) > 0.3
  1361→  1361→      const isAnticipationStep = opts.isAnticipation && step >= groove.stepsPerBar - 2
  1362→  1362→      const progress = i / Math.max(1, stepOrder.length - 1) // 0..1 across the bar's notes
  1363→  1363→
  1364→  1364→      let midi: number
  1365→  1365→      let role: LeadPlanNote['role']
  1366→  1366→
  1367→  1367→      if (isCadenceStep) {
  1368→  1368→        // CADENCE: resolve to the cadence target pc.
  1369→  1369→        midi = cadenceMidi(harmonicPlan, cadenceTarget, targetRegister)
  1370→  1370→        if (step !== (stepOrder[stepOrder.length - 1] ?? groove.stepsPerBar - 1)) {
  1371→  1371→          midi = cadenceMidi(harmonicPlan, cadenceTarget, targetRegister) - 2
  1372→  1372→        }
  1373→  1373→        role = 'CADENCE'
  1374→  1374→      } else if (isAnticipationStep && nextChord) {
  1375→  1375→        // ANTICIPATION: play a chord tone of the NEXT chord.
  1376→  1376→        const nextPcs = nextChord.chordTones
  1377→  1377→        midi = (nextPcs[Math.floor(rng.next() * nextPcs.length)] ?? nextChord.rootPc) + 60
  1378→  1378→        while (midi < LEAD_MIN_MIDI) midi += 12
  1379→  1379→        while (midi > LEAD_MAX_MIDI) midi -= 12
  1380→  1380→        role = 'ANTICIPATION'
  1381→  1381→      } else {
  1382→  1382→        // Contour-driven pitch selection.
  1383→  1383→        // Pick a pitch from the primary pool that matches the contour direction.
  1384→  1384→        const candidates = primaryPool.map((candMidi) => {
  1385→  1385→          const interval = prevMidi !== null ? candMidi - prevMidi : 0
  1386→  1386→          const pc = ((candMidi % 12) + 12) % 12
  1387→  1387→          const isChordTone = chordPcs.includes(pc)
  1388→  1388→          const harmonyScore = isChordTone ? 0.7 : 0.05
  1389→  1389→          const intervalScore = leadIntervalScore({
  1390→  1390→            interval,
  1391→  1391→            rootPc,
  1392→  1392→            grammar: this.interactionGrammar,
  1393→  1393→            confidence: grammarConfidence,
  1394→  1394→          })
  1395→  1395→          // Contour preference: score based on whether the candidate matches
  1396→  1396→          // the contour shape's direction at this position.
  1397→  1397→          let contourScore = 0
  1398→  1398→          if (prevMidi !== null) {
  1399→  1399→            const dir = Math.sign(candMidi - prevMidi)
  1400→  1400→            switch (contourShape) {
  1401→  1401→              case 'ascending':
  1402→  1402→                contourScore = dir > 0 ? 0.3 : dir < 0 ? -0.2 : 0
  1403→  1403→                break
  1404→  1404→              case 'descending':
  1405→  1405→                contourScore = dir < 0 ? 0.3 : dir > 0 ? -0.2 : 0
  1406→  1406→                break
  1407→  1407→              case 'arch':
  1408→  1408→                contourScore = progress < 0.5 ? (dir > 0 ? 0.2 : -0.1) : dir < 0 ? 0.2 : -0.1
  1409→  1409→                break
  1410→  1410→              case 'valley':
  1411→  1411→                contourScore = progress < 0.5 ? (dir < 0 ? 0.2 : -0.1) : dir > 0 ? 0.2 : -0.1
  1412→  1412→                break
  1413→  1413→              case 'wave':
  1414→  1414→                contourScore = i % 2 === 0 ? (dir > 0 ? 0.2 : -0.1) : dir < 0 ? 0.2 : -0.1
  1415→  1415→                break
  1416→  1416→              case 'flat':
  1417→  1417→                contourScore = dir === 0 ? 0.2 : 0
  1418→  1418→                break
  1419→  1419→            }
  1420→  1420→          }
  1421→  1421→          // TENSION → MELODIC: penalize intervals wider than maxInterval.
  1422→  1422→          const melodicPenalty = Math.abs(interval) > maxInterval ? -0.5 : 0
  1423→  1423→          // Leap penalty.
  1424→  1424→          const leapPenalty =
  1425→  1425→            prevMidi !== null && Math.abs(interval) > this.grammar.maxLeap ? -0.5 : 0
  1426→  1426→          return {
  1427→  1427→            midi: candMidi,
  1428→  1428→            score: harmonyScore + intervalScore * 0.3 + contourScore + melodicPenalty + leapPenalty,
  1429→  1429→          }
  1430→  1430→        })
  1431→  1431→        candidates.sort((a, b) => b.score - a.score)
  1432→  1432→        const pick =
  1433→  1433→          rng.next() < 0.7
  1434→  1434→            ? (candidates[0] ?? { midi: targetRegister })
  1435→  1435→            : (candidates[Math.floor(rng.next() * Math.min(3, candidates.length))] ?? {
  1436→  1436→                midi: targetRegister,
  1437→  1437→              })
  1438→  1438→        midi = pick.midi
  1439→  1439→        role = isResponseStep ? 'RESPONSE' : opts.isResponse ? 'RESPONSE' : 'CONTINUATION'
  1440→  1440→      }
  1441→  1441→
  1442→  1442→      // Enforce maxLeap by clamping.
  1443→  1443→      if (prevMidi !== null) {
  1444→  1444→        const interval = midi - prevMidi
  1445→  1445→        if (Math.abs(interval) > this.grammar.maxLeap) {
  1446→  1446→          midi = prevMidi + Math.sign(interval) * this.grammar.maxLeap
  1447→  1447→        }
  1448→  1448→      }
  1449→  1449→      while (midi < LEAD_MIN_MIDI) midi += 12
  1450→  1450→      while (midi > LEAD_MAX_MIDI) midi -= 12
  1451→  1451→
  1452→  1452→      const velocity =
  1453→  1453→        role === 'CADENCE'
  1454→  1454→          ? 0.95
  1455→  1455→          : role === 'ANTICIPATION'
  1456→  1456→            ? 0.8
  1457→  1457→            : role === 'RESPONSE'
  1458→  1458→              ? 0.6
  1459→  1459→              : isFocalBar
  1460→  1460→                ? 0.85
  1461→  1461→                : 0.7
  1462→  1462→
  1463→  1463→      notes.push({
  1464→  1464→        midi,
  1465→  1465→        step,
  1466→  1466→        durationSteps: isCadenceStep ? 2 : 1,
  1467→  1467→        velocity,
  1468→  1468→        role,
  1469→  1469→      })
  1470→  1470→      prevMidi = midi
  1471→  1471→    }
  1472→  1472→
  1473→  1473→    return { notes }
  1474→  1474→  }
  1475→  1475→
  1476→  1476→  // ───────────── LEGACY FALLBACKS (for the A/B OFF vs ON test) ─────────────
  1477→  1477→  // These reproduce the pre-F20 "deaf" generators. Used only when
  1478→  1478→  // `relationalGenerationOff` is set, so the A/B test H can measure the
  1479→  1479→  // improvement the relational engine produces.
  1480→  1480→
  1481→  1481→  /**
  1482→  1482→   * Legacy bass: reads groove.kickSteps (the style skeleton), does NOT
  1483→  1483→   * consume the interaction grammar. This is the pre-F20 behaviour.
  1484→  1484→   */
  1485→  1485→  private composeBassLegacy(_bar: number, groove: GroovePlan, rng: Rng, isLast: boolean): BassPlan {
  1486→  1486→    const scale = getScale(this.context.scaleName)
  1487→  1487→    if (!scale) return emptyBassPlan()
  1488→  1488→    const rootMidi = degreeToMidi(this.context.tonic, scale, 0, BASS_OCTAVE)
  1489→  1489→    const fifthMidi = degreeToMidi(this.context.tonic, scale, 4, BASS_OCTAVE)
  1490→  1490→    const octaveMidi = degreeToMidi(this.context.tonic, scale, 0, BASS_OCTAVE + 1)
  1491→  1491→    const thirdMidi = degreeToMidi(this.context.tonic, scale, 2, BASS_OCTAVE)
  1492→  1492→    const seventhMidi = degreeToMidi(this.context.tonic, scale, 6, BASS_OCTAVE)
  1493→  1493→    const beat = Math.max(1, Math.round(groove.stepsPerBar / 4))
  1494→  1494→    const tension = this.grammar.tensionPreference
  1495→  1495→    const bassDegPrefs = this.learned.bass.degreePreferences
  1496→  1496→    const hasLearnedBass =
  1497→  1497→      Object.keys(bassDegPrefs).length > 0 && this.learned.meta.confidence > 0.3
  1498→  1498→
  1499→  1499→    const chooseBassPitch = (rng: Rng): { midi: number; fn: string } => {
  1500→  1500→      if (hasLearnedBass) {
  1501→  1501→        const degrees = Object.entries(bassDegPrefs)
  1502→  1502→        const totalWeight = degrees.reduce((s, [, w]) => s + w, 0)
  1503→  1503→        let r = rng.next() * totalWeight
  1504→  1504→        for (const [degStr, weight] of degrees) {
  1505→  1505→          r -= weight
  1506→  1506→          if (r <= 0) {
  1507→  1507→            const deg = Number.parseInt(degStr, 10)
  1508→  1508→            const midi = degreeToMidi(this.context.tonic, scale, deg, BASS_OCTAVE)
  1509→  1509→            const fns = ['ROOT', 'THIRD', 'FIFTH', 'SEVENTH', 'OCTAVE']
  1510→  1510→            return { midi, fn: fns[deg] ?? 'PASSING' }
  1511→  1511→          }
  1512→  1512→        }
  1513→  1513→      }
  1514→  1514→      const useFifth = rng.next() < 0.5
  1515→  1515→      return useFifth ? { midi: fifthMidi, fn: 'FIFTH' } : { midi: octaveMidi, fn: 'OCTAVE' }
  1516→  1516→    }
  1517→  1517→
  1518→  1518→    const notes: BassPlanNote[] = []
  1519→  1519→    notes.push({
  1520→  1520→      midi: rootMidi,
  1521→  1521→      step: 0,
  1522→  1522→      durationSteps: 2,
  1523→  1523→      function: 'ROOT',
  1524→  1524→      isAnticipation: false,
  1525→  1525→    })
  1526→  1526→
  1527→  1527→    // LEGACY: reads groove.kickSteps (style skeleton), NOT the actual kickPlan.
  1528→  1528→    if (groove.bassKickAlignment === 'LOCKED') {
  1529→  1529→      for (const step of groove.kickSteps) {
  1530→  1530→        if (step === 0) continue
  1531→  1531→        if (rng.next() < 0.3 + tension * 0.2) {
  1532→  1532→          const pitch = chooseBassPitch(rng)
  1533→  1533→          const dur = rng.next() < 0.5 ? 1 : 2
  1534→  1534→          notes.push({
  1535→  1535→            midi: pitch.midi,
  1536→  1536→            step,
  1537→  1537→            durationSteps: dur,
  1538→  1538→            function: pitch.fn as BassPlanNote['function'],
  1539→  1539→            isAnticipation: false,
  1540→  1540→          })
  1541→  1541→        } else {
  1542→  1542→          notes.push({
  1543→  1543→            midi: rootMidi,
  1544→  1544→            step,
  1545→  1545→            durationSteps: 2,
  1546→  1546→            function: 'ROOT',
  1547→  1547→            isAnticipation: false,
  1548→  1548→          })
  1549→  1549→        }
  1550→  1550→      }
  1551→  1551→    } else {
  1552→  1552→      const half = Math.max(1, Math.round(groove.stepsPerBar / 8))
  1553→  1553→      const complementary = [half, beat + half, beat * 2 + half, beat * 3 + half]
  1554→  1554→      for (const step of complementary) {
  1555→  1555→        if (groove.kickSteps.includes(step) || step === 0) continue
  1556→  1556→        const pitch = chooseBassPitch(rng)
  1557→  1557→        notes.push({
  1558→  1558→          midi: pitch.midi,
  1559→  1559→          step,
  1560→  1560→          durationSteps: 1,
  1561→  1561→          function: pitch.fn as BassPlanNote['function'],
  1562→  1562→          isAnticipation: false,
  1563→  1563→        })
  1564→  1564→      }
  1565→  1565→    }
  1566→  1566→
  1567→  1567→    if (rng.next() < 0.15 + tension * 0.15) {
  1568→  1568→      const jumpStep = beat * 2
  1569→  1569→      if (!notes.some((n) => n.step === jumpStep)) {
  1570→  1570→        notes.push({
  1571→  1571→          midi: octaveMidi,
  1572→  1572→          step: jumpStep,
  1573→  1573→          durationSteps: 1,
  1574→  1574→          function: 'OCTAVE',
  1575→  1575→          isAnticipation: false,
  1576→  1576→        })
  1577→  1577→      }
  1578→  1578→    }
  1579→  1579→
  1580→  1580→    if (isLast) {
  1581→  1581→      const lastStep = groove.stepsPerBar - 2
  1582→  1582→      const filtered = notes.filter((n) => n.step < lastStep)
  1583→  1583→      notes.length = 0
  1584→  1584→      notes.push(...filtered)
  1585→  1585→      notes.push({
  1586→  1586→        midi: fifthMidi,
  1587→  1587→        step: lastStep,
  1588→  1588→        durationSteps: 1,
  1589→  1589→        function: 'CADENCE',
  1590→  1590→        isAnticipation: false,
  1591→  1591→      })
  1592→  1592→      notes.push({
  1593→  1593→        midi: rootMidi,
  1594→  1594→        step: lastStep + 1,
  1595→  1595→        durationSteps: 2,
  1596→  1596→        function: 'CADENCE',
  1597→  1597→        isAnticipation: false,
  1598→  1598→      })
  1599→  1599→    }
  1600→  1600→
  1601→  1601→    for (const n of notes) {
  1602→  1602→      while (n.midi < 36) n.midi += 12
  1603→  1603→      while (n.midi > 59) n.midi -= 12
  1604→  1604→    }
  1605→  1605→    notes.sort((a, b) => a.step - b.step)
  1606→  1606→    const deduped: BassPlanNote[] = []
  1607→  1607→    for (const n of notes) {
  1608→  1608→      if (!deduped.some((d) => d.step === n.step)) deduped.push(n)
  1609→  1609→    }
  1610→  1610→    void thirdMidi
  1611→  1611→    void seventhMidi
  1612→  1612→    return { notes: deduped, onsets: deduped.map((n) => n.step) }
  1613→  1613→  }
  1614→  1614→
  1615→  1615→  /**
  1616→  1616→   * Legacy lead: motif + chord snapping only. Does NOT see bass, kick, groove,
  1617→  1617→   * space map, or interaction grammar. This is the pre-F20 "deaf" lead.
  1618→  1618→   */
  1619→  1619→  private composeLeadLegacy(
  1620→  1620→    bar: number,
  1621→  1621→    bars: number,
  1622→  1622→    phraseMotif: Motif,
  1623→  1623→    harmonicContext: number[],
  1624→  1624→    rng: Rng,
  1625→  1625→    isResponse: boolean
  1626→  1626→  ): { midi: number; step: number; durationSteps: number; velocity: number }[] {
  1627→  1627→    const scale = getScale(this.context.scaleName)
  1628→  1628→    if (!scale || phraseMotif.notes.length === 0) return []
  1629→  1629→
  1630→  1630→    let motif = phraseMotif
  1631→  1631→    if (isResponse) {
  1632→  1632→      motif = callResponse(phraseMotif, this.context.tonic, scale, (this.seed + bar * 17) >>> 0)
  1633→  1633→    } else if (bar > 0) {
  1634→  1634→      const variationChoice = (bar * 3 + this.seed) % 5
  1635→  1635→      if (variationChoice === 0) {
  1636→  1636→        const t = rng.pick([-3, -2, 2, 3])
  1637→  1637→        motif = transposeMotif(phraseMotif, t, this.context.tonic, scale)
  1638→  1638→      } else if (variationChoice === 1) {
  1639→  1639→        if (rng.next() < 0.3) {
  1640→  1640→          motif = invertMotif(phraseMotif, this.context.tonic, scale)
  1641→  1641→        } else {
  1642→  1642→          const t = rng.pick([-2, 0, 2])
  1643→  1643→          if (t !== 0) motif = transposeMotif(phraseMotif, t, this.context.tonic, scale)
  1644→  1644→        }
  1645→  1645→      } else if (variationChoice === 2) {
  1646→  1646→        if (rng.next() < 0.2) {
  1647→  1647→          motif = retrogradeMotif(phraseMotif)
  1648→  1648→        }
  1649→  1649→      } else if (variationChoice === 3) {
  1650→  1650→        if (rng.next() < 0.25) {
  1651→  1651→          const halfNotes = phraseMotif.notes.slice(0, Math.ceil(phraseMotif.notes.length / 2))
  1652→  1652→          if (halfNotes.length > 0) {
  1653→  1653→            motif = { ...phraseMotif, notes: halfNotes }
  1654→  1654→          }
  1655→  1655→        }
  1656→  1656→      }
  1657→  1657→    }
  1658→  1658→
  1659→  1659→    const notes = motif.notes.map((n) => {
  1660→  1660→      let midi = n.midi
  1661→  1661→      while (midi < LEAD_MIN_MIDI) midi += 12
  1662→  1662→      while (midi > LEAD_MAX_MIDI) midi -= 12
  1663→  1663→      return { midi, step: n.step, durationSteps: n.durationSteps, velocity: n.velocity }
  1664→  1664→    })
  1665→  1665→
  1666→  1666→    const maxLeap = this.grammar.maxLeap
  1667→  1667→    for (let i = 1; i < notes.length; i++) {
  1668→  1668→      const prev = notes[i - 1]
  1669→  1669→      const cur = notes[i]
  1670→  1670→      if (!prev || !cur) continue
  1671→  1671→      const interval = cur.midi - prev.midi
  1672→  1672→      if (Math.abs(interval) > maxLeap) {
  1673→  1673→        cur.midi = prev.midi + Math.sign(interval) * maxLeap
  1674→  1674→      }
  1675→  1675→    }
  1676→  1676→
  1677→  1677→    const chordPcs = new Set(harmonicContext)
  1678→  1678→    if (chordPcs.size > 0) {
  1679→  1679→      for (const n of notes) {
  1680→  1680→        const pc = ((n.midi % 12) + 12) % 12
  1681→  1681→        if (!chordPcs.has(pc) && rng.next() < 0.5) {
  1682→  1682→          for (let off = 1; off <= 2; off++) {
  1683→  1683→            const upPc = (((n.midi + off) % 12) + 12) % 12
  1684→  1684→            if (chordPcs.has(upPc)) {
  1685→  1685→              n.midi += off
  1686→  1686→              break
  1687→  1687→            }
  1688→  1688→            const downPc = (((n.midi - off) % 12) + 12) % 12
  1689→  1689→            if (chordPcs.has(downPc)) {
  1690→  1690→              n.midi -= off
  1691→  1691→              break
  1692→  1692→            }
  1693→  1693→          }
  1694→  1694→        }
  1695→  1695→      }
  1696→  1696→    }
  1697→  1697→    void bars
  1698→  1698→    return notes
  1699→  1699→  }
  1700→  1700→}

  1716→  1716→      role: motif.role,
  1717→  1717→      sourceMotifId: motif.id,
  1718→  1718→      transformHistory: [...motif.transformHistory, 'invertPitchPure'],
  1719→  1719→    })
  1720→  1720→  }

  1722→  1722→  const notes = motif.notes.map((n) => {
  1723→  1723→    if (n === first) return { ...n }

  1732→  1732→    role: motif.role,

  1747→  1747→/** Convenience: how tightly the bass locks with the kick on beat 1. */
  1748→  1748→export function measureBassKickAlignment(
  1749→  1749→  bassNotes: { step: number; bar: number }[],
  1750→  1750→  kickNotes: { step: number; bar: number }[],
  1751→  1751→  bars: number
  1752→  1752→): number {
  1753→  1753→  if (bars === 0) return 0
  1754→  1754→  let aligned = 0
  1755→  1755→  for (let bar = 0; bar < bars; bar++) {
  1756→  1756→    const bassOnBeat1 = bassNotes.some((n) => n.bar === bar && n.step === 0)
  1757→  1757→    const kickOnBeat1 = kickNotes.some((n) => n.bar === bar && n.step === 0)
  1758→  1758→    if (bassOnBeat1 && kickOnBeat1) aligned++
  1759→  1759→  }
  1760→  1760→  return aligned / bars
  1761→  1761→}
