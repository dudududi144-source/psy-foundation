     1→/**
     2→ * PSY4 Engine — Single AudioWorklet processor.
     3→ *
     4→ * This is the REAL-TIME PRODUCTION ENGINE. It replaces the setInterval(25ms)
     5→ * main-thread scheduler and the per-hit Web Audio node creation that caused
     6→ * latency, jitter, and GC pressure under dense events.
     7→ *
     8→ * Architecture:
     9→ *   Main thread (controller)
    10→ *     ↓ port.postMessage (commands + event batches)
    11→ *   AudioWorklet (this file)
    12→ *     ├── Transport (BPM, step, section — sample-accurate clock)
    13→ *     ├── Ring-buffer event queue (zero allocation)
    14→ *     ├── Preallocated voice pool (kick, bass, lead, acid, pad, hats, ...)
    15→ *     ├── Voice DSP (Moog ladder, BL saw, envelopes — all inline)
    16→ *     ├── Bus mixing (drum/bass/music/atmos/fx → master)
    17→ *     └── Master chain (saturation + limiter)
    18→ *     ↓ stereo output
    19→ *   Speakers
    20→ *
    21→ * The main thread NEVER determines when a kick fires. It sends high-level
    22→ * musical events ("kick at time T, velocity 0.9") and the worklet executes
    23→ * them sample-accurately. The main thread can be blocked by React/GC without
    24→ * affecting audio timing.
    25→ *
    26→ * Ported DSP from PSY3:
    27→ *   - pro_dsp.py moog()       → MoogLadder class (4-stage tanh)
    28→ *   - pro_dsp.py bl_saw()     → polyBLEP sawtooth
    29→ *   - engine.py kick()        → KickVoice (sub + mid + click)
    30→ *   - engine.py bass()        → BassVoice (saw + Moog + sub)
    31→ *   - engine.py hat()         → HatVoice (differentiated pink noise)
    32→ *   - engine.py clap()        → ClapVoice (multi-burst noise)
    33→ *   - style_master.py _sat()  → master saturation
    34→ *   - style_master.py limiter → master limiter
    35→ */

    61→// ─── Fast polynomial tanh (Pade approximation, PSY5 pattern) ───────────────
    62→// 10x cheaper than Math.tanh (no transcendental call, just multiply + add).
    63→// Accuracy: max error ~0.005 in [-3, 3]; saturates cleanly outside.
    64→// Replaces the lookup-table fastTanh (which required a table + interpolation).
    65→function fastTanh(x) {
    66→  if (x > 3) return 1;
    67→  if (x < -3) return -1;
    68→  const x2 = x * x;
    69→  return x * (27 + x2) / (27 + 9 * x2);
    70→}
    71→// Alias so existing call sites that use `ftanh` (PSY5 naming) also work.
    72→const ftanh = fastTanh;
    73→
    74→// ─── polyBLEP ──────────────────────────────────────────────────────────────
    75→
    76→function polyBlep(phase, inc) {
    77→  if (phase < inc) {
    78→    const t = phase / inc;
    79→    return 2 * t - t * t - 1;
    80→  } else if (phase > 1 - inc) {
    81→    const t = (phase - 1) / inc;
    82→    return t * t + 2 * t + 1;
    83→  }
    84→  return 0;
    85→}

    87→// ─── Moog Ladder Filter (4-stage tanh, stateful) ───────────────────────────
    88→// Port of PSY3 pro_dsp.py moog(). Reusable per-voice instance.
    89→
    90→class MoogLadder {
    91→  constructor() {
    92→    this.s0 = 0; this.s1 = 0; this.s2 = 0; this.s3 = 0;
    93→    this.g = 0;
    94→    this.lastCutoff = -1;
    95→  }
    96→
    97→  reset() { this.s0 = this.s1 = this.s2 = this.s3 = 0; }
    98→
    99→  process(x, cutoff, res, drive, sr, tol) {
   100→    // Recompute g when cutoff changes
   101→    if (Math.abs(cutoff - this.lastCutoff) > 0.5) {
   102→      const fc = Math.min(0.45, cutoff / sr);
   103→      this.g = 1 - Math.exp(-2 * Math.PI * fc);
   104→      this.lastCutoff = cutoff;
   105→    }
   106→    const g = this.g;
   107→    const fb = res * 4 * fastTanh(this.s3);
   108→    const u = fastTanh((x - fb) * drive);
   109→    let prev = u;
   110→    // Component tolerance: 4 filter stages have slightly different characteristics
   111→    // (PSY3 analog modeling — aTol = [0.98, 1.02, 0.99, 1.01])
   112→    // Each stage's integrator coefficient is slightly modulated by its tolerance.
   113→    // When tol is undefined (most voices), all stages are identical (tol = 1).
   114→    const t0 = tol ? tol[0] : 1, t1 = tol ? tol[1] : 1;
   115→    const t2 = tol ? tol[2] : 1, t3 = tol ? tol[3] : 1;
   116→    this.s0 += g * t0 * (fastTanh(prev) - this.s0); prev = this.s0;
   117→    this.s1 += g * t1 * (fastTanh(prev) - this.s1); prev = this.s1;
   118→    this.s2 += g * t2 * (fastTanh(prev) - this.s2); prev = this.s2;
   119→    this.s3 += g * t3 * (fastTanh(prev) - this.s3);
   120→    return this.s3 / (1 + res * 0.5);
   121→  }
   122→}

   124→// ─── One-pole lowpass (for envelopes, simple filters) ──────────────────────

   126→class OnePoleLP {
   127→  constructor() { this.v = 0; }
   128→  reset() { this.v = 0; }
   129→  process(x, cutoff, sr) {
   130→    const a = (1 / sr) * 2 * Math.PI * cutoff;
   131→    this.v += a * (x - this.v) / (1 + a);
   132→    return this.v;
   133→  }
   134→}

   136→// ─── Pink noise generator (stateful, Voss-McCartney) ───────────────────────
   137→
   138→class PinkNoise {
   139→  constructor() {
   140→    this.b = new Float32Array(7);
   141→    this.rngState = 12345;
   142→  }
   143→  reset() { this.b.fill(0); }
   144→  // Gaussian approximation (sum of 3 uniforms → triangular ≈ Gaussian by CLT)
   145→  // PSY3 uses rng.standard_normal() — this gives a more natural noise character
   146→  // than uniform Math.random(). Summing 3 independent uniforms produces a
   147→  // triangular distribution that closely approximates Gaussian.
   148→  next() {
   149→    let s = 0;
   150→    for (let i = 0; i < 3; i++) {
   151→      this.rngState = (this.rngState * 1103515245 + 12345) & 0x7fffffff;
   152→      s += (this.rngState / 0x3fffffff) - 1; // each in [-1, 1]
   153→    }
   154→    return s * 0.3333; // ~Gaussian, range ≈ [-1, 1]
   155→  }
   156→  // Pink noise sample
   157→  process() {
   158→    const w = this.next();
   159→    this.b[0] = 0.99886 * this.b[0] + w * 0.0555179;
   160→    this.b[1] = 0.99332 * this.b[1] + w * 0.0750759;
   161→    this.b[2] = 0.96900 * this.b[2] + w * 0.1538520;
   162→    this.b[3] = 0.86650 * this.b[3] + w * 0.3104856;
   163→    this.b[4] = 0.55000 * this.b[4] + w * 0.5329522;
   164→    this.b[5] = -0.7616 * this.b[5] - w * 0.0168980;
   165→    const p = this.b[0] + this.b[1] + this.b[2] + this.b[3] + this.b[4] + this.b[5] + this.b[6] + w * 0.5362;
   166→    this.b[6] = w * 0.115926;
   167→    return p * 0.11;
   168→  }
   169→}

   171→// ─── ADSR Envelope ─────────────────────────────────────────────────────────

   173→class ADSR {
   174→  constructor() { this.stage = 4; this.t = 0; this.value = 0; }
   175→  trigger(a, d, s, r) { this.stage = 0; this.t = 0; this.a = a; this.d = d; this.s = s; this.r = r; this.value = 0; }
   176→  release() { if (this.stage < 3) { this.stage = 3; this.t = 0; } }
   177→  process(dt) {
   178→    if (this.stage >= 4) return 0;
   179→    this.t += dt;
   180→    if (this.stage === 0) { // attack
   181→      this.value = this.t / Math.max(0.0001, this.a);
   182→      if (this.t >= this.a) { this.stage = 1; this.t = 0; this.value = 1; }
   183→    } else if (this.stage === 1) { // decay
   184→      this.value = 1 - (1 - this.s) * (this.t / Math.max(0.0001, this.d));
   185→      if (this.t >= this.d) { this.stage = 2; this.value = this.s; }
   186→    } else if (this.stage === 2) { // sustain
   187→      this.value = this.s;
   188→    } else if (this.stage === 3) { // release
   189→      this.value = this.s * (1 - this.t / Math.max(0.0001, this.r));
   190→      if (this.t >= this.r) { this.stage = 4; this.value = 0; }
   191→    }
   192→    return Math.max(0, Math.min(1, this.value));
   193→  }
   194→  get done() { return this.stage >= 4; }
   195→}

   197→// ─── Exponential decay envelope (for percussive voices) ────────────────────
   198→
   199→class DecayEnv {
   200→  constructor() { this.t = 0; this.decay = 0.1; this.active = false; }
   201→  trigger(decay) { this.t = 0; this.decay = Math.max(0.001, decay); this.active = true; }
   202→  process(dt) {
   203→    if (!this.active) return 0;
   204→    this.t += dt;
   205→    const v = Math.exp(-this.t / this.decay);
   206→    if (v < 0.0001) { this.active = false; return 0; }
   207→    return v;
   208→  }
   209→  get done() { return !this.active; }
   210→}

   212→// ─── Band-limited sawtooth oscillator (polyBLEP) ───────────────────────────

   214→class BLSaw {
   215→  constructor() { this.phase = 0; this.freq = 220; }
   216→  setFreq(f) { this.freq = f; }
   217→  process(inc) {
   218→    const val = 2 * this.phase - 1;
   219→    const corrected = val - polyBlep(this.phase, inc);
   220→    this.phase += inc;
   221→    if (this.phase >= 1) this.phase -= 1;
   222→    return corrected;
   223→  }
   224→  reset() { this.phase = 0; }
   225→}

   227→// ─── Band-limited square oscillator (polyBLEP) ─────────────────────────────
   228→
   229→class BLSquare {
   230→  constructor() { this.phase = 0; this.freq = 220; }
   231→  setFreq(f) { this.freq = f; }
   232→  process(inc) {
   233→    let val = this.phase < 0.5 ? 1 : -1;
   234→    val += polyBlep(this.phase, inc);
   235→    let p2 = this.phase + 0.5;
   236→    if (p2 >= 1) p2 -= 1;
   237→    val -= polyBlep(p2, inc);
   238→    this.phase += inc;
   239→    if (this.phase >= 1) this.phase -= 1;
   240→    return val;
   241→  }
   242→  reset() { this.phase = 0; }
   243→}

   245→// ─── Voice: Kick (PSY3 engine.py kick) ─────────────────────────────────────
   246→// sub (pitched sine) + mid (saturated triangle) + click (differentiated noise)
   247→
   248→class KickVoice {
   249→  constructor() {
   250→    this.active = false;
   251→    this.t = 0;
   252→    this.amp = 1;
   253→    this.fund = 50;
   254→    // Multi-layer params (PSY3 analog modeling)
   255→    // PSY3 kick has: subDecay, midDecay, clickDecay, subLevel, midLevel, clickLevel,
   256→    //                fundamental, startMult, pitchDecay, saturation
   257→    this.subDecay = 0.2;
   258→    this.midDecay = 0.05;
   259→    this.clickDecay = 0.002;
   260→    this.subLevel = 0.8;
   261→    this.midLevel = 0.5;
   262→    this.clickLevel = 0.35;
   263→    this.startMult = 2.4;
   264→    this.pitchDecay = 0.04;
   265→    this.saturation = 1.5;
   266→    this.phase = 0;
   267→    this.midPhase = 0;
   268→    this.prevNoise = 0;
   269→    this.noise = new PinkNoise();
   270→  }
   271→
   272→  trigger(time, amp, fund, decay, sr) {
   273→    this.active = true;
   274→    this.t = 0;
   275→    this.amp = amp;
   276→    this.fund = fund;
   277→    // Independent decay per layer (PSY3 model) — sub carries the weight,
   278→    // mid provides body/punch, click provides beater attack
   279→    this.subDecay = Math.max(0.05, decay);
   280→    this.midDecay = this.subDecay * 0.25;   // mid decays 4x faster
   281→    this.clickDecay = 0.002;                 // 2ms click
   282→    this.phase = 0;
   283→    this.midPhase = 0;
   284→    this.prevNoise = 0;
   285→    this.noise.reset();
   286→  }
   287→
   288→  // Returns [sample, done]
   289→  render(currentTime, sr) {
   290→    if (!this.active) return [0, true];
   291→    const dt = 1 / sr;
   292→    this.t += dt;
   293→    if (this.t > this.subDecay + 0.05) { this.active = false; return [0, true]; }
   294→
   295→    const t = this.t;
   296→    const f0 = this.fund;
   297→
   298→    // Pitch envelope: f0*startMult → f0 over pitchDecay
   299→    const f = (f0 * this.startMult - f0) * Math.exp(-t / this.pitchDecay) + f0;
   300→
   301→    // Layer 1: SUB — pitched sine with integrated phase (pitch sweep)
   302→    // This is the weight/foundation of the kick
   303→    this.phase += 2 * Math.PI * f / sr;
   304→    const subEnv = Math.exp(-t / (this.subDecay * 0.9));
   305→    const sub = Math.sin(this.phase) * subEnv * this.subLevel;
   306→
   307→    // Layer 2: MID — saturated triangle at fundamental, short decay
   308→    // This is the punch/body — gives the kick its "thwack"
   309→    this.midPhase += 2 * Math.PI * f0 / sr;
   310→    const triPhase = (this.midPhase / (2 * Math.PI)) % 1;
   311→    const tri = 2 * Math.abs(2 * triPhase - 1) - 1;
   312→    const midEnv = Math.exp(-t / this.midDecay);
   313→    const mid = fastTanh(tri * 1.5) * midEnv * this.midLevel;
   314→
   315→    // Layer 3: CLICK — differentiated Gaussian noise, very short
   316→    // This is the beater attack — the transient that cuts through the mix
   317→    const n = this.noise.next();
   318→    const click = (n - this.prevNoise) * Math.exp(-t / this.clickDecay) * this.clickLevel;
   319→    this.prevNoise = n;
   320→
   321→    // Mix layers
   322→    let sample = sub + mid + click;
   323→    // SATURATION: Post-mix tanh (adds harmonics + punch — commercial kicks
   324→    // always have saturation. Without it, the kick sounds flat and digital.)
   325→    sample = fastTanh(sample * this.saturation);
   326→    sample *= this.amp * 0.8;
   327→    return [sample, false];
   328→  }
   329→}

   331→// ─── Voice: Bass (PSY3 engine.py bass) ─────────────────────────────────────
   332→// BL saw → Moog filter (cutoff envelope) + sub sine
   333→
   334→class BassVoice {
   335→  constructor() {
   336→    this.active = false;
   337→    this.t = 0;
   338→    this.freq = 80;
   339→    this.amp = 0.5;
   340→    this.dur = 0.2;
   341→    this.acid = false;
   342→    this.square = new BLSquare();
   343→    this.saw = new BLSaw();
   344→    this.filter = new MoogLadder();
   345→    this.phase = 0;
   346→    this.cutoffStart = 800;
   347→    this.cutoffEnd = 200;
   348→    this.res = 0.1;
   349→    this.bassDecay = 0.12;
   350→    // Post-filter state (one-pole HP for cleaning mud)
   351→    this.hpState = 0;
   352→    // PSY3 bass params: subLevel, harmonicLevel, cutoffFloor, cutoffDecay
   353→    this.subLevel = 0.45;       // sub oscillator level (was hardcoded 0.45)
   354→    this.harmonicLevel = 0.55;  // harmonic (filtered osc) level (was hardcoded 0.55)
   355→    this.cutoffFloor = 80;      // minimum cutoff (prevents filter from closing fully)
   356→    this.cutoffDecay = 0.04;    // cutoff envelope decay time (was hardcoded 0.04)
   357→  }
   358→
   359→  trigger(time, freq, dur, amp, acid, sr, params) {
   360→    this.active = true;
   361→    this.t = 0;
   362→    this.freq = freq;
   363→    this.dur = dur;
   364→    this.amp = amp;
   365→    this.acid = acid;
   366→    this.phase = 0;
   367→    this.hpState = 0;
   368→    this.square.reset();
   369→    this.square.setFreq(freq);
   370→    this.saw.reset();
   371→    this.saw.setFreq(freq);
   372→    this.filter.reset();
   373→    // PSY3 bass params (with defaults for backward compat)
   374→    this.subLevel = params?.subLevel ?? 0.45;
   375→    this.harmonicLevel = params?.harmonicLevel ?? 0.55;
   376→    this.cutoffFloor = params?.cutoffFloor ?? 80;
   377→    this.cutoffDecay = params?.cutoffDecay ?? 0.04;
   378→    if (acid) {
   379→      this.cutoffStart = 2500;
   380→      this.cutoffEnd = 100;
   381→      this.res = 0.85;
   382→      this.bassDecay = 0.15;
   383→    } else {
   384→      this.cutoffStart = params?.cutoffStart ?? 800;
   385→      this.cutoffEnd = params?.cutoffEnd ?? 200;
   386→      this.res = Math.min(0.3, (params?.resonance ?? 3) / 20);
   387→      this.bassDecay = 0.12;
   388→    }
   389→  }
   390→
   391→  render(currentTime, sr) {
   392→    if (!this.active) return [0, true];
   393→    const dt = 1 / sr;
   394→    this.t += dt;
   395→    if (this.t > this.bassDecay) { this.active = false; return [0, true]; }
   396→
   397→    const inc = this.freq / sr;
   398→    const osc = this.acid ? this.saw.process(inc) : this.square.process(inc);
   399→
   400→    // 1. FILTER: Moog ladder with envelope (this is the tone-shaping stage)
   401→    //    Uses configurable cutoffDecay (PSY3) instead of hardcoded 0.04
   402→    //    Floor prevents the filter from fully closing (cutoffFloor)
   403→    const cutoffEnv = (this.cutoffStart - this.cutoffEnd) * Math.exp(-this.t / this.cutoffDecay) + this.cutoffEnd;
   404→    const cutoff = Math.max(this.cutoffFloor, cutoffEnv);
   405→    const drive = this.acid ? 2.5 : 1.3;
   406→    const filtered = this.filter.process(osc, cutoff, this.res, drive, sr);
   407→
   408→    // 2. SUB: Clean sine at fundamental (separate from body — provides weight)
   409→    //    Uses configurable subLevel (PSY3) instead of hardcoded 0.45
   410→    this.phase += 2 * Math.PI * this.freq / sr;
   411→    const sub = Math.sin(this.phase) * this.subLevel;
   412→
   413→    // 3. MIX: Body (filtered) + Sub (clean) — body provides character, sub provides weight
   414→    //    Uses configurable harmonicLevel (PSY3) instead of hardcoded 0.55
   415→    let mixed = filtered * this.harmonicLevel + sub * this.subLevel;
   416→
   417→    // 4. SATURATION: Post-mix tanh saturation (adds harmonics + warmth — this is what makes
   418→    //    a bass sound "produced" rather than "raw oscillator")
   419→    //    Commercial bass always has saturation. Without it, the bass sounds thin and digital.
   420→    mixed = fastTanh(mixed * 1.8);  // drive=1.8 — moderate, adds warmth without distortion
   421→
   422→    // 5. HP FILTER: Remove subsonic mud below 30Hz (one-pole HP)
   423→    //    Prevents the bass from interfering with the kick's sub region
   424→    const hpCutoff = 30;  // Hz
   425→    const hpA = (1 / sr) * 2 * Math.PI * hpCutoff;
   426→    this.hpState += hpA * (mixed - this.hpState) / (1 + hpA);
   427→    mixed = mixed - this.hpState * 0.7;  // partial HP — keep some sub but remove mud
   428→
   429→    // 6. AMP ENVELOPE: Fast attack (1ms) + exponential decay
   430→    const attackEnv = Math.min(1, this.t / 0.001);
   431→    const decayEnv = Math.exp(-this.t / (this.bassDecay * 0.5));
   432→    const ampEnv = attackEnv * decayEnv;
   433→
   434→    return [mixed * ampEnv * this.amp, false];
   435→  }
   436→}

   464→    this.detune = params?.detune ?? 10;
   465→    this.cutoff = params?.cutoff ?? 1800;
   466→    this.res = Math.min(1, (params?.resonance ?? 2) / 20);
   467→    this.lfoRate = params?.lfoRate ?? 0.8;
   468→    this.lfoDepth = params?.lfoDepth ?? 0.3;
   469→    this.lfoPhase = 0;
   470→    // PSY3 lead: filter envelope amount (how much the envelope opens the filter)
   471→    // Higher = more dramatic filter sweep on each note
   472→    this.filterEnvAmount = params?.filterEnvAmount ?? 1.0;
   473→    for (const s of this.saws) { s.reset(); }
   474→    const n = this.saws.length;
   475→    for (let i = 0; i < n; i++) {
   476→      const cents = (i - (n - 1) / 2) * this.detune;
   477→      const mult = Math.pow(2, cents / 1200);
   478→      this.saws[i].setFreq(freq * mult);
   479→    }
   480→    // Octave-up layer — adds brightness and richness
   481→    for (let i = 0; i < this.octaveSaws.length; i++) {
   482→      this.octaveSaws[i].reset();
   483→      const cents = (i - 1) * this.detune * 0.6;
   484→      this.octaveSaws[i].setFreq(freq * 2 * Math.pow(2, cents / 1200));
   485→    }
   486→    this.filter.reset();
   487→    this.noise.reset();
   488→  }
   489→
   490→  render(currentTime, sr) {
   491→    if (!this.active) return [0, true];
   492→    const dt = 1 / sr;
   493→    this.t += dt;
   494→    if (this.t > this.dur + 0.05) { this.active = false; return [0, true]; }
   495→
   496→    // BUG FIX: Use each saw's OWN frequency (set via setFreq in trigger) — NOT the base freq.
   497→    // Previously used `const inc = this.freq / sr` for all saws, which ignored the detune
   498→    // and made all saws play the same frequency. This made leadDetune a DEAD parameter.
   499→
   500→    // Layer 1: Fundamental — 5 detuned saws
   501→    let fundamental = 0;
   502→    for (const s of this.saws) fundamental += s.process(s.freq / sr);
   503→    fundamental /= this.saws.length;
   504→
   505→    // Layer 2: Octave-up — 3 detuned saws at 2x freq (adds brightness/air)
   506→    let octaveLayer = 0;
   507→    for (const s of this.octaveSaws) octaveLayer += s.process(s.freq / sr);
   508→    octaveLayer /= this.octaveSaws.length;
   509→
   510→    // Layer 3: Air — pink noise through high-pass (adds "breath" and sheen)
   511→    const noiseSample = this.noise.process();
   512→    const air = (noiseSample - this.noise.prevOutput || 0) * 0.08; // differentiated = HP
   513→
   514→    // Mix: fundamental dominant, octave at 30%, air at 8%
   515→    let mix = fundamental * 0.7 + octaveLayer * 0.3 + air * 0.08;
   516→
   517→    // LFO modulates filter cutoff (psychedelic movement)
   518→    this.lfoPhase += this.lfoRate * dt;
   519→    const lfo = 0.5 + 0.5 * Math.sin(2 * Math.PI * this.lfoPhase);
   520→    const modCutoff = this.cutoff * (1 + this.lfoDepth * (lfo * 2 - 1) * 0.5);
   521→
   522→    // Filter envelope: open → settle
   523→    // PSY3: filterEnvAmount controls how much the envelope opens the filter
   524→    // (was hardcoded to 2x — now configurable for proper supersaw expression)
   525→    const fEnv = this.cutoff * (1 + this.filterEnvAmount) * Math.exp(-this.t / (this.dur * 0.5)) + this.cutoff;
   526→    const cutoff = Math.min(18000, Math.max(100, fEnv * 0.5 + modCutoff * 0.5));
   527→
   528→    const filtered = this.filter.process(mix, cutoff, this.res, 1.5, sr);
   529→
   530→    // SATURATION: Post-filter tanh — adds character and warmth
   531→    const saturated = fastTanh(filtered * 1.6);
   532→
   533→    // Amp envelope
   534→    const ampEnv = Math.min(1, this.t / 0.006) * Math.exp(-this.t / this.dur);
   535→    const sample = saturated * ampEnv * this.amp;
   536→    return [sample, false];
   537→  }
   538→}

   540→// ─── Voice: Acid (square → high-res Moog → distortion) ─────────────────────
   541→// PSY3 ANALOG MODELING: accent cap, thermal drift, power sag, slide,
   542→// component tolerance. These are what make the acid voice sound like a
   543→// real TB-303 rather than a sterile digital square wave.
   544→
   545→class AcidVoice {
   546→  constructor() {
   547→    this.active = false;
   548→    this.t = 0;
   549→    this.square = new BLSquare();
   550→    this.filter = new MoogLadder();
   551→    this.lfoPhase = 0; // bidirectional filter movement
   552→    // ── PSY3 ANALOG MODELING STATE ──
   553→    this.aAccCap = 0;       // accent cap accumulation ("the cry") — builds with accents, colors filter
   554→    this.aDrift = 0;        // current thermal drift (slow frequency modulation)
   555→    this.aDriftTarget = 0;  // drift target (random walk)
   556→    this.aPowerSag = 0;     // power sag (accent → momentary voltage drop → volume dip)
   557→    this.aActivity = 0;     // activity level (how busy the voice is — affects drift)
   558→    // Component tolerance: 4 filter stages have slightly different characteristics
   559→    // PSY3: aTol = [0.98, 1.02, 0.99, 1.01] — ±2% variation per stage
   560→    this.aTol = [0.98, 1.02, 0.99, 1.01];
   561→    // Slide state (constant-time portamento between notes)
   562→    this.prevFreq = 0;
   563→    this.slideFreq = 0;
   564→    this.slideActive = false;
   565→  }
   566→
   567→  trigger(time, freq, dur, amp, sr, param) {
   568→    this.active = true;
   569→    this.t = 0;
   570→    this.freq = freq;
   571→    this.dur = dur;
   572→    this.amp = amp;
   573→
   574→    // Accent detection — param >= 0.5 means accented note (PSY3 accent)
   575→    const isAccent = (param !== undefined && param >= 0.5);
   576→
   577→    // ── SLIDE: constant-time portamento from previous freq to new freq ──
   578→    // PSY3 uses 60ms slide time. Only slide if there's a previous note and
   579→    // the frequency difference is significant (avoids slide on first note
   580→    // or on same-note retriggers).
   581→    if (this.prevFreq > 0 && Math.abs(freq - this.prevFreq) > 1) {
   582→      this.slideFreq = this.prevFreq;
   583→      this.slideActive = true;
   584→    } else {
   585→      this.slideFreq = freq;
   586→      this.slideActive = false;
   587→    }
   588→    this.prevFreq = freq;
   589→
   590→    // ── ACCENT CAP: accumulates with each accent, colors the filter ──
   591→    // PSY3: aAccCap = min(1, aAccCap + 0.35 * isAccent)
   592→    // This builds up "the cry" — repeated accents make the filter brighter
   593→    // and more open, mimicking the way a real 303's envelope capacitor
   594→    // charges up with repeated accents.
   595→    this.aAccCap = Math.min(1, this.aAccCap + 0.35 * (isAccent ? 1 : 0));
   596→
   597→    // ── POWER SAG: accent causes momentary voltage drop ──
   598→    // PSY3: if (isAccent) aPowerSag = 0.15; aPowerSag *= 0.995
   599→    // The power supply sags under accent load → momentary volume dip.
   600→    // This is the "punch" of a real 303 — the note dips slightly then recovers.
   601→    if (isAccent) this.aPowerSag = 0.15;
   602→
   603→    // ── THERMAL DRIFT: random slow frequency drift target ──
   604→    // PSY3: if (Math.random() < 0.0004) aDriftT = (Math.random()-0.5)*2;
   605→    //       aDrift += (aDriftT - aDrift) * 0.0002
   606→    // We set a new drift target on each trigger (deterministic per-note).
   607→    // Drift is ±1% of frequency — inaudible as detuning but adds "life".
   608→    this.aDriftTarget = (Math.random() - 0.5) * 0.02;
   609→
   610→    // Activity increases with each note (affects drift intensity)
   611→    this.aActivity = Math.min(1, this.aActivity + 0.1);
   612→
   613→    this.square.reset();
   614→    this.filter.reset();
   615→    this.cutoffStart = 200 + 3000;
   616→    this.cutoffEnd = 100;
   617→    this.res = 0.95; // near self-oscillation for squelch
   618→    this.lfoPhase = 0;
   619→  }
   620→
   621→  render(currentTime, sr) {
   622→    if (!this.active) return [0, true];
   623→    const dt = 1 / sr;
   624→    this.t += dt;
   625→    if (this.t > this.dur + 0.05) { this.active = false; return [0, true]; }
   626→
   627→    // ── THERMAL DRIFT: slow random frequency modulation ──
   628→    // PSY3: aDrift += (aDriftT - aDrift) * 0.0002
   629→    // The oscillator frequency drifts slightly with temperature/activity.
   630→    // Inaudible as detuning but adds analog "life" — the note breathes.
   631→    this.aDrift += (this.aDriftTarget - this.aDrift) * 0.0002;
   632→    const driftMult = 1 + this.aDrift * (0.5 + this.aActivity * 0.5);
   633→
   634→    // ── SLIDE: constant-time portamento (60ms exponential glide) ──
   635→    let currentFreq = this.freq;
   636→    if (this.slideActive) {
   637→      const slideTime = 0.06; // 60ms constant-time slide
   638→      const slideProgress = Math.min(1, this.t / slideTime);
   639→      // Exponential glide (pitch slides exponentially, not linearly)
   640→      const ratio = this.freq / this.slideFreq;
   641→      currentFreq = this.slideFreq * Math.pow(ratio, slideProgress);
   642→      if (slideProgress >= 1) this.slideActive = false;
   643→    }
   644→    currentFreq *= driftMult;
   645→
   646→    const inc = currentFreq / sr;
   647→    const sq = this.square.process(inc);
   648→
   649→    // ── POWER SAG: accent causes momentary voltage drop (volume dip) ──
   650→    // PSY3: aPowerSag *= 0.995; osc *= (1 - aPowerSag)
   651→    // The note dips in volume then recovers — this is the analog "punch".
   652→    this.aPowerSag *= 0.995;
   653→    const sagGain = 1 - this.aPowerSag;
   654→    const sqSagged = sq * sagGain;
   655→
   656→    // ── ACCENT CAP: colors the filter cutoff (accent energy builds up) ──
   657→    // Higher accent cap → brighter, more open filter ("the cry")
   658→    // The cap decays slowly so repeated accents build brightness over time.
   659→    this.aAccCap *= 0.99999; // ~2s decay time constant
   660→    const accentBoost = this.aAccCap * 0.5; // up to +50% cutoff
   661→
   662→    // BIDIRECTIONAL filter movement — envelope + LFO combined
   663→    // Envelope: fast drop from high to low (classic acid)
   664→    const envCutoff = (this.cutoffStart - this.cutoffEnd) * Math.exp(-this.t / (this.dur * 0.4)) + this.cutoffEnd;
   665→    // LFO: slow sine that adds up-down movement on top of the envelope
   666→    // This creates the "wobble" that real 303 acid has
   667→    this.lfoPhase += 4.0 * dt; // 4Hz LFO
   668→    const lfo = Math.sin(2 * Math.PI * this.lfoPhase);
   669→    let cutoff = Math.max(80, envCutoff * (1 + lfo * 0.3) * (1 + accentBoost));
   670→    cutoff = Math.min(18000, cutoff);
   671→
   672→    // Component tolerance: 4 filter stages slightly detuned (PSY3 aTol)
   673→    const filtered = this.filter.process(sqSagged, cutoff, 0.95, 3.0, sr, this.aTol);
   674→    const distorted = fastTanh(filtered * 4); // heavy distortion
   675→
   676→    const ampEnv = Math.min(1, this.t / 0.003) * Math.exp(-this.t / this.dur);
   677→    const sample = distorted * ampEnv * this.amp;
   678→    return [sample, false];
   679→  }
   680→}

   682→// ─── Voice: FM (carrier + modulator + envelope, PSY3 acid FM) ──────────────
   683→// Two-operator FM: modulator (sine) → carrier (sine) frequency modulation.
   684→// Modulator index envelope (fast decay) gives the classic "FM pluck" attack
   685→// that PSY3's acid voice uses for metallic squelch. Drives through a Moog
   686→// ladder for warmth + a tanh saturator for grit.
   687→
   688→class FMVoice {
   689→  constructor() {
   690→    this.active = false;
   691→    this.t = 0;
   692→    this.carPhase = 0;
   693→    this.modPhase = 0;
   694→    this.filter = new MoogLadder();
   695→  }
   696→
   697→  trigger(time, freq, dur, amp, sr, params) {
   698→    this.active = true;
   699→    this.t = 0;
   700→    this.freq = freq;
   701→    this.dur = dur;
   702→    this.amp = amp;
   703→    this.carPhase = 0;
   704→    this.modPhase = 0;
   705→    this.ratio = (params && params.fmRatio) || 2.0;       // modulator:carrier ratio
   706→    this.depthStart = (params && params.fmDepth) || 6.0;   // modulation index (start)
   707→    this.depthEnd = (params && params.fmDepthEnd) || 0.5;  // modulation index (end)
   708→    this.cutoff = (params && params.cutoff) || 2200;
   709→    this.res = (params && params.resonance) || 0.4;
   710→    this.filter.reset();
   711→  }
   712→
   713→  render(currentTime, sr) {
   714→    if (!this.active) return [0, true];
   715→    const dt = 1 / sr;
   716→    this.t += dt;
   717→    if (this.t > this.dur + 0.05) { this.active = false; return [0, true]; }
   718→
   719→    // Modulator: sine at freq * ratio, with envelope on modulation index
   720→    this.modPhase += 2 * Math.PI * this.freq * this.ratio * dt;
   721→    // Exponential index decay (PSY3 "accent thermal" — fast attack, exp decay)
   722→    const idx = (this.depthStart - this.depthEnd) * Math.exp(-this.t / 0.05) + this.depthEnd;
   723→    const modulator = Math.sin(this.modPhase) * this.freq * idx;
   724→
   725→    // Carrier: sine at freq + modulator
   726→    this.carPhase += 2 * Math.PI * (this.freq + modulator) * dt;
   727→    const carrier = Math.sin(this.carPhase);
   728→
   729→    // Through Moog ladder for warmth (PSY3 always filters FM)
   730→    const filtered = this.filter.process(carrier, this.cutoff, this.res, 1.4, sr);
   731→    // Saturation for grit
   732→    const saturated = fastTanh(filtered * 1.8);
   733→
   734→    // Amp envelope: 3ms attack + exp decay over dur
   735→    const ampEnv = Math.min(1, this.t / 0.003) * Math.exp(-this.t / this.dur);
   736→    return [saturated * ampEnv * this.amp, false];
   737→  }
   738→}

   740→// ─── Voice: Pad (detuned saws → Moog → slow env) ───────────────────────────
   741→
   742→class PadVoice {
   743→  constructor() {
   744→    this.active = false;
   745→    this.t = 0;
   746→    this.saws = [new BLSaw(), new BLSaw(), new BLSaw()]; // 3 oscillators (was 2)
   747→    this.filter = new MoogLadder();
   748→    this.lfoPhase = 0;
   749→    this.filterSweepPhase = 0; // slow filter sweep
   750→  }
   751→
   752→  trigger(time, freq, dur, amp, sr, params) {
   753→    this.active = true;
   754→    this.t = 0;
   755→    this.dur = dur;
   756→    this.amp = amp;
   757→    this.freq = freq;
   758→    this.cutoffBase = params?.cutoff ?? 1200;
   759→    this.res = 0.08; // slightly higher resonance for filter movement
   760→    this.attack = params?.attack ?? 0.5;
   761→    this.detune = params?.detune ?? 7;
   762→    this.evolveRate = params?.evolveRate ?? 0.1;
   763→    this.lfoPhase = 0;
   764→    this.filterSweepPhase = 0;
   765→    for (const s of this.saws) { s.reset(); }
   766→    // 3-osc detuned: -detune, center, +detune (wider than 2-osc)
   767→    this.saws[0].setFreq(freq * Math.pow(2, -this.detune / 1200));
   768→    this.saws[1].setFreq(freq);
   769→    this.saws[2].setFreq(freq * Math.pow(2, this.detune / 1200));
   770→    this.filter.reset();
   771→  }
   772→
   773→  // Mono render (backward compat — delegates to renderStereo and sums to mono)
   774→  render(currentTime, sr) {
   775→    const [l, r] = this.renderStereo(currentTime, sr);
   776→    return [(l + r) * 0.5, false];
   777→  }
   778→
   779→  // STEREO render — PSY3 stereo spread: detuned oscs panned L/C/R
   780→  // PSY3 pad has: numOscs=2, detune=0.004, cutoff=900, attack=0.6, release=1.2
   781→  // We use 3 oscs panned L/C/R for wider stereo image.
   782→  // Filter is applied to the MID signal (M/S processing) — preserves stereo width.
   783→  renderStereo(currentTime, sr) {
   784→    if (!this.active) return [0, 0, true];
   785→    const dt = 1 / sr;
   786→    this.t += dt;
   787→    if (this.t > this.dur + 0.1) { this.active = false; return [0, 0, true]; }
   788→
   789→    // Evolve LFO modulates detune (via frequency)
   790→    this.lfoPhase += this.evolveRate * dt;
   791→    const lfo = Math.sin(2 * Math.PI * this.lfoPhase);
   792→    const detuneMod = 1 + 0.003 * lfo;
   793→    this.saws[0].setFreq(this.freq * Math.pow(2, -this.detune / 1200) * detuneMod);
   794→    this.saws[1].setFreq(this.freq * detuneMod);
   795→    this.saws[2].setFreq(this.freq * Math.pow(2, this.detune / 1200) * detuneMod);
   796→
   797→    // Render each saw with its own frequency
   798→    const s0 = this.saws[0].process(this.saws[0].freq / sr);
   799→    const s1 = this.saws[1].process(this.saws[1].freq / sr);
   800→    const s2 = this.saws[2].process(this.saws[2].freq / sr);
   801→
   802→    // STEREO SPREAD: pan detuned oscs L/C/R
   803→    // s0 (detuned -) → hard left, s1 (center) → both, s2 (detuned +) → hard right
   804→    let left = s0 * 0.7 + s1 * 0.5;
   805→    let right = s2 * 0.7 + s1 * 0.5;
   806→
   807→    // M/S processing: filter the mid, preserve the side (stereo width)
   808→    const mid = (left + right) * 0.5;
   809→    const side = (left - right) * 0.5;
   810→
   811→    // SLOW FILTER SWEEP — cutoff moves up and down over the duration
   812→    // This is what makes a pad "breathe" — without it, it's a static organ
   813→    this.filterSweepPhase += 0.15 * dt; // 0.15Hz — very slow
   814→    const sweep = 0.5 + 0.5 * Math.sin(2 * Math.PI * this.filterSweepPhase);
   815→    const cutoff = this.cutoffBase * (0.6 + sweep * 0.8); // 60% to 140% of base
   816→
   817→    const filteredMid = this.filter.process(mid, cutoff, this.res, 1.2, sr);
   818→
   819→    // Recombine: filtered mid + unfiltered side (preserves stereo width)
   820→    left = filteredMid + side;
   821→    right = filteredMid - side;
   822→
   823→    // Slow attack/release envelope
   824→    const attackEnv = Math.min(1, this.t / this.attack);
   825→    const releaseEnv = Math.min(1, (this.dur - this.t) / 0.4);
   826→    const ampEnv = Math.max(0, Math.min(1, Math.min(attackEnv, releaseEnv)));
   827→    return [left * ampEnv * this.amp, right * ampEnv * this.amp, false];
   828→  }
   829→}

   833→class HatVoice {
   834→  constructor() {
   835→    this.active = false;
   836→    this.t = 0;
   837→    this.noise = new PinkNoise();
   838→    this.prevNoise = 0;
   839→  }

   841→  trigger(time, open, amp, sr) {

   843→    this.t = 0;
   844→    this.open = open;

   846→    this.decay = open ? 0.22 : 0.03;
   847→    this.prevNoise = 0;
   848→    this.noise.reset();
   849→  }

   858→    const hp = n - this.prevNoise;

   866→// ─── Voice: Clap (multi-burst noise, PSY3 engine.py clap) ──────────────────
   867→
   868→class ClapVoice {
   869→  constructor() {
   870→    this.active = false;
   871→    this.t = 0;
   872→    this.noise = new PinkNoise();
   873→  }
   874→
   875→  trigger(time, amp, sr) {
   876→    this.active = true;
   877→    this.t = 0;
   878→    this.amp = amp;
   879→    this.noise.reset();
   880→    this.bursts = [0, 0.012, 0.024, 0.036];
   881→    this.decays = [0.02, 0.02, 0.02, 0.09];
   882→  }
   883→
   884→  render(currentTime, sr) {
   885→    if (!this.active) return [0, true];
   886→    this.t += 1 / sr;
   887→    if (this.t > 0.3) { this.active = false; return [0, true]; }
   888→
   889→    const n = this.noise.next();
   890→    let g = 0;
   891→    for (let k = 0; k < 4; k++) {
   892→      if (this.t >= this.bursts[k]) {
   893→        g += Math.exp(-(this.t - this.bursts[k]) / this.decays[k]);
   894→      }
   895→    }
   896→    const sample = n * g * 0.6 * this.amp / 0.4;
   897→    return [sample, false];
   898→  }
   899→}

   901→// ─── Voice: Perc (pitched sine with pitch envelope + saturation) ───────────
   902→// BEFORE: bare sine with fixed frequency and decay = telephone bell.
   903→// AFTER: sine with pitch envelope (descending) + saturation + Moog filter = tribal perc.
   904→
   905→class PercVoice {
   906→  constructor() {
   907→    this.active = false;
   908→    this.t = 0;
   909→    this.phase = 0;
   910→    this.filter = new MoogLadder();
   911→  }
   912→
   913→  trigger(time, freq, amp, sr) {
   914→    this.active = true;
   915→    this.t = 0;
   916→    this.freq = freq;
   917→    this.amp = amp;
   918→    this.phase = 0;
   919→    this.filter.reset();
   920→  }
   921→
   922→  render(currentTime, sr) {
   923→    if (!this.active) return [0, true];
   924→    this.t += 1 / sr;
   925→    if (this.t > 0.1) { this.active = false; return [0, true]; }
   926→
   927→    // Pitch envelope: starts 1.5x higher, drops to fundamental
   928→    const pitchEnv = 1.5 * Math.exp(-this.t / 0.01) + 0.5;
   929→    this.phase += 2 * Math.PI * this.freq * pitchEnv / sr;
   930→    const osc = Math.sin(this.phase);
   931→
   932→    // Filter for body — LP at 800Hz with slight resonance
   933→    const filtered = this.filter.process(osc, 800, 0.2, 1.5, sr);
   934→
   935→    // Saturation for warmth
   936→    const saturated = fastTanh(filtered * 1.8);
   937→
   938→    const env = Math.exp(-this.t / 0.05);
   939→    const sample = saturated * env * this.amp;
   940→    return [sample, false];
   941→  }
   942→}
   943→
   944→// ─── Voice: Shaker (filtered noise with proper HP + saturation) ────────────
   945→// BEFORE: differentiated noise (primitive HP). Thin and digital.
   946→// AFTER: noise through Moog HP + saturation = warm shaker with body.
   947→
   948→class ShakerVoice {
   949→  constructor() {
   950→    this.active = false;
   951→    this.t = 0;
   952→    this.noise = new PinkNoise();
   953→    this.prevNoise = 0;
   954→    this.filter = new MoogLadder(); // for HP shaping
   955→  }
   956→
   957→  trigger(time, amp, sr) {
   958→    this.active = true;
   959→    this.t = 0;
   960→    this.amp = amp;
   961→    this.noise.reset();
   962→    this.prevNoise = 0;
   963→    this.filter.reset();
   964→  }
   965→
   966→  render(currentTime, sr) {
   967→    if (!this.active) return [0, true];
   968→    this.t += 1 / sr;
   969→    if (this.t > 0.08) { this.active = false; return [0, true]; }
   970→
   971→    const n = this.noise.process();
   972→    // HP via differentiation (fast)
   973→    const hp = n - this.prevNoise;
   974→    this.prevNoise = n;
   975→    // Additional HP shaping through Moog (highpass approximation via lowpass inversion)
   976→    const shaped = this.filter.process(hp, 6000, 0.1, 1.0, sr);
   977→    // Saturation for warmth
   978→    const saturated = fastTanh(shaped * 2.5);
   979→    const env = Math.exp(-this.t / 0.03);
   980→    const sample = saturated * env * 2 * this.amp;
   981→    return [sample, false];
   982→  }
   983→}

   985→// ─── Voice: Texture (multi-layer psychedelic evolving bed) ──────────────────
   986→// BEFORE: FM sine or raw noise = siren or wind. Not psychedelic.
   987→// AFTER: 3 layers — detuned osc bed + filtered noise + slow filter morph.
   988→// Creates evolving atmospheric texture that sounds "psychedelic" not "generated".
   989→
   990→class TextureVoice {
   991→  constructor() {
   992→    this.active = false;
   993→    this.t = 0;
   994→    this.saw1 = new BLSaw();
   995→    this.saw2 = new BLSaw();
   996→    this.filter = new MoogLadder();
   997→    this.noise = new PinkNoise();
   998→    this.morphPhase = 0;
   999→    this.noiseFilter = new MoogLadder(); // separate filter for noise layer
  1000→  }
  1001→
  1002→  trigger(time, dur, amp, type, sr) {
  1003→    this.active = true;
  1004→    this.t = 0;
  1005→    this.dur = dur;
  1006→    this.amp = amp;
  1007→    this.type = type || 'fm';
  1008→    this.morphPhase = 0;
  1009→    this.saw1.reset();
  1010→    this.saw2.reset();
  1011→    this.filter.reset();
  1012→    this.noiseFilter.reset();
  1013→    this.noise.reset();
  1014→    // Detuned oscillators — slow evolving bed
  1015→    const baseFreq = 110 + Math.random() * 220;
  1016→    this.saw1.setFreq(baseFreq);
  1017→    this.saw2.setFreq(baseFreq * 1.01); // very slight detune
  1018→    this.baseFreq = baseFreq;
  1019→  }
  1020→
  1021→  render(currentTime, sr) {
  1022→    if (!this.active) return [0, true];
  1023→    this.t += 1 / sr;
  1024→    if (this.t > this.dur + 0.1) { this.active = false; return [0, true]; }
  1025→
  1026→    const dt = 1 / sr;
  1027→    const env = Math.min(1, this.t / 0.5) * Math.min(1, (this.dur - this.t) / 0.5);
  1028→    if (env <= 0) return [0, false];
  1029→
  1030→    // Layer 1: Detuned saw bed — provides harmonic content
  1031→    const inc = this.baseFreq / sr;
  1032→    let oscBed = (this.saw1.process(inc) + this.saw2.process(inc)) * 0.3;
  1033→
  1034→    // Layer 2: Filtered noise — provides "air" and texture
  1035→    const noiseSamp = this.noise.process();
  1036→    const noiseFiltered = this.noiseFilter.process(noiseSamp, 2000, 0.3, 1.0, sr) * 0.4;
  1037→
  1038→    // Layer 3: Slow filter morph — cutoff moves up and down
  1039→    this.morphPhase += 0.3 * dt; // 0.3Hz morph
  1040→    const morph = 0.5 + 0.5 * Math.sin(2 * Math.PI * this.morphPhase);
  1041→    const morphCutoff = 300 + morph * 2000; // 300Hz to 2300Hz
  1042→
  1043→    // Mix layers and apply morph filter
  1044→    let mix = oscBed + noiseFiltered;
  1045→    mix = this.filter.process(mix, morphCutoff, 0.15, 1.2, sr);
  1046→
  1047→    // Saturation for warmth
  1048→    mix = fastTanh(mix * 1.3);
  1049→
  1050→    return [mix * env * this.amp, false];
  1051→  }
  1052→}

  1054→// ─── Voice: FX (riser, impact, sweep, zap, blip, downlifter) ──────────────
  1055→// BEFORE: Riser = noise getting louder. Impact = sine going down. Primitive.
  1056→// AFTER: Riser = noise + filter sweep opening up. Impact = sub boom + noise burst.
  1057→//        Sweep = filtered noise with stereo movement. Each FX has more body.
  1058→
  1059→class FXVoice {
  1060→  constructor() {
  1061→    this.active = false;
  1062→    this.t = 0;
  1063→    this.noise = new PinkNoise();
  1064→    this.phase = 0;
  1065→    this.filter = new MoogLadder(); // filter for riser/sweep
  1066→  }
  1067→
  1068→  trigger(type, time, dur, amp, sr) {
  1069→    this.active = true;
  1070→    this.type = type;
  1071→    this.t = 0;
  1072→    this.dur = dur || 0.3;
  1073→    this.amp = amp || 0.2;
  1074→    this.phase = 0;
  1075→    this.noise.reset();
  1076→    this.filter.reset();
  1077→  }
  1078→
  1079→  render(currentTime, sr) {
  1080→    if (!this.active) return [0, true];
  1081→    const dt = 1 / sr;
  1082→    this.t += dt;
  1083→    if (this.t > this.dur + 0.2) { this.active = false; return [0, true]; }
  1084→
  1085→    let sample = 0;
  1086→    const t = this.t;
  1087→    switch (this.type) {
  1088→      case V_RISER: {
  1089→        // Riser = noise through filter that opens up + amplitude rise
  1090→        // BEFORE: just noise * env. No filter, no character.
  1091→        const n = this.noise.process();
  1092→        // Filter opens from 200Hz to 8000Hz over the duration
  1093→        const cutoff = 200 + (t / this.dur) * 7800;
  1094→        const filtered = this.filter.process(n, cutoff, 0.2, 1.5, sr);
  1095→        // Amplitude rises exponentially (not linear)
  1096→        const env = Math.pow(t / this.dur, 2) * 0.35;
  1097→        sample = fastTanh(filtered * env * 3); // saturate for punch
  1098→        break;
  1099→      }
  1100→      case V_IMPACT: {
  1101→        // Impact = sub sine boom + noise burst (two layers)
  1102→        // BEFORE: just sine going down. No body, no texture.
  1103→        // Sub boom: sine from 120Hz to 35Hz with exp decay
  1104→        const f = 120 * Math.exp(-t / 0.15) + 35;
  1105→        this.phase += 2 * Math.PI * f * dt;
  1106→        const subEnv = Math.exp(-t / 0.2);
  1107→        const sub = Math.sin(this.phase) * subEnv * 0.7;
  1108→        // Noise burst: short percussive crack
  1109→        const n = this.noise.process();
  1110→        const noiseEnv = Math.exp(-t / 0.02); // 20ms crack
  1111→        const crack = n * noiseEnv * 0.3;
  1112→        sample = sub + crack;
  1113→        sample = fastTanh(sample * 1.5); // saturate
  1114→        break;
  1115→      }
  1116→      case V_SWEEP: {
  1117→        // Sweep = filtered noise with filter moving + amplitude curve
  1118→        // BEFORE: noise * sin envelope. No filter movement.
  1119→        const n = this.noise.process();
  1120→        // Filter sweeps from low to high and back
  1121→        const sweepPos = t / this.dur;
  1122→        const cutoff = 200 + Math.sin(Math.PI * sweepPos) * 4000 + 2000;
  1123→        const filtered = this.filter.process(n, cutoff, 0.3, 1.3, sr);
  1124→        const env = Math.sin(Math.PI * sweepPos) * 0.2;
  1125→        sample = filtered * env;
  1126→        break;
  1127→      }
  1128→      case V_ZAP: {
  1129→        // FM zap — carrier + modulator with exponential index decay
  1130→        const car = 880, mod = 1760;
  1131→        const idx = 3 * Math.exp(-t / 0.03);
  1132→        this.phase += 2 * Math.PI * (car + idx * Math.sin(2 * Math.PI * mod * t)) * dt;
  1133→        const env = Math.exp(-t / 0.04);
  1134→        sample = Math.sin(this.phase) * env;
  1135→        sample = fastTanh(sample * 2); // saturate for grit
  1136→        break;
  1137→      }
  1138→      case V_BLIP: {
  1139→        // Pure sine blip with pitch envelope (descending)
  1140→        const f = 1200 * Math.exp(-t / 0.01) + 400;
  1141→        this.phase += 2 * Math.PI * f * dt;
  1142→        const env = Math.exp(-t / 0.02);
  1143→        sample = Math.sin(this.phase) * env;
  1144→        break;
  1145→      }
  1146→      case V_DOWNLIFTER: {
  1147→        // Downlifter = saw wave with descending pitch + filter closing
  1148→        const f = 800 * Math.exp(-t / 0.15) + 100;
  1149→        this.phase += 2 * Math.PI * f * dt;
  1150→        const saw = 2 * (this.phase / (2 * Math.PI) % 1) - 1; // naive saw
  1151→        const cutoff = 3000 * Math.exp(-t / 0.2) + 200;
  1152→        const filtered = this.filter.process(saw, cutoff, 0.1, 1.0, sr);
  1153→        const env = Math.exp(-t / 0.2);
  1154→        sample = filtered * env * 0.4;
  1155→        break;
  1156→      }
  1157→    }
  1158→    return [sample * this.amp, false];
  1159→  }
  1160→}

  1162→// ─── Sample Voice (plays preloaded AudioBuffer data) ──────────────────────
  1163→// Plays a sample with linear interpolation, pitch shift, and gain.
  1164→// Used for kick/hat/clap — the REAL PSY3 samples give professional sound quality
  1165→// that pure synth DSP cannot match.
  1166→
  1167→class SampleVoice {
  1168→  constructor() {
  1169→    this.active = false;
  1170→    this.t = 0;
  1171→    this.sampleData = null;     // Float32Array
  1172→    this.sampleRate = 44100;
  1173→    this.playbackRate = 1.0;    // pitch shift
  1174→    this.amp = 1.0;
  1175→    this.gainEnv = 1.0;
  1176→    this.decay = 0.3;
  1177→    this.position = 0;          // fractional sample position
  1178→    this.pan = 0;               // -1..1
  1179→  }
  1180→
  1181→  trigger(sampleData, sampleRate, playbackRate, amp, decay, pan) {
  1182→    this.active = true;
  1183→    this.t = 0;
  1184→    this.sampleData = sampleData;
  1185→    this.sampleRate = sampleRate;
  1186→    this.playbackRate = playbackRate || 1.0;
  1187→    this.amp = amp;
  1188→    this.decay = decay || 0.3;
  1189→    this.position = 0;
  1190→    this.pan = pan || 0;
  1191→  }
  1192→
  1193→  // Returns [leftSample, rightSample, done]
  1194→  renderStereo(currentTime, sr) {
  1195→    if (!this.active || !this.sampleData) return [0, 0, true];
  1196→    this.t += 1 / sr;
  1197→    const env = Math.exp(-this.t / this.decay);
  1198→    if (env < 0.001 || this.position >= this.sampleData.length) {
  1199→      this.active = false;
  1200→      return [0, 0, true];
  1201→    }
  1202→
  1203→    // Linear interpolation playback
  1204→    const idx = Math.floor(this.position);
  1205→    const frac = this.position - idx;
  1206→    const s1 = this.sampleData[idx] || 0;
  1207→    const s2 = this.sampleData[idx + 1] || 0;
  1208→    let sample = (s1 + (s2 - s1) * frac) * env * this.amp;
  1209→
  1210→    // SATURATION: Add warmth and punch to samples (especially kick)
  1211→    // Commercial kicks/snares always have saturation. Without it, samples
  1212→    // sound flat and lifeless. This tanh adds harmonics that make the
  1213→    // kick "punch through" the mix.
  1214→    sample = fastTanh(sample * 1.4);  // moderate drive — warm, not distorted
  1215→
  1216→    // Advance position based on playback rate and sample rate ratio
  1217→    this.position += this.playbackRate * (this.sampleRate / sr);
  1218→
  1219→    // Stereo: apply pan (equal power)
  1220→    const pan = Math.max(-1, Math.min(1, this.pan));
  1221→    const leftGain = pan <= 0 ? 1 : 1 - pan;
  1222→    const rightGain = pan >= 0 ? 1 : 1 + pan;
  1223→
  1224→    return [sample * leftGain, sample * rightGain, false];
  1225→  }
  1226→}

  1228→// ─── Algorithmic Reverb (Schroeder-style: 4 comb + 2 allpass) ──────────────
  1229→// Creates space and depth. A dry psytrance mix sounds flat/amateur.
  1230→// Reverb is a SEND — voices send a portion of their signal here, and the
  1231→// reverb output feeds back to the master. This is how professional mixes work.
  1232→
  1233→class SchroederReverb {
  1234→  constructor() {
  1235→    // 4 parallel comb filters (different delays for density)
  1236→    this.combDelays = [1687, 1601, 2053, 2251]; // samples at 44100 (prime)
  1237→    this.combBuffers = [];
  1238→    this.combIdx = [];
  1239→    this.combFeedback = 0.84;
  1240→    this.combDamping = 0.2;
  1241→    this.combLP = []; // one-pole LP per comb for high-freq damping
  1242→    for (let i = 0; i < 4; i++) {
  1243→      this.combBuffers.push(new Float32Array(this.combDelays[i]));
  1244→      this.combIdx.push(0);
  1245→      this.combLP.push(0);
  1246→    }
  1247→    // 2 series allpass filters (diffusion)
  1248→    this.allpassDelays = [347, 113]; // samples
  1249→    this.allpassBuffers = [];
  1250→    this.allpassIdx = [];
  1251→    this.allpassFeedback = 0.7;
  1252→    for (let i = 0; i < 2; i++) {
  1253→      this.allpassBuffers.push(new Float32Array(this.allpassDelays[i]));
  1254→      this.allpassIdx.push(0);
  1255→    }
  1256→    this.wet = 0.45;  // INCREASED from 0.3 — more audible reverb
  1257→    this.inputGain = 0.15; // send level
  1258→  }
  1259→
  1260→  setWet(wet) { this.wet = wet; }
  1261→  setInputGain(g) { this.inputGain = g; }
  1262→
  1263→  // Process a mono input, return stereo [left, right] reverb output
  1264→  process(input, sr) {
  1265→    // Scale input by send level
  1266→    const inSample = input * this.inputGain;
  1267→
  1268→    // ── Comb filters (parallel) ──
  1269→    let combSum = 0;
  1270→    for (let i = 0; i < 4; i++) {
  1271→      const buf = this.combBuffers[i];
  1272→      const idx = this.combIdx[i];
  1273→      const delayed = buf[idx];
  1274→      // One-pole lowpass for damping (high frequencies decay faster)
  1275→      this.combLP[i] = delayed + this.combDamping * (this.combLP[i] - delayed);
  1276→      const out = inSample + this.combLP[i] * this.combFeedback;
  1277→      buf[idx] = out;
  1278→      this.combIdx[i] = (idx + 1) % this.combDelays[i];
  1279→      combSum += out;
  1280→    }
  1281→    combSum *= 0.25; // normalize
  1282→
  1283→    // ── Allpass filters (series) for diffusion ──
  1284→    let ap = combSum;
  1285→    for (let i = 0; i < 2; i++) {
  1286→      const buf = this.allpassBuffers[i];
  1287→      const idx = this.allpassIdx[i];
  1288→      const delayed = buf[idx];
  1289→      const out = -ap * this.allpassFeedback + delayed;
  1290→      buf[idx] = ap + delayed * this.allpassFeedback;
  1291→      this.allpassIdx[i] = (idx + 1) % this.allpassDelays[i];
  1292→      ap = out;
  1293→    }
  1294→
  1295→    // Stereo: slight delay between L and R for width
  1296→    // (re-use allpass output, offset by a few samples for stereo effect)
  1297→    const left = ap * this.wet;
  1298→    const right = combSum * this.wet * 0.9; // slightly different for width
  1299→    return [left, right];
  1300→  }
  1301→
  1302→  reset() {
  1303→    for (const buf of this.combBuffers) buf.fill(0);
  1304→    for (const buf of this.allpassBuffers) buf.fill(0);
  1305→    this.combLP.fill(0);
  1306→  }
  1307→}

  1309→// ─── Tempo-Synced Stereo Delay (ping-pong) ────────────────────────────────
  1310→// Creates psychedelic movement. Left and right channels have different
  1311→// delay times (e.g., 3/16 and 3/8) for a wide, evolving echo.
  1312→
  1313→class StereoDelay {
  1314→  constructor() {
  1315→    // REDUCED from 2s to 0.5s — saves 1.3MB memory
  1316→    // 0.5s is plenty for psytrance delay (3/8 at 140bpm = 0.32s)
  1317→    this.bufferSize = 44100 / 2; // 0.5 seconds max (was 2 seconds)
  1318→    this.leftBuf = new Float32Array(this.bufferSize);
  1319→    this.rightBuf = new Float32Array(this.bufferSize);
  1320→    this.leftIdx = 0;
  1321→    this.rightIdx = 0;
  1322→    this.leftDelay = 0.375;  // seconds (3/8 at 120bpm)
  1323→    this.rightDelay = 0.281; // seconds (slightly different for ping-pong)
  1324→    this.feedback = 0.35;
  1325→    this.wet = 0.35;  // INCREASED from 0.25 — more audible delay
  1326→    this.inputGain = 0.2;
  1327→    this.sr = 44100;
  1328→    // LP filter on feedback for darker echoes
  1329→    this.fbLP = [0, 0];
  1330→  }
  1331→
  1332→  setDelayTimes(leftMs, rightMs) {
  1333→    this.leftDelay = leftMs / 1000;
  1334→    this.rightDelay = rightMs / 1000;
  1335→  }
  1336→
  1337→  setFeedback(fb) { this.feedback = fb; }
  1338→  setWet(wet) { this.wet = wet; }
  1339→  setInputGain(g) { this.inputGain = g; }
  1340→
  1341→  // Process stereo input [left, right], return stereo [left, right] delay output
  1342→  process(leftIn, rightIn, sr) {
  1343→    this.sr = sr;
  1344→    const leftDelaySamples = Math.floor(this.leftDelay * sr);
  1345→    const rightDelaySamples = Math.floor(this.rightDelay * sr);
  1346→
  1347→    // Read delayed samples
  1348→    const leftReadIdx = (this.leftIdx - leftDelaySamples + this.bufferSize) % this.bufferSize;
  1349→    const rightReadIdx = (this.rightIdx - rightDelaySamples + this.bufferSize) % this.bufferSize;
  1350→    const leftDelayed = this.leftBuf[leftReadIdx];
  1351→    const rightDelayed = this.rightBuf[rightReadIdx];
  1352→
  1353→    // Feedback with LP filtering (darker echoes)
  1354→    const fbCutoff = 0.3;
  1355→    this.fbLP[0] = this.fbLP[0] + fbCutoff * (leftDelayed - this.fbLP[0]);
  1356→    this.fbLP[1] = this.fbLP[1] + fbCutoff * (rightDelayed - this.fbLP[1]);
  1357→
  1358→    // Ping-pong: left feedback goes to right, right to left
  1359→    const leftWrite = leftIn * this.inputGain + this.fbLP[1] * this.feedback;
  1360→    const rightWrite = rightIn * this.inputGain + this.fbLP[0] * this.feedback;
  1361→
  1362→    this.leftBuf[this.leftIdx] = leftWrite;
  1363→    this.rightBuf[this.rightIdx] = rightWrite;
  1364→    this.leftIdx = (this.leftIdx + 1) % this.bufferSize;
  1365→    this.rightIdx = (this.rightIdx + 1) % this.bufferSize;
  1366→
  1367→    return [leftDelayed * this.wet, rightDelayed * this.wet];
  1368→  }
  1369→
  1370→  reset() {
  1371→    this.leftBuf.fill(0);
  1372→    this.rightBuf.fill(0);
  1373→    this.fbLP.fill(0);
  1374→  }
  1375→}

  1377→// ─── Bus Processor (compression + saturation + EQ per bus) ────────────────
  1378→// Each bus (drum/bass/music/atmos/fx) gets its own processing.
  1379→// This is what makes the mix sound "produced" — without bus processing,
  1380→// it sounds like isolated sounds, not a cohesive track.
  1381→
  1382→class BusProcessor {
  1383→  constructor(config) {
  1384→    this.config = config;
  1385→    // Compressor state
  1386→    this.compEnv = 0;
  1387→    // HP filter state (clean low end)
  1388→    this.hpState = 0;
  1389→    // Saturation drive
  1390→    this.drive = config.drive || 1.0;
  1391→    // Output gain
  1392→    this.gain = config.gain || 1.0;
  1393→  }
  1394→
  1395→  process(sample, sr) {
  1396→    const dt = 1 / sr;
  1397→
  1398→    // 1. HP FILTER: Remove subsonic mud (configurable per bus)
  1399→    if (this.config.hpFreq && this.config.hpFreq > 0) {
  1400→      const hpA = (1 / sr) * 2 * Math.PI * this.config.hpFreq;
  1401→      this.hpState += hpA * (sample - this.hpState) / (1 + hpA);
  1402→      sample = sample - this.hpState;
  1403→    }
  1404→
  1405→    // 2. COMPRESSION: Simple envelope-follower compressor
  1406→    //    Drum bus: fast attack/release, moderate ratio (punchy)
  1407→    //    Bass bus: medium attack/release, low ratio (controlled)
  1408→    //    Music bus: slow attack/release, low ratio (glue)
  1409→    if (this.config.compThr) {
  1410→      const abs = Math.abs(sample);
  1411→      const att = this.config.compAtt || 0.003;
  1412→      const rel = this.config.compRel || 0.1;
  1413→      if (abs > this.compEnv) {
  1414→        this.compEnv += (abs - this.compEnv) * (dt / att);
  1415→      } else {
  1416→        this.compEnv += (abs - this.compEnv) * (dt / rel);
  1417→      }
  1418→      if (this.compEnv > this.config.compThr) {
  1419→        const over = this.compEnv - this.config.compThr;
  1420→        const ratio = this.config.compRatio || 2;
  1421→        const reduction = over * (1 - 1 / ratio);
  1422→        const compGain = (this.compEnv - reduction) / this.compEnv;
  1423→        sample *= compGain;
  1424→      }
  1425→      // Makeup gain
  1426→      sample *= this.config.compMakeup || 1.2;
  1427→    }
  1428→
  1429→    // 3. SATURATION: Add warmth and harmonics
  1430→    if (this.drive > 1.0) {
  1431→      sample = fastTanh(sample * this.drive);
  1432→    }
  1433→
  1434→    return sample * this.gain;
  1435→  }
  1436→}
  1437→
  1438→// ─── Biquad filter (RBJ cookbook, transposed direct form II) ───────────────
  1439→// Used for multiband crossover filters in the master chain.
  1440→
  1441→class Biquad {
  1442→  constructor() {
  1443→    this.z1 = 0; this.z2 = 0;
  1444→    this.b0 = 1; this.b1 = 0; this.b2 = 0;
  1445→    this.a1 = 0; this.a2 = 0;
  1446→  }
  1447→  setLowpass(fc, sr, Q) {
  1448→    const w0 = 2 * Math.PI * fc / sr;
  1449→    const cw = Math.cos(w0), sw = Math.sin(w0);
  1450→    const alpha = sw / (2 * Q);
  1451→    const a0 = 1 + alpha;
  1452→    this.b0 = ((1 - cw) / 2) / a0;
  1453→    this.b1 = (1 - cw) / a0;
  1454→    this.b2 = ((1 - cw) / 2) / a0;
  1455→    this.a1 = (-2 * cw) / a0;
  1456→    this.a2 = (1 - alpha) / a0;
  1457→  }
  1458→  setHighpass(fc, sr, Q) {
  1459→    const w0 = 2 * Math.PI * fc / sr;
  1460→    const cw = Math.cos(w0), sw = Math.sin(w0);
  1461→    const alpha = sw / (2 * Q);
  1462→    const a0 = 1 + alpha;
  1463→    this.b0 = ((1 + cw) / 2) / a0;
  1464→    this.b1 = -(1 + cw) / a0;
  1465→    this.b2 = ((1 + cw) / 2) / a0;
  1466→    this.a1 = (-2 * cw) / a0;
  1467→    this.a2 = (1 - alpha) / a0;
  1468→  }
  1469→  process(x) {
  1470→    // Transposed direct form II
  1471→    const y = this.b0 * x + this.z1;
  1472→    this.z1 = this.b1 * x - this.a1 * y + this.z2;
  1473→    this.z2 = this.b2 * x - this.a2 * y;
  1474→    return y;
  1475→  }
  1476→  reset() { this.z1 = 0; this.z2 = 0; }
  1477→}
  1478→
  1479→// ─── Multiband Compressor (3-band: low <180Hz, mid 180-4000Hz, high >4000Hz) ──
  1480→// PSY3 style_master.py master_pro() step 2: multiband compression.
  1481→// Uses Linkwitz-Riley 2nd-order crossovers (Q=0.5) for flat summing.
  1482→// Each band has independent compressor with per-band attack/release/ratio.
  1483→
  1484→class MultibandComp {
  1485→  constructor(sr) {
  1486→    // LR2 crossovers at 180Hz and 4000Hz
  1487→    // 4 biquads: LP1(180), HP1(180), LP2(4000), HP2(4000)
  1488→    this.lp1 = new Biquad(); this.lp1.setLowpass(180, sr, 0.5);
  1489→    this.hp1 = new Biquad(); this.hp1.setHighpass(180, sr, 0.5);
  1490→    this.lp2 = new Biquad(); this.lp2.setLowpass(4000, sr, 0.5);
  1491→    this.hp2 = new Biquad(); this.hp2.setHighpass(4000, sr, 0.5);
  1492→    // Per-band compressor envelope state
  1493→    this.envLow = 0; this.envMid = 0; this.envHigh = 0;
  1494→    // Per-band settings (PSY3-style: low=more ratio, high=faster)
  1495→    this.lowThr = 0.5;  this.lowRatio = 3;   this.lowAtt = 0.005; this.lowRel = 0.15;  this.lowMakeup = 1.3;
  1496→    this.midThr = 0.5;  this.midRatio = 2;   this.midAtt = 0.004; this.midRel = 0.12;  this.midMakeup = 1.2;
  1497→    this.highThr = 0.45; this.highRatio = 2.5; this.highAtt = 0.002; this.highRel = 0.08; this.highMakeup = 1.2;
  1498→  }
  1499→
  1500→  process(sample, sr) {
  1501→    const dt = 1 / sr;
  1502→
  1503→    // ── Split into 3 bands (LR2 crossover = flat sum) ──
  1504→    const low = this.lp1.process(sample);
  1505→    const midHigh = this.hp1.process(sample);
  1506→    const mid = this.lp2.process(midHigh);
  1507→    const high = this.hp2.process(midHigh);
  1508→
  1509→    // ── Compress LOW band (controls bass energy) ──
  1510→    const absLow = Math.abs(low);
  1511→    if (absLow > this.envLow) this.envLow += (absLow - this.envLow) * (dt / this.lowAtt);
  1512→    else this.envLow += (absLow - this.envLow) * (dt / this.lowRel);
  1513→    let gainLow = 1;
  1514→    if (this.envLow > this.lowThr) {
  1515→      const over = this.envLow - this.lowThr;
  1516→      gainLow = (this.envLow - over * (1 - 1 / this.lowRatio)) / this.envLow;
  1517→    }
  1518→    const cLow = low * gainLow * this.lowMakeup;
  1519→
  1520→    // ── Compress MID band (controls vocal/instrument presence) ──
  1521→    const absMid = Math.abs(mid);
  1522→    if (absMid > this.envMid) this.envMid += (absMid - this.envMid) * (dt / this.midAtt);
  1523→    else this.envMid += (absMid - this.envMid) * (dt / this.midRel);
  1524→    let gainMid = 1;
  1525→    if (this.envMid > this.midThr) {
  1526→      const over = this.envMid - this.midThr;
  1527→      gainMid = (this.envMid - over * (1 - 1 / this.midRatio)) / this.envMid;
  1528→    }
  1529→    const cMid = mid * gainMid * this.midMakeup;
  1530→
  1531→    // ── Compress HIGH band (controls harshness/air) ──
  1532→    const absHigh = Math.abs(high);
  1533→    if (absHigh > this.envHigh) this.envHigh += (absHigh - this.envHigh) * (dt / this.highAtt);
  1534→    else this.envHigh += (absHigh - this.envHigh) * (dt / this.highRel);
  1535→    let gainHigh = 1;
  1536→    if (this.envHigh > this.highThr) {
  1537→      const over = this.envHigh - this.highThr;
  1538→      gainHigh = (this.envHigh - over * (1 - 1 / this.highRatio)) / this.envHigh;
  1539→    }
  1540→    const cHigh = high * gainHigh * this.highMakeup;
  1541→
  1542→    // Sum bands (LR2 = flat, so sum ≈ original + compression artifacts)
  1543→    return cLow + cMid + cHigh;
  1544→  }
  1545→}
  1546→
  1547→// ─── Stereo Widener (PSY3 to_stereo: Haas delay + decorrelated HP side) ─────
  1548→// PSY3 style_master.py to_stereo():
  1549→//   d = int(0.012 * SR)  // 12ms Haas
  1550→//   side = roll(x, d); side[:d] = 0
  1551→//   side = side - roll(side, 1)  // decorrelated HP side
  1552→//   return [x + side*width, x - side*width]
  1553→// Creates stereo width from mono signal. Here we enhance existing stereo:
  1554→// extract mid, delay+HP it, add/subtract as side channel.
  1555→
  1556→class StereoWidener {
  1557→  constructor() {
  1558→    // 12ms Haas delay buffer (generous size for up to 96kHz: 0.012 * 96000 = 1152)
  1559→    this.delayBuf = new Float32Array(2048);
  1560→    this.delayIdx = 0;
  1561→    this.delaySamples = Math.max(1, Math.floor(0.012 * sampleRate));
  1562→    this.prevDelayed = 0;
  1563→    this.width = 0.3; // PSY3 default width
  1564→  }
  1565→
  1566→  setWidth(w) { this.width = Math.max(0, Math.min(0.5, w)); }
  1567→
  1568→  // Takes stereo [left, right], returns widened stereo [left, right]
  1569→  process(left, right, sr) {
  1570→    // Mid signal
  1571→    const mid = (left + right) * 0.5;
  1572→
  1573→    // Haas delay on mid (12ms)
  1574→    const delayed = this.delayBuf[this.delayIdx];
  1575→    this.delayBuf[this.delayIdx] = mid;
  1576→    this.delayIdx = (this.delayIdx + 1) % this.delaySamples;
  1577→
  1578→    // HP via differentiation (decorrelated side — PSY3: side = side - roll(side,1))
  1579→    const side = delayed - this.prevDelayed;
  1580→    this.prevDelayed = delayed;
  1581→
  1582→    // Add width: L += side*width, R -= side*width
  1583→    // This adds a delayed+HP'd version of the mid to the side channel,
  1584→    // creating a sense of space without destroying the original image.
  1585→    return [left + side * this.width, right - side * this.width];
  1586→  }
  1587→
  1588→  reset() {
  1589→    this.delayBuf.fill(0);
  1590→    this.prevDelayed = 0;
  1591→    this.delayIdx = 0;
  1592→  }
  1593→}
  1594→
  1595→// ─── Master chain (multiband comp + glue + saturation + LUFS + true-peak) ───
  1596→// PSY3 style_master.py master_pro() ports:
  1597→//   1. Multiband compression (3-band) — NEW
  1598→//   2. Glue compression (thr=0.6, ratio=2, makeup=1.3) — PSY3 params
  1599→//   3. Saturation (drive=1.15, mix=0.15) — PSY3 params
  1600→//   4. LUFS targeting (-9 LUFS) — NEW
  1601→//   5. True-peak limiting (2x oversample, ceiling 0.89) — NEW
  1602→//   6. Final tanh (soft clip safety)
  1603→
  1604→class MasterChain {
  1605→  constructor() {
  1606→    this.gain = 1.0;
  1607→    this.ceiling = 0.89;     // PSY3 true-peak ceiling
  1608→
  1609→    // Multiband compressor (3-band: low <180Hz, mid 180-4000Hz, high >4000Hz)
  1610→    this.mb = new MultibandComp(sampleRate);
  1611→
  1612→    // Glue compression (PSY3: thr=0.6, ratio=2, makeup=1.3)
  1613→    this.glueEnv = 0;
  1614→    this.glueThr = 0.60;
  1615→    this.glueRatio = 2.0;      // PSY3 ratio
  1616→    this.glueAttack = 0.004;
  1617→    this.glueRelease = 0.12;
  1618→    this.glueMakeup = 1.3;     // PSY3 makeup
  1619→
  1620→    // Saturation (PSY3: drive=1.15, mix=0.15)
  1621→    this.satDrive = 1.15;
  1622→    this.satMix = 0.15;
  1623→
  1624→    // LUFS targeting (simplified K-weighted loudness → makeup gain)
  1625→    this.lufsMs = 0;           // running mean square
  1626→    this.lufsGain = 1.0;       // current applied gain
  1627→    this.lufsTargetGain = 1.0; // computed target gain
  1628→    this.lufsTargetLufs = -9;  // target loudness (-8 to -10 LUFS)
  1629→    this.lufsCounter = 0;      // update counter (every 32 samples)
  1630→
  1631→    // True-peak limiter (2x oversample, 1-sample lookahead)
  1632→    this.tpPrevInput = 0;      // previous input sample (for inter-sample peak)
  1633→    this.tpGainEnv = 1;        // limiter gain (smoothed)
  1634→    this.tpAttack = 0.0001;    // very fast attack (catches peaks)
  1635→    this.tpRelease = 0.06;     // moderate release
  1636→
  1637→    this.sr = sampleRate;
  1638→  }
  1639→
  1640→  process(sample, sr) {
  1641→    const dt = 1 / sr;
  1642→
  1643→    // 1. MULTIBAND COMPRESSION (3-band: low/mid/high)
  1644→    //    Compresses each band independently → tighter low end, controlled
  1645→    //    mids, smoothed highs. This is what commercial masters have.
  1646→    sample = this.mb.process(sample, sr);
  1647→
  1648→    // 2. GLUE COMPRESSION (PSY3: thr=0.6, ratio=2, makeup=1.3)
  1649→    //    "Glues" the multiband output into a cohesive track.
  1650→    const abs = Math.abs(sample);
  1651→    if (abs > this.glueEnv) {
  1652→      this.glueEnv += (abs - this.glueEnv) * (dt / this.glueAttack);
  1653→    } else {
  1654→      this.glueEnv += (abs - this.glueEnv) * (dt / this.glueRelease);
  1655→    }
  1656→    let glueGain = 1;
  1657→    if (this.glueEnv > this.glueThr) {
  1658→      const over = this.glueEnv - this.glueThr;
  1659→      glueGain = (this.glueEnv - over * (1 - 1 / this.glueRatio)) / this.glueEnv;
  1660→    }
  1661→    sample *= glueGain * this.glueMakeup;
  1662→
  1663→    // 3. SATURATION (PSY3: drive=1.15, mix=0.15)
  1664→    //    Mix of dry + tanh-saturated (adds harmonic richness + warmth)
  1665→    const saturated = fastTanh(sample * this.satDrive);
  1666→    sample = saturated * this.satMix + sample * (1 - this.satMix);
  1667→
  1668→    // 4. LUFS TARGETING (simplified K-weighted loudness → makeup gain)
  1669→    //    Measures running mean square, converts to LUFS approximation,
  1670→    //    adjusts gain to hit target (-9 LUFS). Slow time constant → no pumping.
  1671→    this.lufsMs = this.lufsMs * 0.99999 + sample * sample * 0.00001;
  1672→    // Update target gain every 32 samples (saves Math.log10/pow CPU)
  1673→    if ((this.lufsCounter++ & 31) === 0) {
  1674→      const lufs = this.lufsMs > 1e-10 ? -0.691 + 10 * Math.log10(this.lufsMs) : -70;
  1675→      const gainDb = this.lufsTargetLufs - lufs;
  1676→      this.lufsTargetGain = Math.max(0.5, Math.min(2.5, Math.pow(10, gainDb / 20)));
  1677→    }
  1678→    this.lufsGain += (this.lufsTargetGain - this.lufsGain) * 0.000005; // ~4s time constant
  1679→    sample *= this.lufsGain;
  1680→
  1681→    // 5. TRUE-PEAK LIMITING (2x oversample, 1-sample lookahead)
  1682→    //    Detects inter-sample peaks (linear interpolation midpoint) and
  1683→    //    limits them. Prevents clipping that sample-peak limiters miss.
  1684→    const interp = (this.tpPrevInput + sample) * 0.5;
  1685→    const peak = Math.max(Math.abs(this.tpPrevInput), Math.abs(interp), Math.abs(sample));
  1686→    let tpTarget = 1;
  1687→    if (peak > this.ceiling) {
  1688→      tpTarget = this.ceiling / peak;
  1689→    }
  1690→    // Smooth gain: fast attack (catch peaks), slow release (avoid pumping)
  1691→    if (tpTarget < this.tpGainEnv) {
  1692→      this.tpGainEnv += (tpTarget - this.tpGainEnv) * (dt / this.tpAttack);
  1693→    } else {
  1694→      this.tpGainEnv += (tpTarget - this.tpGainEnv) * (dt / this.tpRelease);
  1695→    }
  1696→    // Output the previous sample with current gain (1-sample lookahead delay)
  1697→    const output = this.tpPrevInput * this.tpGainEnv;
  1698→    this.tpPrevInput = sample;
  1699→
  1700→    // 6. FINAL TANH (soft clip safety — prevents any remaining overshoot)
  1701→    return fastTanh(output * this.gain);
  1702→  }
  1703→}
