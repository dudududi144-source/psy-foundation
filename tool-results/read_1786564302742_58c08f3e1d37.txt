   563→   563→// ====================== 10. A/B comparison ======================
   564→   564→
   565→   565→describe('A/B comparison: new architecture vs old', () => {
   566→   566→  // Comprehensive kick-bass alignment: fraction of kick steps that have a
   567→   567→  // co-occurring bass note. The old bass-behavior generates notes from a
   568→   568→  // fixed K-B-B-B grammar ([0, 8, 12, 15]) that doesn't know the kick
   569→   569→  // pattern; the new CompositionEngine reads the groove and aligns the
   570→   570→  // bass to the kick (LOCKED) or to the gaps (COMPLEMENTARY — still hits
   571→   571→  // step 0, which is always a kick step).
   572→   572→  function kickBassCooccurrence(
   573→   573→    kick: { step: number; bar: number }[],
   574→   574→    bass: { step: number; bar: number }[],
   575→   575→    bars: number
   576→   576→  ): number {
   577→   577→    let total = 0
   578→   578→    let hit = 0
   579→   579→    for (let bar = 0; bar < bars; bar++) {
   580→   580→      const kickSteps = new Set(kick.filter((k) => k.bar === bar).map((k) => k.step))
   581→   581→      const bassSteps = new Set(bass.filter((b) => b.bar === bar).map((b) => b.step))
   582→   582→      if (kickSteps.size === 0) continue
   583→   583→      for (const ks of kickSteps) {
   584→   584→        total++
   585→   585→        if (bassSteps.has(ks)) hit++
   586→   586→      }
   587→   587→    }
   588→   588→    return total > 0 ? hit / total : 0
   589→   589→  }
   590→   590→
   591→   591→  test('new CompositionEngine has higher kick-bass alignment than old bass-behavior', () => {
   592→   592→    // Use a custom grammar: BROKEN kick ([0, 3, 6, 10]) with LOCKED bass.
   593→   593→    // The old K-B-B-B bass ([0, 8, 12, 15]) only co-occurs with the kick on
   594→   594→    // step 0, so old alignment = 1/4 = 0.25.
   595→   595→    // The new LOCKED bass hits every kick step (ROOT on 0, 3, 6, 10), so
   596→   596→    // new alignment = 4/4 = 1.0 — a 4x improvement.
   597→   597→    const ctx = phrygianContext({ sectionRole: 'acid' })
   598→   598→    const baseGrammar = getStyleGrammar('acid')
   599→   599→    const grammar = { ...baseGrammar, bassAlignment: 'LOCKED' as const }
   600→   600→    const groove = buildGroovePlan({ context: ctx, seed: 1, bars: 32, grammar })
   601→   601→    expect(groove.kickSteps).toEqual([0, 3, 6, 10])
   602→   602→    expect(groove.bassKickAlignment).toBe('LOCKED')
   603→   603→
   604→   604→    // --- OLD architecture: bass generated independently of kick ---
   605→   605→    const oldBassBehavior = generateBassBehavior({
   606→   606→      context: ctx,
   607→   607→      harmonicContext: [4, 8, 11],
   608→   608→      seed: 1,
   609→   609→      bars: 32,
   610→   610→      stepsPerBar: 16,
   611→   611→    })
   612→   612→    const oldBassNotes = oldBassBehavior.notes.map((n) => ({
   613→   613→      midi: n.midi,
   614→   614→      step: n.step % 16,
   615→   615→      bar: Math.floor(n.step / 16),
   616→   616→      function: n.function,
   617→   617→    }))
   618→   618→    const oldKickNotes: { step: number; bar: number }[] = []
   619→   619→    for (let bar = 0; bar < 32; bar++) {
   620→   620→      for (const step of groove.kickSteps) {
   621→   621→        oldKickNotes.push({ step, bar })
   622→   622→      }
   623→   623→    }
   624→   624→    const oldAlignment = kickBassCooccurrence(oldKickNotes, oldBassNotes, 32)
   625→   625→
   626→   626→    // --- NEW architecture: bass composed against the groove ---
   627→   627→    const engine = new CompositionEngine({ seed: 1, context: ctx, grammar })
   628→   628→    const section = engine.composeSection({ bars: 32 })
   629→   629→    const notes = engine.renderNotes(section)
   630→   630→    const newAlignment = kickBassCooccurrence(
   631→   631→      notes.kick,
   632→   632→      notes.bass.map((b) => ({ step: b.step, bar: b.bar })),
   633→   633→      32
   634→   634→    )
   635→   635→
   636→   636→    // The new architecture should produce strictly higher kick-bass alignment
   637→   637→    // than the old independent-bass grammar.
   638→   638→    expect(newAlignment).toBeGreaterThan(oldAlignment)
   639→   639→    // Sanity: new alignment should be at least 2x the old.
   640→   640→    expect(newAlignment / Math.max(oldAlignment, 0.01)).toBeGreaterThan(2)
   641→   641→  })
   642→   642→
   643→   643→  test('measureBassKickAlignment utility reports the expected ratio', () => {
   644→   644→    const bass = [
   645→   645→      { step: 0, bar: 0 },
   646→   646→      { step: 0, bar: 1 },
   647→   647→      { step: 4, bar: 2 }, // bar 2 has no beat-1 bass
   648→   648→    ]
   649→   649→    const kick = [
   650→   650→      { step: 0, bar: 0 },
   651→   651→      { step: 0, bar: 1 },
   652→   652→      { step: 0, bar: 2 },
   653→   653→      { step: 0, bar: 3 }, // bar 3 has no bass
   654→   654→    ]
   655→   655→    // 2 bars (0, 1) out of 4 have both bass+kick on beat 1.
   656→   656→    const alignment = measureBassKickAlignment(bass, kick, 4)
   657→   657→    expect(alignment).toBeCloseTo(0.5, 5)
   658→   658→  })
