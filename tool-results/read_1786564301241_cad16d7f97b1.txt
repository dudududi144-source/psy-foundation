    48→function makeTestMotif(midis: number[], rootPc = 4): ReturnType<typeof createMotif> {
    49→  const notes = midis.map((m, i) => ({
    50→    step: i * 2,
    51→    midi: m,
    52→    velocity: 0.7,
    53→    durationSteps: 2,
    54→    accent: i === 0,
    55→  }))
    56→  return createMotif(notes, {
    57→    id: `test-${midis.join('-')}`,
    58→    rootPc,
    59→    scaleName: 'phrygian-dominant',
    60→    steps: 16,
    61→    role: 'lead',
    62→  })
    63→}

   563→// ====================== 10. A/B comparison ======================
   564→
   565→describe('A/B comparison: new architecture vs old', () => {
   566→  // Comprehensive kick-bass alignment: fraction of kick steps that have a
   567→  // co-occurring bass note. The old bass-behavior generates notes from a
   568→  // fixed K-B-B-B grammar ([0, 8, 12, 15]) that doesn't know the kick
   569→  // pattern; the new CompositionEngine reads the groove and aligns the
   570→  // bass to the kick (LOCKED) or to the gaps (COMPLEMENTARY — still hits
   571→  // step 0, which is always a kick step).
   572→  function kickBassCooccurrence(
   573→    kick: { step: number; bar: number }[],
   574→    bass: { step: number; bar: number }[],
   575→    bars: number
   576→  ): number {
   577→    let total = 0
   578→    let hit = 0
   579→    for (let bar = 0; bar < bars; bar++) {
   580→      const kickSteps = new Set(kick.filter((k) => k.bar === bar).map((k) => k.step))
   581→      const bassSteps = new Set(bass.filter((b) => b.bar === bar).map((b) => b.step))
   582→      if (kickSteps.size === 0) continue
   583→      for (const ks of kickSteps) {
   584→        total++
   585→        if (bassSteps.has(ks)) hit++
   586→      }
   587→    }
   588→    return total > 0 ? hit / total : 0
   589→  }
   590→
   591→  test('new CompositionEngine has higher kick-bass alignment than old bass-behavior', () => {
   592→    // Use a custom grammar: BROKEN kick ([0, 3, 6, 10]) with LOCKED bass.
   593→    // The old K-B-B-B bass ([0, 8, 12, 15]) only co-occurs with the kick on
   594→    // step 0, so old alignment = 1/4 = 0.25.
   595→    // The new LOCKED bass hits every kick step (ROOT on 0, 3, 6, 10), so
   596→    // new alignment = 4/4 = 1.0 — a 4x improvement.
   597→    const ctx = phrygianContext({ sectionRole: 'acid' })
   598→    const baseGrammar = getStyleGrammar('acid')
   599→    const grammar = { ...baseGrammar, bassAlignment: 'LOCKED' as const }
   600→    const groove = buildGroovePlan({ context: ctx, seed: 1, bars: 32, grammar })
   601→    expect(groove.kickSteps).toEqual([0, 3, 6, 10])
   602→    expect(groove.bassKickAlignment).toBe('LOCKED')
   603→
   604→    // --- OLD architecture: bass generated independently of kick ---
   605→    const oldBassBehavior = generateBassBehavior({
   606→      context: ctx,
   607→      harmonicContext: [4, 8, 11],
   608→      seed: 1,
   609→      bars: 32,
   610→      stepsPerBar: 16,
   611→    })
   612→    const oldBassNotes = oldBassBehavior.notes.map((n) => ({
   613→      midi: n.midi,
   614→      step: n.step % 16,
   615→      bar: Math.floor(n.step / 16),
   616→      function: n.function,
   617→    }))
   618→    const oldKickNotes: { step: number; bar: number }[] = []
   619→    for (let bar = 0; bar < 32; bar++) {
   620→      for (const step of groove.kickSteps) {
   621→        oldKickNotes.push({ step, bar })
   622→      }
   623→    }
   624→    const oldAlignment = kickBassCooccurrence(oldKickNotes, oldBassNotes, 32)
   625→
   626→    // --- NEW architecture: bass composed against the groove ---
   627→    const engine = new CompositionEngine({ seed: 1, context: ctx, grammar })
   628→    const section = engine.composeSection({ bars: 32 })
   629→    const notes = engine.renderNotes(section)
   630→    const newAlignment = kickBassCooccurrence(
   631→      notes.kick,
   632→      notes.bass.map((b) => ({ step: b.step, bar: b.bar })),
   633→      32
   634→    )
   635→
   636→    // The new architecture should produce strictly higher kick-bass alignment
   637→    // than the old independent-bass grammar.
   638→    expect(newAlignment).toBeGreaterThan(oldAlignment)
   639→    // Sanity: new alignment should be at least 2x the old.
   640→    expect(newAlignment / Math.max(oldAlignment, 0.01)).toBeGreaterThan(2)
   641→  })
   642→
   643→  test('measureBassKickAlignment utility reports the expected ratio', () => {
   644→    const bass = [
   645→      { step: 0, bar: 0 },
   646→      { step: 0, bar: 1 },
   647→      { step: 4, bar: 2 }, // bar 2 has no beat-1 bass
   648→    ]
   649→    const kick = [
   650→      { step: 0, bar: 0 },
   651→      { step: 0, bar: 1 },
   652→      { step: 0, bar: 2 },
   653→      { step: 0, bar: 3 }, // bar 3 has no bass
   654→    ]
   655→    // 2 bars (0, 1) out of 4 have both bass+kick on beat 1.
   656→    const alignment = measureBassKickAlignment(bass, kick, 4)
   657→    expect(alignment).toBeCloseTo(0.5, 5)
   658→  })
   659→})
