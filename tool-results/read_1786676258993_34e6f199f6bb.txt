     1→/**
     2→ * AudioCritic — analyzes rendered PCM and returns an actionable musical/sonic
     3→ * diagnosis.
     4→ *
     5→ * F22 requirement: the critic must NOT just return peak/RMS/centroid. It must
     6→ * return a diagnosis that maps perceptual problems to probable causes to
     7→ * parameter families to musical/synthesis corrections.
     8→ *
     9→ * The critique covers 8 areas:
    10→ *   lowEnd — kick clarity, bass clarity, kick/bass separation, sub mud, phase risk
    11→ *   transient — punch, attack sharpness, kick definition
    12→ *   bass — note separation, decay overlap, pitch stability, spectral consistency
    13→ *   groove — onset clarity, pocket consistency, kick/bass lock, excessive uniformity
    14→ *   lead — articulation, melodic clarity, phrase contrast, repetition balance, harmonic clarity
    15→ *   timbre — brightness, roughness, noisiness, spectral movement, modulation depth, identity strength
    16→ *   mix — low-mid mud, harshness, high-end presence, stereo contrast, dynamic range, masking
    17→ *   musicality — tension/release, motif identity, development, call/response, rhythmic interest
    18→ *
    19→ * Each failure includes a diagnosis string + a correction hint that the
    20→ * generator/synth can act on.
    21→ */
    22→
    23→export interface AudioCritique {
    24→  lowEnd: {
    25→    kickClarity: number
    26→    bassClarity: number
    27→    kickBassSeparation: number
    28→    subMud: number
    29→    phaseRisk: number
    30→  }
    31→  transient: {
    32→    punch: number
    33→    attackSharpness: number
    34→    kickDefinition: number
    35→  }
    36→  bass: {
    37→    noteSeparation: number
    38→    decayOverlap: number
    39→    pitchStability: number
    40→    spectralConsistency: number
    41→  }
    42→  groove: {
    43→    onsetClarity: number
    44→    pocketConsistency: number
    45→    kickBassLock: number
    46→    excessiveUniformity: number
    47→  }
    48→  lead: {
    49→    articulation: number
    50→    melodicClarity: number
    51→    phraseContrast: number
    52→    repetitionBalance: number
    53→    harmonicClarity: number
    54→  }
    55→  timbre: {
    56→    brightness: number
    57→    roughness: number
    58→    noisiness: number
    59→    spectralMovement: number
    60→    modulationDepth: number
    61→    identityStrength: number
    62→  }
    63→  mix: {
    64→    lowMidMud: number
    65→    harshness: number
    66→    highEndPresence: number
    67→    stereoContrast: number
    68→    dynamicRange: number
    69→    masking: number
    70→  }
    71→  musicality: {
    72→    tensionRelease: number
    73→    motifIdentity: number
    74→    development: number
    75→    callResponse: number
    76→    rhythmicInterest: number
    77→  }
    78→  overallScore: number
    79→  failures: AudioFailure[]
    80→}

    82→export interface AudioFailure {
    83→  /** Failure code (e.g. BASS_DECAY_TOO_LONG). */
    84→  code: string
    85→  /** Human-readable diagnosis. */
    86→  diagnosis: string
    87→  /** Which parameter family to change. */
    88→  correctionTarget: string
    89→  /** Suggested correction direction (e.g. "shorten bass decay", "raise click brightness"). */
    90→  correctionHint: string
    91→  /** Severity 0..1. */
    92→  severity: number
    93→}

    95→/**
    96→ * Analyze a rendered PCM buffer and return an actionable critique.
    97→ *
    98→ * @param pcm — mono PCM Float32Array
    99→ * @param sampleRate — sample rate
   100→ * @param bpm — BPM (for rhythmic analysis)
   101→ * @param stepsPerBar — steps per bar (for onset grid analysis)
   102→ */
   103→export function critiqueAudio(
   104→  pcm: Float32Array,
   105→  sampleRate: number,
   106→  bpm: number,
   107→  stepsPerBar = 16
   108→): AudioCritique {
   109→  const failures: AudioFailure[] = []
   110→
   111→  // ── Basic levels ──
   112→  const peak = findPeak(pcm)
   113→  const rms = computeRMS(pcm)
   114→  const dynamicRange = peak / Math.max(0.0001, rms)
   115→
   116→  // ── Spectral analysis ──
   117→  const fftSize = 2048
   118→  const spectra = computeSpectrogram(pcm, fftSize, sampleRate)
   119→  const avgSpectrum = averageSpectrum(spectra)
   120→  const spectralCentroid = computeCentroid(avgSpectrum, sampleRate, fftSize)
   121→  const _spectralRolloff = computeRolloff(avgSpectrum, sampleRate, fftSize)
   122→
   123→  // ── Band energy ──
   124→  const subEnergy = bandEnergy(avgSpectrum, sampleRate, fftSize, 20, 80)
   125→  const bassEnergy = bandEnergy(avgSpectrum, sampleRate, fftSize, 80, 250)
   126→  const lowMidEnergy = bandEnergy(avgSpectrum, sampleRate, fftSize, 250, 500)
   127→  const midEnergy = bandEnergy(avgSpectrum, sampleRate, fftSize, 500, 2000)
   128→  const highMidEnergy = bandEnergy(avgSpectrum, sampleRate, fftSize, 2000, 5000)
   129→  const highEnergy = bandEnergy(avgSpectrum, sampleRate, fftSize, 5000, 12000)
   130→
   131→  // ── Onset detection ──
   132→  const onsets = detectOnsets(pcm, sampleRate, bpm, stepsPerBar)
   133→  const onsetSharpness = computeOnsetSharpness(pcm, onsets, sampleRate)
   134→
   135→  // ── Transient analysis ──
   136→  const _transientEnv = computeTransientEnvelope(pcm, sampleRate)
   137→  const punch = computePunch(pcm, onsets, sampleRate)
   138→
   139→  // ── Spectral movement (how much the spectrum changes over time) ──
   140→  const spectralMovement = computeSpectralMovement(spectra)
   141→
   142→  // ── Low-end analysis ──
   143→  const kickClarity = onsetSharpness // kick clarity = how sharp the onsets are
   144→  const bassClarity = Math.min(1, bassEnergy / Math.max(0.01, subEnergy + bassEnergy))
   145→  const kickBassSeparation = computeKickBassSeparation(avgSpectrum, sampleRate, fftSize)
   146→  const subMud = computeSubMud(avgSpectrum, sampleRate, fftSize)
   147→  const phaseRisk = computePhaseRisk(pcm, sampleRate)
   148→
   149→  // ── Bass analysis ──
   150→  const noteSeparation = computeNoteSeparation(pcm, onsets, sampleRate, bpm)
   151→  const decayOverlap = computeDecayOverlap(pcm, onsets, sampleRate, bpm)
   152→  const pitchStability = computePitchStability(pcm, sampleRate)
   153→  const spectralConsistency = 1 - spectralMovement
   154→
   155→  // ── Groove analysis ──
   156→  const onsetClarity = onsetSharpness
   157→  const pocketConsistency = computePocketConsistency(onsets, sampleRate, bpm, stepsPerBar)
   158→  const kickBassLock = computeKickBassLock(pcm, onsets, sampleRate, bpm)
   159→  const excessiveUniformity = computeExcessiveUniformity(pcm, sampleRate, bpm)
   160→
   161→  // ── Lead analysis ──
   162→  const articulation = computeArticulation(pcm, sampleRate)
   163→  const melodicClarity = computeMelodicClarity(pcm, sampleRate)
   164→  const phraseContrast = computePhraseContrast(pcm, sampleRate, bpm)
   165→  const repetitionBalance = computeRepetitionBalance(pcm, sampleRate, bpm)
   166→  const harmonicClarity = computeHarmonicClarity(avgSpectrum, sampleRate, fftSize)
   167→
   168→  // ── Timbre analysis ──
   169→  const brightness = Math.min(1, spectralCentroid / 5000)
   170→  const roughness = computeRoughness(avgSpectrum, sampleRate, fftSize)
   171→  const noisiness = computeNoisiness(avgSpectrum, sampleRate, fftSize)
   172→  const modulationDepth = spectralMovement
   173→  const identityStrength = computeIdentityStrength(spectra)
   174→
   175→  // ── Mix analysis ──
   176→  const lowMidMud = computeLowMidMud(avgSpectrum, sampleRate, fftSize)
   177→  const harshness = computeHarshness(avgSpectrum, sampleRate, fftSize)
   178→  const highEndPresence =
   179→    highEnergy /
   180→    Math.max(0.01, subEnergy + bassEnergy + lowMidEnergy + midEnergy + highMidEnergy + highEnergy)
   181→  const stereoContrast = 0.5 // mono for now
   182→  const masking = computeMasking(avgSpectrum, sampleRate, fftSize)
   183→
   184→  // ── Musicality analysis ──
   185→  const tensionRelease = computeTensionRelease(pcm, sampleRate, bpm)
   186→  const motifIdentity = computeMotifIdentity(pcm, sampleRate, bpm)
   187→  const development = computeDevelopment(pcm, sampleRate, bpm)
   188→  const callResponse = computeCallResponse(pcm, sampleRate, bpm)
   189→  const rhythmicInterest = 1 - excessiveUniformity
   190→
   191→  // ── Diagnose failures ──
   192→  if (subMud > 0.6) {
   193→    failures.push({
   194→      code: 'LOW_MID_MUD',
   195→      diagnosis: `Low-mid energy is too high (${lowMidEnergy.toFixed(3)}), creating mud that masks kick and bass clarity.`,
   196→      correctionTarget: 'bass.mid.cutoffHz',
   197→      correctionHint: 'Lower the mid bass filter cutoff and reduce sub sustain',
   198→      severity: subMud,
   199→    })
   200→  }
   201→  if (kickClarity < 0.4) {
   202→    failures.push({
   203→      code: 'KICK_TRANSIENT_MASKED',
   204→      diagnosis: `Kick transient is not sharp enough (sharpness=${onsetSharpness.toFixed(3)}), likely masked by bass decay overlap.`,
   205→      correctionTarget: 'kick.clickAmount / bass.decay',
   206→      correctionHint: 'Increase click amount and brightness; shorten bass decay to leave space',
   207→      severity: 1 - kickClarity,
   208→    })
   209→  }
   210→  if (decayOverlap > 0.5) {
   211→    failures.push({
   212→      code: 'BASS_DECAY_TOO_LONG',
   213→      diagnosis: `Bass decay overlaps ${decayOverlap.toFixed(2)} of the next kick onset — bass is smearing into the kick.`,
   214→      correctionTarget: 'bass.decay / bass.release',
   215→      correctionHint: 'Shorten bass decay to 0.06s or less; reduce sustain to 0',
   216→      severity: decayOverlap,
   217→    })
   218→  }
   219→  if (punch < 0.3) {
   220→    failures.push({
   221→      code: 'WEAK_PUNCH',
   222→      diagnosis: `Kick punch is weak (${punch.toFixed(3)}) — the body envelope isn't sharp enough.`,
   223→      correctionTarget: 'kick.bodyDecay / kick.pitchDropTime',
   224→      correctionHint: 'Shorten body decay to 0.12s; increase pitch drop speed',
   225→      severity: 1 - punch,
   226→    })
   227→  }
   228→  if (spectralMovement < 0.15) {
   229→    failures.push({
   230→      code: 'NO_TIMBRAL_MOVEMENT',
   231→      diagnosis: `Spectral movement is low (${spectralMovement.toFixed(3)}) — the sound is static across the section.`,
   232→      correctionTarget: 'lead.filterEnvAmount / lead.fmAmount',
   233→      correctionHint:
   234→        'Increase filter envelope amount; add FM modulation; introduce sound trajectory',
   235→      severity: 0.5 - spectralMovement,
   236→    })
   237→  }
   238→  if (brightness > 0.7) {
   239→    failures.push({
   240→      code: 'LEAD_TOO_BRIGHT',
   241→      diagnosis: `Lead is too bright (centroid=${spectralCentroid.toFixed(0)}Hz) — harshness in the high mids.`,
   242→      correctionTarget: 'lead.cutoffHz / lead.resonance',
   243→      correctionHint: 'Lower filter cutoff; reduce resonance; add warmer saturation',
   244→      severity: brightness - 0.7,
   245→    })
   246→  }
   247→  if (brightness < 0.2 && highEndPresence < 0.05) {
   248→    failures.push({
   249→      code: 'HIGH_END_TOO_WEAK',
   250→      diagnosis: `High-end energy is weak (${(highEndPresence * 100).toFixed(1)}%) — hats and lead are masked by low-mid energy.`,
   251→      correctionTarget: 'hatGain / lead.cutoffHz',
   252→      correctionHint: 'Raise hat gain; raise lead cutoff; introduce upper harmonic layer',
   253→      severity: 0.2 - brightness,
   254→    })
   255→  }
   256→  if (excessiveUniformity > 0.7) {
   257→    failures.push({
   258→      code: 'RHYTHMIC_PATTERN_TOO_UNIFORM',
   259→      diagnosis: `Rhythmic pattern is too uniform (${excessiveUniformity.toFixed(3)}) — lacks dynamic variation.`,
   260→      correctionTarget: 'groove.velocityContour / ghost.notes',
   261→      correctionHint: 'Add velocity variation; introduce ghost notes; vary accent patterns',
   262→      severity: excessiveUniformity - 0.7,
   263→    })
   264→  }
   265→  if (kickBassSeparation < 0.3) {
   266→    failures.push({
   267→      code: 'KICK_BASS_PHASE_RISK',
   268→      diagnosis: `Kick and bass are not spectrally separated (${kickBassSeparation.toFixed(3)}) — they occupy the same frequency range.`,
   269→      correctionTarget: 'kick.pitchEnd / bass.sub.cutoffHz',
   270→      correctionHint:
   271→        'Lower kick sub frequency; raise bass sub cutoff; ensure kick is done before bass starts',
   272→      severity: 0.3 - kickBassSeparation,
   273→    })
   274→  }
   275→  if (melodicClarity < 0.3) {
   276→    failures.push({
   277→      code: 'LEAD_TOO_STATIC',
   278→      diagnosis: `Lead melodic clarity is low (${melodicClarity.toFixed(3)}) — the phrase lacks articulation and movement.`,
   279→      correctionTarget: 'lead.attack / lead.filterEnvAmount',
   280→      correctionHint: 'Shorten attack; increase filter envelope; add pitch movement',
   281→      severity: 0.3 - melodicClarity,
   282→    })
   283→  }
   284→  if (masking > 0.6) {
   285→    failures.push({
   286→      code: 'LEAD_MASKING_BASS',
   287→      diagnosis: `Masking detected (${masking.toFixed(3)}) — lead frequencies overlap with bass, reducing clarity.`,
   288→      correctionTarget: 'lead.cutoffHz / bass.mid.cutoffHz',
   289→      correctionHint: 'Raise lead cutoff above 2000Hz; lower bass mid cutoff below 800Hz',
   290→      severity: masking - 0.6,
   291→    })
   292→  }
   293→  if (motifIdentity < 0.3) {
   294→    failures.push({
   295→      code: 'WEAK_MOTIF_IDENTITY',
   296→      diagnosis: `Motif identity is weak (${motifIdentity.toFixed(3)}) — the listener cannot recognize a returning motif.`,
   297→      correctionTarget: 'phraseMaterial.developmentHistory',
   298→      correctionHint: 'Reduce variation amount; use CONTINUE more; callback to earlier motifs',
   299→      severity: 0.3 - motifIdentity,
   300→    })
   301→  }
   302→
   303→  // ── Overall score ──
   304→  const scores = [
   305→    kickClarity,
   306→    bassClarity,
   307→    kickBassSeparation,
   308→    1 - subMud,
   309→    1 - phaseRisk,
   310→    punch,
   311→    onsetSharpness,
   312→    kickDefinition(punch, onsetSharpness),
   313→    noteSeparation,
   314→    1 - decayOverlap,
   315→    pitchStability,
   316→    spectralConsistency,
   317→    onsetClarity,
   318→    pocketConsistency,
   319→    kickBassLock,
   320→    1 - excessiveUniformity,
   321→    articulation,
   322→    melodicClarity,
   323→    phraseContrast,
   324→    repetitionBalance,
   325→    harmonicClarity,
   326→    1 - Math.abs(brightness - 0.5),
   327→    1 - roughness,
   328→    1 - noisiness,
   329→    spectralMovement,
   330→    modulationDepth,
   331→    identityStrength,
   332→    1 - lowMidMud,
   333→    1 - harshness,
   334→    highEndPresence * 5,
   335→    stereoContrast,
   336→    Math.min(1, dynamicRange / 10),
   337→    1 - masking,
   338→    tensionRelease,
   339→    motifIdentity,
   340→    development,
   341→    callResponse,
   342→    rhythmicInterest,
   343→  ]
   344→  const overallScore = scores.reduce((s, v) => s + Math.max(0, Math.min(1, v)), 0) / scores.length
   345→
   346→  return {
   347→    lowEnd: { kickClarity, bassClarity, kickBassSeparation, subMud, phaseRisk },
   348→    transient: {
   349→      punch,
   350→      attackSharpness: onsetSharpness,
   351→      kickDefinition: kickDefinition(punch, onsetSharpness),
   352→    },
   353→    bass: { noteSeparation, decayOverlap, pitchStability, spectralConsistency },
   354→    groove: { onsetClarity, pocketConsistency, kickBassLock, excessiveUniformity },
   355→    lead: { articulation, melodicClarity, phraseContrast, repetitionBalance, harmonicClarity },
   356→    timbre: {
   357→      brightness,
   358→      roughness,
   359→      noisiness,
   360→      spectralMovement,
   361→      modulationDepth,
   362→      identityStrength,
   363→    },
   364→    mix: {
   365→      lowMidMud,
   366→      harshness,
   367→      highEndPresence,
   368→      stereoContrast,
   369→      dynamicRange: Math.min(1, dynamicRange / 10),
   370→      masking,
   371→    },
   372→    musicality: { tensionRelease, motifIdentity, development, callResponse, rhythmicInterest },
   373→    overallScore,
   374→    failures,
   375→  }
   376→}

   378→function kickDefinition(punch: number, sharpness: number): number {
   379→  return (punch + sharpness) / 2
   380→}

   384→function findPeak(pcm: Float32Array): number {
   385→  let peak = 0
   386→  for (let i = 0; i < pcm.length; i++) {
   387→    const abs = Math.abs(pcm[i] ?? 0)
   388→    if (abs > peak) peak = abs
   389→  }
   390→  return peak
   391→}
   392→
   393→function computeRMS(pcm: Float32Array): number {
   394→  let sum = 0
   395→  for (let i = 0; i < pcm.length; i++) {
   396→    sum += (pcm[i] ?? 0) ** 2
   397→  }
   398→  return Math.sqrt(sum / pcm.length)
   399→}

   401→/**
   402→ * Compute a simple spectrogram using a sliding window FFT.
   403→ * For performance, uses a simplified DFT (not a full FFT) at reduced resolution.
   404→ */
   405→function computeSpectrogram(pcm: Float32Array, fftSize: number, _sampleRate: number): number[][] {
   406→  const hopSize = fftSize
   407→  const numFrames = Math.floor(pcm.length / hopSize)
   408→  const numBins = Math.min(128, fftSize / 2) // reduced resolution for performance
   409→  const spectra: number[][] = []
   410→  const window = hannWindow(fftSize)
   411→  for (let f = 0; f < numFrames; f++) {
   412→    const start = f * hopSize
   413→    const frame = new Float32Array(fftSize)
   414→    for (let i = 0; i < fftSize; i++) {
   415→      frame[i] = (pcm[start + i] ?? 0) * (window[i] ?? 0)
   416→    }
   417→    const spectrum = computeDFT(frame, numBins)
   418→    spectra.push(spectrum)
   419→  }
   420→  return spectra
   421→}
   422→
   423→function hannWindow(size: number): Float32Array {
   424→  const w = new Float32Array(size)
   425→  for (let i = 0; i < size; i++) {
   426→    w[i] = 0.5 * (1 - Math.cos((2 * Math.PI * i) / (size - 1)))
   427→  }
   428→  return w
   429→}
   430→
   431→/**
   432→ * Simplified DFT — computes the magnitude spectrum at numBins frequency bins.
   433→ * Uses direct computation (O(n*k)) rather than FFT for simplicity. This is
   434→ * slow for large fftSize but adequate for analysis (not real-time).
   435→ */
   436→function computeDFT(frame: Float32Array, numBins: number): number[] {
   437→  const N = frame.length
   438→  const spectrum = new Array(numBins).fill(0)
   439→  for (let k = 0; k < numBins; k++) {
   440→    let real = 0
   441→    let imag = 0
   442→    const freq = k / N
   443→    for (let n = 0; n < N; n++) {
   444→      const angle = -2 * Math.PI * freq * n
   445→      real += (frame[n] ?? 0) * Math.cos(angle)
   446→      imag += (frame[n] ?? 0) * Math.sin(angle)
   447→    }
   448→    spectrum[k] = Math.sqrt(real * real + imag * imag) / N
   449→  }
   450→  return spectrum
   451→}
   452→
   453→function averageSpectrum(spectra: number[][]): number[] {
   454→  if (spectra.length === 0) return []
   455→  const numBins = spectra[0]?.length
   456→  const avg = new Array(numBins).fill(0)
   457→  for (const s of spectra) {
   458→    for (let i = 0; i < numBins; i++) {
   459→      avg[i] = (avg[i] ?? 0) + (s[i] ?? 0)
   460→    }
   461→  }
   462→  for (let i = 0; i < numBins; i++) {
   463→    avg[i] = (avg[i] ?? 0) / spectra.length
   464→  }
   465→  return avg
   466→}

   468→function computeCentroid(spectrum: number[], sampleRate: number, fftSize: number): number {
   469→  let weighted = 0
   470→  let total = 0
   471→  for (let i = 0; i < spectrum.length; i++) {
   472→    const freq = (i * sampleRate) / fftSize
   473→    const mag = spectrum[i] ?? 0
   474→    weighted += freq * mag
   475→    total += mag
   476→  }
   477→  return total > 0 ? weighted / total : 0
   478→}
   479→
   480→function computeRolloff(spectrum: number[], sampleRate: number, fftSize: number): number {
   481→  let total = 0
   482→  for (const m of spectrum) total += m
   483→  if (total === 0) return 0
   484→  let cumulative = 0
   485→  for (let i = 0; i < spectrum.length; i++) {
   486→    cumulative += spectrum[i] ?? 0
   487→    if (cumulative > 0.85 * total) {
   488→      return (i * sampleRate) / fftSize
   489→    }
   490→  }
   491→  return 0
   492→}

   494→function bandEnergy(
   495→  spectrum: number[],
   496→  sampleRate: number,
   497→  fftSize: number,
   498→  lowHz: number,
   499→  highHz: number
   500→): number {
   501→  const lowBin = Math.floor((lowHz * fftSize) / sampleRate)
   502→  const highBin = Math.ceil((highHz * fftSize) / sampleRate)
   503→  let energy = 0
   504→  for (let i = lowBin; i <= highBin && i < spectrum.length; i++) {
   505→    energy += spectrum[i] ?? 0
   506→  }
   507→  return energy
   508→}

   510→function detectOnsets(
   511→  pcm: Float32Array,
   512→  sampleRate: number,
   513→  bpm: number,
   514→  stepsPerBar: number
   515→): number[] {
   516→  const secondsPerStep = 60 / bpm / (stepsPerBar / 4)
   517→  const samplesPerStep = Math.ceil(secondsPerStep * sampleRate)
   518→  const onsets: number[] = []
   519→  for (let pos = 0; pos < pcm.length; pos += samplesPerStep) {
   520→    // Measure the energy in a short window after the onset.
   521→    const windowSize = Math.floor(sampleRate * 0.01) // 10ms
   522→    let energy = 0
   523→    for (let i = 0; i < windowSize && pos + i < pcm.length; i++) {
   524→      energy += Math.abs(pcm[pos + i] ?? 0)
   525→    }
   526→    onsets.push(energy / windowSize)
   527→  }
   528→  return onsets
   529→}
   530→
   531→function computeOnsetSharpness(pcm: Float32Array, onsets: number[], sampleRate: number): number {
   532→  // Measure how quickly energy rises at each onset.
   533→  let totalSharpness = 0
   534→  let count = 0
   535→  const samplesPerStep = Math.floor(pcm.length / Math.max(1, onsets.length))
   536→  const preWindow = Math.floor(sampleRate * 0.005) // 5ms before
   537→  const postWindow = Math.floor(sampleRate * 0.005) // 5ms after
   538→  for (let s = 0; s < onsets.length; s++) {
   539→    const pos = s * samplesPerStep
   540→    let pre = 0
   541→    let post = 0
   542→    for (let i = 0; i < preWindow; i++) {
   543→      pre += Math.abs(pcm[pos - preWindow + i] ?? 0)
   544→    }
   545→    for (let i = 0; i < postWindow; i++) {
   546→      post += Math.abs(pcm[pos + i] ?? 0)
   547→    }
   548→    pre /= preWindow
   549→    post /= postWindow
   550→    const sharpness = post / Math.max(0.001, pre + post)
   551→    totalSharpness += sharpness
   552→    count++
   553→  }
   554→  return count > 0 ? totalSharpness / count : 0
   555→}

   570→function computePunch(pcm: Float32Array, onsets: number[], sampleRate: number): number {

   572→  let totalPunch = 0

   577→  for (let s = 0; s < onsets.length; s++) {

   580→    let peakLong = 0
   581→    for (let i = 0; i < shortWindow && pos + i < pcm.length; i++) {
   582→      peakShort = Math.max(peakShort, Math.abs(pcm[pos + i] ?? 0))
   583→    }
   584→    for (let i = 0; i < longWindow && pos + i < pcm.length; i++) {
   585→      peakLong = Math.max(peakLong, Math.abs(pcm[pos + i] ?? 0))
   586→    }
   587→    totalPunch += peakShort / Math.max(0.001, peakLong)
   588→    count++
   589→  }
   590→  return count > 0 ? totalPunch / count : 0

   593→function computeSpectralMovement(spectra: number[][]): number {

   639→function computeNoteSeparation(
   640→  pcm: Float32Array,
   641→  onsets: number[],
   642→  _sampleRate: number,
   643→  _bpm: number
   644→): number {
   645→  // Note separation = how much energy drops between onsets.
   646→  const samplesPerStep = Math.floor(pcm.length / Math.max(1, onsets.length))
   647→  let totalDip = 0
   648→  let count = 0
   649→  for (let s = 1; s < onsets.length; s++) {
   650→    const pos = s * samplesPerStep
   651→    const dipPos = pos - Math.floor(samplesPerStep * 0.5)
   652→    let onsetEnergy = 0
   653→    let dipEnergy = 0
   654→    for (let i = 0; i < 50; i++) {
   655→      onsetEnergy += Math.abs(pcm[pos + i] ?? 0)
   656→      dipEnergy += Math.abs(pcm[dipPos + i] ?? 0)
   657→    }
   658→    totalDip += 1 - dipEnergy / Math.max(1, onsetEnergy)
   659→    count++
   660→  }
   661→  return count > 0 ? Math.max(0, totalDip / count) : 0
   662→}

   664→function computeDecayOverlap(
   665→  pcm: Float32Array,
   666→  onsets: number[],
   667→  sampleRate: number,
   668→  _bpm: number
   669→): number {
   670→  // Overlap = how much BASS-MID energy from one kick onset is still present at the next kick.
   671→  //
   672→  // FIX (v2): The original measured full-range energy at every onset → always ~1.0.
   673→  // Fix v1 measured 0-200Hz band → still ~1.0 because sub-bass sustains + kick sub tail.
   674→  //
   675→  // This v2 isolates the BASS-MID band (80-250Hz) via a band-pass filter:
   676→  //   - HP at 80Hz removes the kick sub (0-80Hz) and sub-bass fundamental
   677→  //   - LP at 250Hz removes the lead/hats/upper harmonics
   678→  // The remaining signal is the bass MID layer (saw through Moog), which SHOULD
   679→  // decay tightly (< 0.06s) between kicks. If this band shows high overlap, the
   680→  // bass is genuinely smearing.
   681→  //
   682→  // Additionally, we measure at KICK onsets only (every 4th onset, since the
   683→  // kick is four-on-the-floor). This avoids measuring between consecutive 16th
   684→  // bass notes (which are supposed to be continuous in psytrance).
   685→
   686→  // Band-pass: HP at 80Hz (one-pole) then LP at 250Hz (one-pole).
   687→  const hpAlpha = 2 * Math.PI * 80 / sampleRate
   688→  const hpA = hpAlpha / (1 + hpAlpha)
   689→  const lpAlpha = 2 * Math.PI * 250 / sampleRate
   690→  const lpA = lpAlpha / (1 + lpAlpha)
   691→
   692→  const bassMid = new Float32Array(pcm.length)
   693→  let hpPrev = 0
   694→  let lpPrev = 0
   695→  for (let i = 0; i < pcm.length; i++) {
   696→    const x = pcm[i] ?? 0
   697→    // High-pass: y = x - lp(x)
   698→    hpPrev = hpPrev + hpA * (x - hpPrev)
   699→    const hpOut = x - hpPrev
   700→    // Low-pass on the HP output
   701→    lpPrev = lpPrev + lpA * (hpOut - lpPrev)
   702→    bassMid[i] = lpPrev
   703→  }
   704→
   705→  // Use every 4th onset (kick onsets in four-on-the-floor).
   706→  // If onsets length < 8, use all onsets (short render fallback).
   707→  const step = onsets.length >= 8 ? 4 : 1
   708→  const samplesPerStep = Math.floor(bassMid.length / Math.max(1, onsets.length))
   709→  let totalOverlap = 0
   710→  let count = 0
   711→  for (let s = 0; s < onsets.length - step; s += step) {
   712→    const pos = s * samplesPerStep
   713→    const nextPos = (s + step) * samplesPerStep
   714→    const preNext = nextPos - 5
   715→    if (preNext + 50 >= bassMid.length) continue
   716→    let peakOnset = 0
   717→    let preNextEnergy = 0
   718→    for (let i = 0; i < 50; i++) {
   719→      peakOnset = Math.max(peakOnset, Math.abs(bassMid[pos + i] ?? 0))
   720→      preNextEnergy = Math.max(preNextEnergy, Math.abs(bassMid[preNext + i] ?? 0))
   721→    }
   722→    totalOverlap += preNextEnergy / Math.max(0.001, peakOnset)
   723→    count++
   724→  }
   725→  return count > 0 ? Math.min(1, totalOverlap / count) : 0
   726→}

   728→function computePitchStability(pcm: Float32Array, sampleRate: number): number {
   729→  // Simplified: measure autocorrelation at bass frequencies.
   730→  // High autocorrelation = stable pitch.
   731→  const window = pcm.slice(0, Math.min(pcm.length, 4096))
   732→  let autocorr = 0
   733→  let energy = 0
   734→  const lag = Math.floor(sampleRate / 80) // ~80Hz
   735→  for (let i = 0; i < window.length - lag; i++) {
   736→    autocorr += (window[i] ?? 0) * (window[i + lag] ?? 0)
   737→    energy += (window[i] ?? 0) ** 2
   738→  }
   739→  return energy > 0 ? Math.min(1, autocorr / energy) : 0.5
   740→}

   742→function computePocketConsistency(
   743→  onsets: number[],
   744→  _sampleRate: number,
   745→  _bpm: number,
   746→  _stepsPerBar: number
   747→): number {
   748→  // Consistency = how uniform the onset energies are (pocket = steady).
   749→  if (onsets.length < 2) return 0.5
   750→  const mean = onsets.reduce((s, v) => s + v, 0) / onsets.length
   751→  if (mean === 0) return 0
   752→  const variance = onsets.reduce((s, v) => s + (v - mean) ** 2, 0) / onsets.length
   753→  const cv = Math.sqrt(variance) / mean // coefficient of variation
   754→  return Math.max(0, 1 - cv)
   755→}

   767→function computeExcessiveUniformity(pcm: Float32Array, sampleRate: number, bpm: number): number {
   768→  // Uniformity = how similar consecutive bars are.
   769→  const secondsPerBar = (60 / bpm) * 4
   770→  const samplesPerBar = Math.floor(secondsPerBar * sampleRate)
   771→  if (pcm.length < samplesPerBar * 2) return 0.5
   772→  let totalSim = 0
   773→  let count = 0
   774→  for (let b = 1; b < Math.floor(pcm.length / samplesPerBar); b++) {
   775→    let corr = 0
   776→    let energy = 0
   777→    for (let i = 0; i < samplesPerBar; i++) {
   778→      const cur = pcm[b * samplesPerBar + i] ?? 0
   779→      const prev = pcm[(b - 1) * samplesPerBar + i] ?? 0
   780→      corr += cur * prev
   781→      energy += cur * cur + prev * prev
   782→    }
   783→    totalSim += energy > 0 ? (corr / energy) * 2 : 0
   784→    count++
   785→  }
   786→  return count > 0 ? totalSim / count : 0
   787→}

   789→function computeArticulation(pcm: Float32Array, sampleRate: number): number {
   790→  // Articulation = how much the envelope changes (not sustains).
   791→  const windowSize = Math.floor(sampleRate * 0.02)
   792→  let changes = 0
   793→  let prevEnv = 0
   794→  for (let i = 0; i < pcm.length; i += windowSize) {
   795→    let env = 0
   796→    for (let j = 0; j < windowSize && i + j < pcm.length; j++) {
   797→      env += Math.abs(pcm[i + j] ?? 0)
   798→    }
   799→    env /= windowSize
   800→    changes += Math.abs(env - prevEnv)
   801→    prevEnv = env
   802→  }
   803→  return Math.min(1, (changes / (pcm.length / windowSize)) * 10)
   804→}

   817→function computePhraseContrast(pcm: Float32Array, sampleRate: number, _bpm: number): number {
   818→  // Contrast = how different the first and second halves of the section are.

   821→  const secondHalf = pcm.slice(half)

   823→  const centroid2 = computeCentroid(computeDFT(secondHalf.slice(0, 2048), 128), sampleRate, 2048)
   824→  const diff = Math.abs(centroid1 - centroid2) / Math.max(1, centroid1)
   825→  return Math.min(1, diff)

   896→function computeTensionRelease(pcm: Float32Array, _sampleRate: number, _bpm: number): number {
   897→  // Simplified: tension = energy builds then releases.
   898→  const sections = 4
   899→  const sectionLength = Math.floor(pcm.length / sections)
   900→  const energies: number[] = []
   901→  for (let s = 0; s < sections; s++) {
   902→    let energy = 0
   903→    for (let i = 0; i < sectionLength; i++) {
   904→      energy += Math.abs(pcm[s * sectionLength + i] ?? 0)
   905→    }
   906→    energies.push(energy / sectionLength)
   907→  }
   908→  // Good tension/release: energy peaks in the middle, drops at the end.
   909→  const peak = Math.max(...energies)
   910→  const peakIdx = energies.indexOf(peak)
   911→  const endEnergy = energies[energies.length - 1] ?? 0
   912→  const endDrop = endEnergy < peak * 0.7 ? 1 : 0
   913→  const midPeak = peakIdx > 0 && peakIdx < energies.length - 1 ? 1 : 0
   914→  return (endDrop + midPeak) / 2
   915→}

   929→function computeCallResponse(pcm: Float32Array, sampleRate: number, _bpm: number): number {
   930→  // Simplified: check if the second half mirrors the first (response).
   931→  const half = Math.floor(pcm.length / 2)

   936→    let a = 0
   937→    let b = 0
   938→    for (let j = 0; j < windowSize; j++) {
